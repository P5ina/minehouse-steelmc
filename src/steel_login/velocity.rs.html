17/// Forwarding version requested from the proxy: profile and address, no chat keys.
18pub const MODERN_DEFAULT: u8 = 1;
19
20const SIGNATURE_LEN: usize = 32;
21const HMAC_BLOCK: usize = 64;
22const MAX_ADDRESS_LEN: usize = 255;
23
24/// The channel identifier for the forwarding query.
25#[must_use]
26pub const fn channel() -> Identifier {
27    Identifier::new_static("velocity", "player_info")
28}

30/// The player data a proxy forwarded.
31#[derive(Debug)]
32pub struct ForwardedPlayer {
33    /// The player's real address, as seen by the proxy.
34    pub address: IpAddr,
35    /// The profile the proxy authenticated.
36    pub profile: GameProfile,
37}
38
39/// Why a forwarding answer was rejected.
40#[derive(Debug, PartialEq, Eq, thiserror::Error)]
41pub enum ForwardingError {
42    /// The client did not understand the query, so it is not a proxy.
43    #[error("the proxy did not answer the forwarding query")]
44    NoAnswer,
45    /// The answer was signed with another secret or altered.
46    #[error("forwarding signature does not match the configured secret")]
47    BadSignature,
48    /// The proxy answered with a forwarding version we do not read.
49    #[error("unsupported forwarding version {0}")]
50    UnsupportedVersion(i32),
51    /// The signed data could not be decoded.
52    #[error("malformed forwarding data")]
53    Malformed,
54}

56fn hmac_sha256(key: &[u8], message: &[u8]) -> [u8; SIGNATURE_LEN] {
57    let mut block = [0u8; HMAC_BLOCK];
58    if key.len() > HMAC_BLOCK {
59        block[..SIGNATURE_LEN].copy_from_slice(&Sha256::digest(key));
60    } else {
61        block[..key.len()].copy_from_slice(key);
62    }
63    let mut inner_pad = [0x36u8; HMAC_BLOCK];
64    let mut outer_pad = [0x5cu8; HMAC_BLOCK];
65    for ((inner, outer), key_byte) in inner_pad.iter_mut().zip(&mut outer_pad).zip(block) {
66        *inner ^= key_byte;
67        *outer ^= key_byte;
68    }
69    let inner = Sha256::new()
70        .chain_update(inner_pad)
71        .chain_update(message)
72        .finalize();
73    Sha256::new()
74        .chain_update(outer_pad)
75        .chain_update(inner)
76        .finalize()
77        .into()
78}

80/// Compares without an early exit so timing does not reveal the matching prefix.
81fn constant_time_eq(left: &[u8], right: &[u8]) -> bool {
82    left.len() == right.len()
83        && left
84            .iter()
85            .zip(right)
86            .fold(0u8, |diff, (a, b)| diff | (a ^ b))
87            == 0
88}

90/// Verifies and decodes a forwarding answer.
91///
92/// # Errors
93/// Returns an error if the answer is missing, unsigned, signed with another secret or malformed.
94pub fn read_forwarded_player(
95    secret: &[u8],
96    payload: Option<&[u8]>,
97) -> Result<ForwardedPlayer, ForwardingError> {
98    let payload = payload.ok_or(ForwardingError::NoAnswer)?;
99    if payload.len() < SIGNATURE_LEN {
100        return Err(ForwardingError::Malformed);
101    }
102    let (signature, data) = payload.split_at(SIGNATURE_LEN);
103    if !constant_time_eq(signature, &hmac_sha256(secret, data)) {
104        return Err(ForwardingError::BadSignature);
105    }
106
107    let cursor = &mut Cursor::new(data);
108    let version = VarInt::read(cursor)
109        .map_err(|_| ForwardingError::Malformed)?
110        .0;
111    // Newer versions only append chat-key data after the profile; we asked for the default.
112    if !(i32::from(MODERN_DEFAULT)..=4).contains(&version) {
113        return Err(ForwardingError::UnsupportedVersion(version));
114    }
115    let address = String::read_prefixed_bound::<VarInt>(cursor, MAX_ADDRESS_LEN * 3)
116        .map_err(|_| ForwardingError::Malformed)?;
117    let address = address
118        .parse::<IpAddr>()
119        .map_err(|_| ForwardingError::Malformed)?;
120    let profile = GameProfile::read(cursor).map_err(|_| ForwardingError::Malformed)?;
121    if version == i32::from(MODERN_DEFAULT) && cursor.read(&mut [0u8; 1]).unwrap_or(0) != 0 {
122        return Err(ForwardingError::Malformed);
123    }
124    Ok(ForwardedPlayer { address, profile })
125}

134    const SECRET: &[u8] = b"forwarding-secret";

136    fn body(version: i32, address: &str, id: Uuid, name: &str) -> Vec<u8> {
137        let mut data = Vec::new();
138        VarInt(version).write(&mut data).expect("write version");
139        address
140            .write_prefixed::<VarInt>(&mut data)
141            .expect("write address");
142        GameProfile::new(id, name.to_owned(), Vec::new())
143            .expect("valid profile")
144            .write(&mut data)
145            .expect("write profile");
146        data
147    }
148
149    fn signed(secret: &[u8], data: &[u8]) -> Vec<u8> {
150        let mut payload = hmac_sha256(secret, data).to_vec();
151        payload.extend_from_slice(data);
152        payload
153    }

155    #[test]
156    fn hmac_matches_rfc_4231_case_2() {
157        let mac = hmac_sha256(b"Jefe", b"what do ya want for nothing?");
158        assert_eq!(
159            hex::encode(mac),
160            "5bdcc146bf60754e6a042426089575c75a003f089d2739839dec58b964ec3843"
161        );
162    }
163
164    #[test]
165    fn accepts_a_correctly_signed_answer() {
166        let id = Uuid::from_u128(42);
167        let payload = signed(SECRET, &body(1, "203.0.113.7", id, "Steve"));
168
169        let player = read_forwarded_player(SECRET, Some(&payload)).expect("valid answer");
170        assert_eq!(player.address, "203.0.113.7".parse::<IpAddr>().expect("ip"));
171        assert_eq!(player.profile.id(), id);
172        assert_eq!(player.profile.name(), "Steve");
173    }
174
175    #[test]
176    fn rejects_another_secret() {
177        let payload = signed(b"other", &body(1, "203.0.113.7", Uuid::nil(), "Steve"));
178        assert_eq!(
179            read_forwarded_player(SECRET, Some(&payload)).expect_err("answer should be rejected"),
180            ForwardingError::BadSignature
181        );
182    }
183
184    #[test]
185    fn rejects_tampered_data() {
186        let mut payload = signed(SECRET, &body(1, "203.0.113.7", Uuid::nil(), "Steve"));
187        let last = payload.len() - 1;
188        payload[last] ^= 1;
189        assert_eq!(
190            read_forwarded_player(SECRET, Some(&payload)).expect_err("answer should be rejected"),
191            ForwardingError::BadSignature
192        );
193    }
194
195    #[test]
196    fn rejects_a_client_that_does_not_answer() {
197        assert_eq!(
198            read_forwarded_player(SECRET, None).expect_err("answer should be rejected"),
199            ForwardingError::NoAnswer
200        );
201    }

203    #[test]

206        assert_eq!(
207            read_forwarded_player(SECRET, Some(&payload)).expect_err("answer should be rejected"),
208            ForwardingError::UnsupportedVersion(9)
