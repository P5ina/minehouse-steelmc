1//! Import of vanilla Anvil worlds into Steel's region format.
2//!
3//! The importer reads a world saved by vanilla Minecraft of the same version Steel targets
4//! (`steel_utils::version::DATA_VERSION`) and writes Steel `.srg` region files. Steel has no
5//! `DataFixer`, so worlds from older versions must be upgraded by vanilla first
6//! (`java -jar server.jar --forceUpgrade`).
7//!
8//! What is converted per dimension:
9//! - `region/`: block states, biomes, heightmaps, light, block entities, scheduled block and
10//!   fluid ticks, postprocessing marks and structure references of `minecraft:full` chunks.
11//! - `entities/`: entities with their passengers.
12//! - `poi/`: free tickets of points of interest.
13//!
14//! Chunks that are not `minecraft:full` are skipped; Steel generates them again from the seed.
15//! Structure starts are converted when Steel can represent every piece of them; the others are
16//! dropped and counted in the report. Jigsaw and template-backed structures are converted so far.
17//!
18//! Content Steel does not know (blocks, biomes, block entities or entities from mods) cannot be
19//! represented in Steel's typed format. Unknown blocks become air, unknown biomes become
20//! `minecraft:plains`, and unknown block entities and entities are dropped. Every replacement is
21//! counted by id in the [`ImportReport`], so an import can be checked before it is used.

56/// Errors that stop an import.
57#[derive(Debug, thiserror::Error)]
58pub enum AnvilImportError {
59    /// Reading or writing a file failed.
60    #[error("{}: {source}", path.display())]
61    Io {
62        /// File or directory involved.
63        path: PathBuf,
64        /// Underlying error.
65        source: io::Error,
66    },
67    /// A region or `.dat` file is malformed.
68    #[error("{}: {reason}", path.display())]
69    InvalidFile {
70        /// File that failed to parse.
71        path: PathBuf,
72        /// What is wrong.
73        reason: String,
74    },
75    /// The Steel target already contains region files.
76    #[error("{} already contains Steel region files; import into an empty world", path.display())]
77    TargetNotEmpty {
78        /// Target region directory.
79        path: PathBuf,
80    },
81    /// The Steel world already has level data.
82    #[error("{} already exists; import into an empty world", path.display())]
83    LevelDataExists {
84        /// Existing `level.toml`.
85        path: PathBuf,
86    },
87    /// A player already has Steel data in the target domain.
88    #[error("player {uuid} already has data in domain {domain}; import into an empty world")]
89    PlayerDataExists {
90        /// Player UUID.
91        uuid: uuid::Uuid,
92        /// Target domain.
93        domain: String,
94    },
95}

97/// One vanilla dimension to import into one Steel world.
98#[derive(Debug, Clone)]
99pub struct DimensionImport {
100    /// Dimension id, for example `minecraft:overworld`.
101    pub key: Identifier,
102    /// Vanilla dimension directory holding `region/`, `entities/` and `poi/`,
103    /// for example `world/dimensions/minecraft/overworld`.
104    pub source: PathBuf,
105    /// Steel region directory, for example `saves/minecraft/worlds/overworld/region`.
106    pub target: PathBuf,
107    /// Minimum build height of the dimension type.
108    pub min_y: i32,
109    /// Build height of the dimension type.
110    pub height: i32,
111}

113/// Imports one dimension.
114///
115/// Chunk conversion runs on `thread_pool` because a whole world is gigabytes of NBT.
116pub async fn import_dimension(
117    dimension: &DimensionImport,
118    thread_pool: Arc<rayon::ThreadPool>,
119) -> Result<ImportReport, AnvilImportError> {
120    ensure_empty_target(&dimension.target)?;
121
122    let regions = region::list_region_files(&dimension.source.join("region"))?;
123    let manager = RegionManager::new(&dimension.target);
124    let mut report = ImportReport::default();
125
126    for region_path in regions {
127        let task_dimension = dimension.clone();
128        let task_path = region_path.clone();
129        let pool = Arc::clone(&thread_pool);
130        let (chunks, region_report) =
131            spawn_blocking(move || convert_region(&task_dimension, &task_path, &pool))
132                .await
133                .map_err(|error| AnvilImportError::Io {
134                    path: region_path,
135                    source: io::Error::other(error.to_string()),
136                })??;
137        report.merge(region_report);
138
139        let saves = chunks.into_iter().map(|(pos, persistent)| {
140            let prepared = PreparedChunkSave {
141                pos,
142                status: ChunkStatus::Full,
143                persistent,
144                handled_runtime_entity_ids: Vec::new(),
145            };
146            manager.save_chunk_data(prepared, &thread_pool)
147        });
148        try_join_all(saves)
149            .await
150            .map_err(|source| AnvilImportError::Io {
151                path: dimension.target.clone(),
152                source,
153            })?;
154        report.regions += 1;
155    }
156
157    manager
158        .close_all()
159        .await
160        .map_err(|source| AnvilImportError::Io {
161            path: dimension.target.clone(),
162            source,
163        })?;
164    Ok(report)
165}

167fn ensure_empty_target(target: &Path) -> Result<(), AnvilImportError> {
168    let entries = match fs::read_dir(target) {
169        Ok(entries) => entries,
170        Err(error) if error.kind() == io::ErrorKind::NotFound => return Ok(()),
171        Err(source) => {
172            return Err(AnvilImportError::Io {
173                path: target.to_path_buf(),
174                source,
175            });
176        }
177    };
178    for entry in entries.flatten() {
179        if entry.path().extension().is_some_and(|ext| ext == "srg") {
180            return Err(AnvilImportError::TargetNotEmpty {
181                path: target.to_path_buf(),
182            });
183        }
184    }
185    Ok(())
186}

188/// Converts every chunk of one region file.
189fn convert_region(
190    dimension: &DimensionImport,
191    region_path: &Path,
192    thread_pool: &rayon::ThreadPool,
193) -> Result<(Vec<(ChunkPos, PersistentChunk<'static>)>, ImportReport), AnvilImportError> {
194    use rayon::prelude::*;
195
196    let region = AnvilRegion::open(region_path)?;
197    let file_name = region_path.file_name().map_or_default(PathBuf::from);
198    let entities = open_optional_region(&dimension.source.join("entities").join(&file_name))?;
199    let pois = open_optional_region(&dimension.source.join("poi").join(&file_name))?;
200
201    let results: Vec<_> = thread_pool.install(|| {
202        region
203            .chunk_positions()
204            .into_par_iter()
205            .map(|pos| {
206                let mut report = ImportReport::default();
207                let chunk = convert_one(
208                    &region,
209                    entities.as_ref(),
210                    pois.as_ref(),
211                    pos,
212                    dimension,
213                    &mut report,
214                );
215                (pos, chunk, report)
216            })
217            .collect()
218    });
219
220    let mut chunks = Vec::new();
221    let mut report = ImportReport::default();
222    for (pos, chunk, chunk_report) in results {
223        report.merge(chunk_report);
224        match chunk {
225            Ok(Some(persistent)) => chunks.push((pos, persistent)),
226            Ok(None) => {}
227            Err(reason) => report.failed_chunks.push((pos, reason)),
228        }
229    }
230    Ok((chunks, report))
231}

233fn convert_one(
234    region: &AnvilRegion,
235    entities: Option<&AnvilRegion>,
236    poi_region: Option<&AnvilRegion>,
237    pos: ChunkPos,
238    dimension: &DimensionImport,
239    report: &mut ImportReport,
240) -> Result<Option<PersistentChunk<'static>>, String> {
241    let Some(chunk_bytes) = region.read_chunk(pos).map_err(|error| error.to_string())? else {
242        return Ok(None);
243    };
244    let entity_bytes = match entities {
245        Some(entities) => entities
246            .read_chunk(pos)
247            .map_err(|error| error.to_string())?,
248        None => None,
249    };
250    let poi_bytes = match poi_region {
251        Some(region) => region.read_chunk(pos).map_err(|error| error.to_string())?,
252        None => None,
253    };
254    match convert_chunk(
255        &chunk_bytes,
256        entity_bytes.as_deref(),
257        poi_bytes.as_deref(),
258        pos,
259        dimension.min_y,
260        dimension.height,
261        report,
262    )? {
263        ChunkConversion::Imported(persistent) => {
264            report.chunks_imported += 1;
265            Ok(Some(*persistent))
266        }
267        ChunkConversion::NotFull(status) => {
268            *report.skipped_statuses.entry(status).or_default() += 1;
269            Ok(None)
270        }
271    }
272}

274fn open_optional_region(path: &Path) -> Result<Option<AnvilRegion>, AnvilImportError> {
275    if path.exists() {
276        AnvilRegion::open(path).map(Some)
277    } else {
278        Ok(None)
279    }
280}

282/// The vanilla dimensions of a 26.x world save and the Steel worlds they import into.
283///
284/// Vanilla keeps each dimension in `dimensions/<namespace>/<name>`; Steel's default world
285/// layout is `<save_path>/<domain>/worlds/<name>/region`.
286#[must_use]
287pub fn vanilla_dimensions(world: &Path, save_path: &Path, domain: &str) -> Vec<DimensionImport> {
288    use steel_registry::vanilla_dimension_types::{OVERWORLD, THE_END, THE_NETHER};
289
290    [&OVERWORLD, &THE_NETHER, &THE_END]
291        .into_iter()
292        .map(|dimension_type| {
293            let name = dimension_type.key.path.as_ref();
294            DimensionImport {
295                key: dimension_type.key.clone(),
296                source: world.join("dimensions").join("minecraft").join(name),
297                target: save_path
298                    .join(domain)
299                    .join("worlds")
300                    .join(name)
301                    .join("region"),
302                min_y: dimension_type.min_y,
303                height: dimension_type.height,
304            }
305        })
306        .collect()
307}

309/// Reads the world seed from `data/minecraft/world_gen_settings.dat`.
310pub fn read_world_seed(world: &Path) -> Result<i64, AnvilImportError> {
311    let path = world
312        .join("data")
313        .join("minecraft")
314        .join("world_gen_settings.dat");
315    let invalid = |reason: String| AnvilImportError::InvalidFile {
316        path: path.clone(),
317        reason,
318    };
319    let root = level::read_gzip_root(&path)?;
320    chunk::check_data_version(&root).map_err(invalid)?;
321    root.compound("data")
322        .and_then(|data| data.long("seed"))
323        .ok_or_else(|| invalid("missing data.seed".to_owned()))
324}
