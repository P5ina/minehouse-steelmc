1//! Conversion of vanilla structure starts (`StructureStart.createTag`) into Steel's
2//! `PersistentStructureStart`.
3//!
4//! A start is converted only when every one of its pieces is. A start with a piece Steel cannot
5//! represent, or with data that does not parse, is dropped whole and counted by structure id:
6//! a start with missing pieces would generate a different structure than vanilla did.
7

22/// Vanilla `StructureStart.INVALID_START_ID`.
23const INVALID_START: &str = "INVALID";
24/// Vanilla `StructurePieceType.JIGSAW`.
25const JIGSAW_PIECE: &str = "minecraft:jigsaw";
26/// NBT list type id of an empty list.
27const END_TAG: u8 = 0;
28
29/// Converts the structure starts of a chunk's `structures` compound.
30pub(super) fn structure_starts(
31    structures: &NbtCompound,
32    report: &mut ImportReport,
33) -> Vec<PersistentStructureStart> {
34    let Some(starts) = structures.compound("starts") else {
35        return Vec::new();
36    };
37    let mut converted = Vec::new();
38    for (structure, start) in starts.iter() {
39        let Some(start) = start.compound() else {
40            continue;
41        };
42        if nbt::string(start, "id").is_none_or(|id| id == INVALID_START) {
43            continue;
44        }
45        let structure = structure.to_str();
46        match structure_start(&structure, start) {
47            Ok(start) => converted.push(start),
48            Err(_) => {
49                ImportReport::count(&mut report.structure_starts_dropped, structure.into_owned());
50            }
51        }
52    }
53    converted
54}

56/// Converts one start, vanilla `StructureStart.loadStaticStart`.
57fn structure_start(
58    structure: &str,
59    start: &NbtCompound,
60) -> Result<PersistentStructureStart, String> {
61    let structure = nbt::identifier(structure).ok_or("malformed structure id")?;
62    if REGISTRY.structures.by_key(&structure).is_none() {
63        return Err(format!("unknown structure {structure}"));
64    }
65    let pieces = compound_list(start, "Children")?
66        .iter()
67        .map(structure_piece)
68        .collect::<Result<Vec<_>, _>>()?;
69    if pieces.is_empty() {
70        return Err("structure start has no pieces".to_owned());
71    }
72    Ok(PersistentStructureStart {
73        structure,
74        chunk_x: int(start, "ChunkX")?,
75        chunk_z: int(start, "ChunkZ")?,
76        references: int(start, "references")?,
77        pieces,
78    })
79}

81/// Converts one piece. The common fields are vanilla `StructurePiece(type, tag)`.
82fn structure_piece(piece: &NbtCompound) -> Result<PersistentStructurePiece, String> {
83    let piece_type = nbt::string(piece, "id").ok_or("piece without id")?;
84    let bounding_box = bounding_box(piece)?;
85    let gen_depth = int(piece, "GD")?;
86    let orientation = int(piece, "O")?;
87    if !(-1..=3).contains(&orientation) {
88        return Err(format!("invalid piece orientation {orientation}"));
89    }
90    let orientation = orientation as i8;
91    if piece_type == JIGSAW_PIECE {
92        return jigsaw_piece(piece, bounding_box, gen_depth, orientation);
93    }
94    let (payload, bounding_box) = structure_templates::template_payload(&piece_type, piece)
95        .ok_or_else(|| format!("unsupported structure piece {piece_type}"))??;
96    let piece_type = nbt::identifier(&piece_type).ok_or("malformed piece id")?;
97    Ok(PersistentStructurePiece {
98        piece_type,
99        bounding_box,
100        gen_depth,
101        orientation,
102        payload,
103        // Template pieces have no terrain adaptation data.
104        ground_level_delta: 0,
105        projection: -1,
106        junctions: Vec::new(),
107    })
108}

110/// Vanilla `PoolElementStructurePiece(StructurePieceSerializationContext, CompoundTag)`.
111fn jigsaw_piece(
112    piece: &NbtCompound,
113    bounding_box: PersistentBoundingBox,
114    gen_depth: i32,
115    orientation: i8,
116) -> Result<PersistentStructurePiece, String> {
117    let element = piece
118        .compound("pool_element")
119        .ok_or("jigsaw piece without pool_element")?;
120    let (pool_element, projection) = pool_element(element)?;
121    let rotation = match nbt::string(piece, "rotation").as_deref() {
122        Some("NONE") => 0,
123        Some("CLOCKWISE_90") => 1,
124        Some("CLOCKWISE_180") => 2,
125        Some("COUNTERCLOCKWISE_90") => 3,
126        other => return Err(format!("invalid jigsaw rotation {other:?}")),
127    };
128    // `LiquidSettings.CODEC` falls back to `APPLY_WATERLOGGING` when the field is absent.
129    let liquid_settings = match nbt::string(piece, "liquid_settings").as_deref() {
130        None | Some("apply_waterlogging") => 0,
131        Some("ignore_waterlogging") => 1,
132        Some(other) => return Err(format!("invalid liquid settings {other}")),
133    };
134    let junctions = compound_list(piece, "junctions")?
135        .iter()
136        .map(junction)
137        .collect::<Result<Vec<_>, _>>()?;
138    Ok(PersistentStructurePiece {
139        piece_type: Identifier::vanilla_static("jigsaw"),
140        bounding_box,
141        gen_depth,
142        orientation,
143        payload: PersistentStructurePiecePayload::Jigsaw(PersistentJigsawPieceData {
144            pool_element,
145            position: [
146                int(piece, "PosX")?,
147                int(piece, "PosY")?,
148                int(piece, "PosZ")?,
149            ],
150            rotation,
151            liquid_settings,
152        }),
153        ground_level_delta: int(piece, "ground_level_delta")?,
154        projection,
155        junctions,
156    })
157}

159/// Converts a pool element encoded with `StructurePoolElement.CODEC`, returning it with its
160/// projection (`StructurePoolElement.getProjection`).
161///
162/// Elements are checked against what Steel can place: a template that is not bundled, or a
163/// processor list or placed feature Steel does not know, would fail when the piece is placed.
164fn pool_element(element: &NbtCompound) -> Result<(PersistentPoolElement, i8), String> {
165    let element_type = nbt::string(element, "element_type").ok_or("pool element without type")?;
166    if element_type == "minecraft:empty_pool_element" {
167        return Ok((PersistentPoolElement::Empty, 0));
168    }
169    let projection = match nbt::string(element, "projection").as_deref() {
170        Some("rigid") => 0,
171        Some("terrain_matching") => 1,
172        other => return Err(format!("invalid pool element projection {other:?}")),
173    };
174    let converted = match element_type.as_str() {
175        "minecraft:single_pool_element" => {
176            let (location, processors) = single_element(element)?;
177            PersistentPoolElement::Single {
178                location,
179                processors,
180                projection,
181            }
182        }
183        "minecraft:legacy_single_pool_element" => {
184            let (location, processors) = single_element(element)?;
185            PersistentPoolElement::LegacySingle {
186                location,
187                processors,
188                projection,
189            }
190        }
191        "minecraft:feature_pool_element" => {
192            let feature = nbt::string(element, "feature")
193                .and_then(|feature| nbt::identifier(&feature))
194                .ok_or("feature pool element without a feature id")?;
195            if REGISTRY.placed_features.by_key(&feature).is_none() {
196                return Err(format!("unknown placed feature {feature}"));
197            }
198            PersistentPoolElement::Feature {
199                feature,
200                projection,
201            }
202        }
203        "minecraft:list_pool_element" => {
204            let elements = compound_list(element, "elements")?
205                .iter()
206                .map(|element| pool_element(element).map(|(element, _)| element))
207                .collect::<Result<Vec<_>, _>>()?;
208            PersistentPoolElement::List {
209                elements,
210                projection,
211            }
212        }
213        other => return Err(format!("unsupported pool element {other}")),
214    };
215    Ok((converted, projection))
216}

218/// Template location and processors of a (legacy) single pool element.
219fn single_element(element: &NbtCompound) -> Result<(Identifier, PersistentProcessorList), String> {
220    let location = nbt::string(element, "location")
221        .and_then(|location| nbt::identifier(&location))
222        .ok_or("single pool element without a template id")?;
223    if vanilla_template_nbt_bytes(&location).is_none() {
224        return Err(format!("structure template {location} is not bundled"));
225    }
226    // `StructureProcessorType.LIST_CODEC` writes a registered list by id and a direct list as
227    // `{processors: [...]}`. Steel only represents the empty direct list.
228    let processors = if let Some(id) = nbt::string(element, "processors") {
229        let id = nbt::identifier(&id).ok_or("malformed processor list id")?;
230        if REGISTRY.structure_processors.by_key(&id).is_none() {
231            return Err(format!("unknown processor list {id}"));
232        }
233        PersistentProcessorList::Registry(id)
234    } else {
235        let direct = element
236            .compound("processors")
237            .ok_or("single pool element without processors")?;
238        if !compound_list(direct, "processors")?.is_empty() {
239            return Err("direct processor lists are not supported".to_owned());
240        }
241        PersistentProcessorList::Empty
242    };
243    Ok((location, processors))
244}

246/// Vanilla `JigsawJunction.deserialize`.
247fn junction(junction: &NbtCompound) -> Result<PersistentJigsawJunction, String> {
248    let dest_projection = match nbt::string(junction, "dest_proj").as_deref() {
249        Some("rigid") => 0,
250        Some("terrain_matching") => 1,
251        other => return Err(format!("invalid junction projection {other:?}")),
252    };
253    Ok(PersistentJigsawJunction {
254        source_x: int(junction, "source_x")?,
255        source_ground_y: int(junction, "source_ground_y")?,
256        source_z: int(junction, "source_z")?,
257        delta_y: int(junction, "delta_y")?,
258        dest_projection,
259    })
260}

262/// Vanilla `BoundingBox.CODEC`: `[minX, minY, minZ, maxX, maxY, maxZ]`.
263fn bounding_box(piece: &NbtCompound) -> Result<PersistentBoundingBox, String> {
264    let Some(&[min_x, min_y, min_z, max_x, max_y, max_z]) = piece.int_array("BB") else {
265        return Err("piece without a bounding box".to_owned());
266    };
267    if min_x > max_x || min_y > max_y || min_z > max_z {
268        return Err("inverted piece bounding box".to_owned());
269    }
270    Ok(PersistentBoundingBox {
271        min_x,
272        min_y,
273        min_z,
274        max_x,
275        max_y,
276        max_z,
277    })
278}

280fn int(compound: &NbtCompound, key: &str) -> Result<i32, String> {
281    compound.int(key).ok_or_else(|| format!("missing {key}"))
282}

284/// Reads a list of compounds, where an empty list may have the end tag as its element type.
285fn compound_list<'a>(compound: &'a NbtCompound, key: &str) -> Result<&'a [NbtCompound], String> {
286    let list: &NbtList = compound.list(key).ok_or_else(|| format!("missing {key}"))?;
287    if list.id() == END_TAG {
288        return Ok(&[]);
289    }
290    list.compounds()
291        .ok_or_else(|| format!("{key} is not a list of compounds"))
292}
