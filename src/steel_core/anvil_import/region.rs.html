1//! Reader for vanilla Anvil region files (`r.<x>.<z>.mca`).
2//!
3//! A region file starts with a 4 KiB table of 1024 big-endian location entries
4//! (3-byte sector offset + 1-byte sector count) followed by a 4 KiB timestamp table.
5//! Each chunk payload is `length: u32 BE`, `compression: u8`, then `length - 1` bytes.
6//! When bit 7 of the compression byte is set, the payload lives in an external
7//! `c.<x>.<z>.mcc` file next to the region file.
8
9use std::io::Read;
10use std::path::{Path, PathBuf};
11use std::{fs, io};
12
13use flate2::read::{GzDecoder, ZlibDecoder};
14use steel_utils::ChunkPos;
15
16use super::AnvilImportError;
17
18const SECTOR_SIZE: usize = 4096;
19const HEADER_SIZE: usize = SECTOR_SIZE * 2;
20const CHUNKS_PER_REGION: usize = 1024;
21const EXTERNAL_FLAG: u8 = 0x80;
22
23const COMPRESSION_GZIP: u8 = 1;
24const COMPRESSION_ZLIB: u8 = 2;
25const COMPRESSION_NONE: u8 = 3;
26const COMPRESSION_LZ4: u8 = 4;
27const COMPRESSION_CUSTOM: u8 = 127;
28
29/// One region file loaded fully into memory.
30pub struct AnvilRegion {
31    path: PathBuf,
32    region_x: i32,
33    region_z: i32,
34    data: Vec<u8>,
35}
36
37impl AnvilRegion {
38    /// Reads a region file. The region coordinates come from the file name.
39    pub fn open(path: &Path) -> Result<Self, AnvilImportError> {
40        let (region_x, region_z) =
41            parse_region_file_name(path).ok_or_else(|| AnvilImportError::InvalidFile {
42                path: path.to_path_buf(),
43                reason: "file name is not r.<x>.<z>.mca".to_owned(),
44            })?;
45        let data = fs::read(path).map_err(|source| AnvilImportError::Io {
46            path: path.to_path_buf(),
47            source,
48        })?;
49        Ok(Self {
50            path: path.to_path_buf(),
51            region_x,
52            region_z,
53            data,
54        })
55    }
56
57    /// Region X coordinate.
58    #[must_use]
59    pub const fn region_x(&self) -> i32 {
60        self.region_x
61    }
62
63    /// Region Z coordinate.
64    #[must_use]
65    pub const fn region_z(&self) -> i32 {
66        self.region_z
67    }
68
69    /// Chunk positions that have a payload in this region.
70    #[must_use]
71    pub fn chunk_positions(&self) -> Vec<ChunkPos> {
72        if self.data.len() < HEADER_SIZE {
73            return Vec::new();
74        }
75        (0..CHUNKS_PER_REGION)
76            .filter(|&index| self.location(index) != (0, 0))
77            .map(|index| {
78                ChunkPos::new(
79                    self.region_x * 32 + (index % 32) as i32,
80                    self.region_z * 32 + (index / 32) as i32,
81                )
82            })
83            .collect()
84    }
85
86    fn location(&self, index: usize) -> (usize, usize) {
87        let entry = &self.data[index * 4..index * 4 + 4];
88        let offset =
89            (usize::from(entry[0]) << 16) | (usize::from(entry[1]) << 8) | usize::from(entry[2]);
90        (offset, usize::from(entry[3]))
91    }
92
93    /// Returns the decompressed NBT bytes of a chunk, or `None` if it has no payload.
94    pub fn read_chunk(&self, pos: ChunkPos) -> Result<Option<Vec<u8>>, AnvilImportError> {
95        if self.data.len() < HEADER_SIZE {
96            return Ok(None);
97        }
98        let local_x = pos.0.x.rem_euclid(32) as usize;
99        let local_z = pos.0.y.rem_euclid(32) as usize;
100        let (sector, sector_count) = self.location(local_x + local_z * 32);
101        if sector == 0 && sector_count == 0 {
102            return Ok(None);
103        }
104
105        let start = sector * SECTOR_SIZE;
106        let Some(header) = self.data.get(start..start + 5) else {
107            return Err(self.invalid(pos, "chunk offset points past the end of the file"));
108        };
109        let length = u32::from_be_bytes([header[0], header[1], header[2], header[3]]) as usize;
110        if length == 0 {
111            return Err(self.invalid(pos, "chunk payload length is zero"));
112        }
113        let compression = header[4];
114
115        if compression & EXTERNAL_FLAG != 0 {
116            let external = self
117                .path
118                .with_file_name(format!("c.{}.{}.mcc", pos.0.x, pos.0.y));
119            let compressed = fs::read(&external).map_err(|source| AnvilImportError::Io {
120                path: external,
121                source,
122            })?;
123            return self
124                .decompress(pos, compression & !EXTERNAL_FLAG, &compressed)
125                .map(Some);
126        }
127
128        let Some(payload) = self.data.get(start + 5..start + 4 + length) else {
129            return Err(self.invalid(pos, "chunk payload runs past the end of the file"));
130        };
131        self.decompress(pos, compression, payload).map(Some)
132    }
133
134    fn decompress(
135        &self,
136        pos: ChunkPos,
137        compression: u8,
138        payload: &[u8],
139    ) -> Result<Vec<u8>, AnvilImportError> {
140        let mut out = Vec::new();
141        let result = match compression {
142            COMPRESSION_GZIP => GzDecoder::new(payload).read_to_end(&mut out),
143            COMPRESSION_ZLIB => ZlibDecoder::new(payload).read_to_end(&mut out),
144            COMPRESSION_NONE => {
145                out.extend_from_slice(payload);
146                Ok(out.len())
147            }
148            COMPRESSION_LZ4 => {
149                return Err(self.invalid(
150                    pos,
151                    "LZ4 region compression is not supported; set region-file-compression=deflate \
152                     in server.properties, then load and save the world with vanilla once",
153                ));
154            }
155            COMPRESSION_CUSTOM => {
156                return Err(self.invalid(pos, "custom region compression is not supported"));
157            }
158            other => {
159                return Err(self.invalid(pos, &format!("unknown compression type {other}")));
160            }
161        };
162        result.map_err(|source: io::Error| AnvilImportError::InvalidFile {
163            path: self.path.clone(),
164            reason: format!("chunk {pos:?} failed to decompress: {source}"),
165        })?;
166        Ok(out)
167    }
168
169    fn invalid(&self, pos: ChunkPos, reason: &str) -> AnvilImportError {
170        AnvilImportError::InvalidFile {
171            path: self.path.clone(),
172            reason: format!("chunk {pos:?}: {reason}"),
173        }
174    }
175}

177/// Parses `r.<x>.<z>.mca` into region coordinates.
178#[must_use]
179pub fn parse_region_file_name(path: &Path) -> Option<(i32, i32)> {
180    let name = path.file_name()?.to_str()?;
181    let rest = name.strip_prefix("r.")?.strip_suffix(".mca")?;
182    let (x, z) = rest.split_once('.')?;
183    Some((x.parse().ok()?, z.parse().ok()?))
184}

186/// Lists region files in a directory, sorted by coordinates. A missing directory is empty.
187pub fn list_region_files(dir: &Path) -> Result<Vec<PathBuf>, AnvilImportError> {
188    let entries = match fs::read_dir(dir) {
189        Ok(entries) => entries,
190        Err(error) if error.kind() == io::ErrorKind::NotFound => return Ok(Vec::new()),
191        Err(source) => {
192            return Err(AnvilImportError::Io {
193                path: dir.to_path_buf(),
194                source,
195            });
196        }
197    };
198    let mut files = Vec::new();
199    for entry in entries {
200        let entry = entry.map_err(|source| AnvilImportError::Io {
201            path: dir.to_path_buf(),
202            source,
203        })?;
204        let path = entry.path();
205        if let Some(coords) = parse_region_file_name(&path) {
206            files.push((coords, path));
207        }
208    }
209    files.sort_by_key(|(coords, _)| *coords);
210    Ok(files.into_iter().map(|(_, path)| path).collect())
211}
