53/// What the player import wrote and what it could not carry over.
54#[derive(Debug, Default)]
55pub struct PlayerReport {
56    /// Players written to Steel's storage.
57    pub players: usize,
58    /// Players that could not be read, with the reason.
59    pub failed_players: Vec<(PathBuf, String)>,
60    /// Items Steel does not know, dropped from inventories and ender chests.
61    pub unknown_items: BTreeMap<String, usize>,
62    /// Stats Steel does not know.
63    pub unknown_stats: BTreeMap<String, usize>,
64    /// Advancements Steel does not know.
65    pub unknown_advancements: BTreeMap<String, usize>,
66    /// Vehicles and ender pearls dropped because Steel does not know their type.
67    pub unknown_entities: BTreeMap<String, usize>,
68    /// Active effects dropped because Steel does not save player effects yet, by effect id.
69    pub dropped_effects: BTreeMap<String, usize>,
70    /// Players whose recipe book was dropped because Steel has no recipe book yet.
71    pub dropped_recipe_books: usize,
72}
73
74impl PlayerReport {
75    fn merge_entities(&mut self, entities: ImportReport) {
76        for (id, count) in entities.unknown_entities {
77            *self.unknown_entities.entry(id).or_default() += count;
78        }
79    }
80}

82impl fmt::Display for PlayerReport {
83    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
84        writeln!(f, "players: {}", self.players)?;
85        for (title, map) in [
86            ("Unknown items dropped", &self.unknown_items),
87            ("Unknown stats dropped", &self.unknown_stats),
88            ("Unknown advancements dropped", &self.unknown_advancements),
89            (
90                "Unknown vehicles and ender pearls dropped",
91                &self.unknown_entities,
92            ),
93            (
94                "Active effects dropped (Steel does not save them yet)",
95                &self.dropped_effects,
96            ),
97        ] {
98            if !map.is_empty() {
99                writeln!(f, "{title}:")?;
100                for (key, count) in map {
101                    writeln!(f, "  {key}: {count}")?;
102                }
103            }
104        }
105        if self.dropped_recipe_books > 0 {
106            writeln!(
107                f,
108                "Recipe books dropped (Steel has no recipe book yet): {}",
109                self.dropped_recipe_books
110            )?;
111        }
112        if !self.failed_players.is_empty() {
113            writeln!(f, "Failed players:")?;
114            for (path, reason) in &self.failed_players {
115                writeln!(f, "  {}: {reason}", path.display())?;
116            }
117        }
118        Ok(())
119    }
120}

122/// Imports every player of a vanilla save into Steel's player storage under `save_path`.
123///
124/// Fails without writing anything if one of the players already has Steel data in `domain`.
125pub async fn import_players(
126    world: &Path,
127    save_path: &Path,
128    domain: &str,
129) -> Result<PlayerReport, AnvilImportError> {
130    let players_dir = world.join("players");
131    let player_files = list_player_files(&players_dir.join("data"))?;
132    let storage = PlayerDataStorage::on_disk(save_path.to_path_buf())
133        .await
134        .map_err(|source| AnvilImportError::Io {
135            path: save_path.to_path_buf(),
136            source,
137        })?;
138
139    for (uuid, _) in &player_files {
140        let existing =
141            storage
142                .load_domain(domain, *uuid)
143                .await
144                .map_err(|source| AnvilImportError::Io {
145                    path: save_path.to_path_buf(),
146                    source,
147                })?;
148        if existing.is_some() {
149            return Err(AnvilImportError::PlayerDataExists {
150                uuid: *uuid,
151                domain: domain.to_owned(),
152            });
153        }
154    }
155
156    let mut report = PlayerReport::default();
157    for (uuid, path) in player_files {
158        let data = match convert_player(&players_dir, uuid, &path, &mut report) {
159            Ok(data) => data,
160            Err(reason) => {
161                report.failed_players.push((path, reason));
162                continue;
163            }
164        };
165        let io_error = |source| AnvilImportError::Io {
166            path: save_path.to_path_buf(),
167            source,
168        };
169        storage
170            .save_domain_data(domain, uuid, &data)
171            .await
172            .map_err(io_error)?;
173        storage
174            .save_global(
175                uuid,
176                &GlobalPlayerData {
177                    last_active_domain: domain.to_owned(),
178                },
179            )
180            .await
181            .map_err(io_error)?;
182        report.players += 1;
183    }
184    Ok(report)
185}

187/// Lists `<uuid>.dat` files, sorted by UUID. A missing directory has no players.
188fn list_player_files(dir: &Path) -> Result<Vec<(Uuid, PathBuf)>, AnvilImportError> {
189    let entries = match fs::read_dir(dir) {
190        Ok(entries) => entries,
191        Err(error) if error.kind() == io::ErrorKind::NotFound => return Ok(Vec::new()),
192        Err(source) => {
193            return Err(AnvilImportError::Io {
194                path: dir.to_path_buf(),
195                source,
196            });
197        }
198    };
199    let mut files = Vec::new();
200    for entry in entries {
201        let path = entry
202            .map_err(|source| AnvilImportError::Io {
203                path: dir.to_path_buf(),
204                source,
205            })?
206            .path();
207        let uuid = path
208            .file_name()
209            .and_then(|name| name.to_str())
210            .and_then(|name| name.strip_suffix(".dat"))
211            .and_then(|stem| Uuid::parse_str(stem).ok());
212        if let Some(uuid) = uuid {
213            files.push((uuid, path));
214        }
215    }
216    files.sort();
217    Ok(files)
218}

220/// Converts one player. Missing stats or advancements files mean the player has none.
221pub(super) fn convert_player(
222    players_dir: &Path,
223    uuid: Uuid,
224    data_path: &Path,
225    report: &mut PlayerReport,
226) -> Result<PersistentPlayerData, String> {
227    let root = read_gzip_root(data_path).map_err(|error| error.to_string())?;
228    chunk::check_data_version(&root)?;
229
230    let mut data = convert_player_data(&root, report)?;
231
232    let stats_path = players_dir.join("stats").join(format!("{uuid}.json"));
233    if let Some(stats) = read_json(&stats_path)? {
234        data.stats = convert_stats(&stats, report)?;
235    }
236    let advancements_path = players_dir
237        .join("advancements")
238        .join(format!("{uuid}.json"));
239    if let Some(advancements) = read_json(&advancements_path)? {
240        data.advancements = convert_advancements(&advancements, report)?;
241    }
242    Ok(data)
243}

245/// Converts the `ServerPlayer.addAdditionalSaveData` compound.
246fn convert_player_data(
247    root: &NbtCompound,
248    report: &mut PlayerReport,
249) -> Result<PersistentPlayerData, String> {
250    let pos = nbt::doubles::<3>(root, "Pos").ok_or("missing Pos")?;
251    let world = nbt::string(root, "Dimension").unwrap_or_else(|| "minecraft:overworld".to_owned());
252
253    let mut inventory = items(root.list("Inventory"), report);
254    if let Some(equipment) = root.compound("equipment") {
255        for (key, slot) in EQUIPMENT_SLOTS {
256            if let Some(item) = equipment
257                .compound(key)
258                .and_then(|item| item_stack(item, report))
259            {
260                inventory.push(PersistentSlot { slot, item });
261            }
262        }
263    }
264
265    for effect in root
266        .list("active_effects")
267        .and_then(|list| list.compounds())
268        .unwrap_or_default()
269    {
270        let id = nbt::string(effect, "id").unwrap_or_else(|| "<missing id>".to_owned());
271        ImportReport::count(&mut report.dropped_effects, id);
272    }
273    if root.contains("recipeBook") {
274        report.dropped_recipe_books += 1;
275    }
276
277    let mut entities = ImportReport::default();
278    let root_vehicle = root.compound("RootVehicle").and_then(|vehicle| {
279        let attach = entity_uuid(vehicle, "Attach")?;
280        let entity = convert_entity(vehicle.compound("Entity")?, &mut entities)?;
281        Some(PersistentRootVehicle { attach, entity })
282    });
283    let ender_pearls = root
284        .list("ender_pearls")
285        .and_then(|list| list.compounds())
286        .unwrap_or_default()
287        .iter()
288        .filter_map(|pearl| {
289            let world = nbt::string(pearl, "ender_pearl_dimension")?;
290            let pearl = nbt::without_keys(pearl, &["ender_pearl_dimension"]);
291            let entity = convert_entity(&pearl, &mut entities)?;
292            Some(PersistentEnderPearl { world, entity })
293        })
294        .collect();
295    report.merge_entities(entities);
296
297    let abilities = root.compound("abilities");
298    let ability = |key: &str| abilities.is_some_and(|abilities| nbt::boolean(abilities, key));
299    let ability_float = |key: &str, default: f32| {
300        abilities
301            .and_then(|abilities| abilities.float(key))
302            .unwrap_or(default)
303    };
304
305    Ok(PersistentPlayerData {
306        pos,
307        motion: nbt::doubles::<3>(root, "Motion").unwrap_or_default(),
308        rotation: nbt::floats::<2>(root, "Rotation").unwrap_or_default(),
309        on_ground: nbt::boolean(root, "OnGround"),
310        fall_flying: nbt::boolean(root, "FallFlying"),
311        remaining_fire_ticks: i32::from(root.short("Fire").unwrap_or_default()),
312        ticks_frozen: root.int("TicksFrozen").unwrap_or_default(),
313        // Powder snow contact is recomputed every tick and is not saved by vanilla.
314        is_in_powder_snow: false,
315        was_in_powder_snow: false,
316        has_visual_fire: nbt::boolean(root, "HasVisualFire"),
317        health: root.float("Health").unwrap_or(20.0),
318        game_mode: root.int("playerGameType").unwrap_or_default(),
319        prev_game_mode: root.int("previousPlayerGameType"),
320        abilities: PersistentAbilities {
321            invulnerable: ability("invulnerable"),
322            flying: ability("flying"),
323            may_fly: ability("mayfly"),
324            instabuild: ability("instabuild"),
325            may_build: abilities
326                .is_none_or(|abilities| abilities.byte("mayBuild").is_none_or(|value| value != 0)),
327            flying_speed: ability_float("flySpeed", 0.05),
328            walking_speed: ability_float("walkSpeed", 0.1),
329        },
330        inventory,
331        selected_slot: root.int("SelectedItemSlot").unwrap_or_default(),
332        world,
333        food_level: root.int("foodLevel").unwrap_or(20),
334        food_saturation_level: root.float("foodSaturationLevel").unwrap_or(5.0),
335        food_exhaustion_level: root.float("foodExhaustionLevel").unwrap_or_default(),
336        food_tick_timer: root.int("foodTickTimer").unwrap_or_default(),
337        data_version: PLAYER_DATA_VERSION,
338        experience_level: root.int("XpLevel").unwrap_or_default(),
339        experience_progress: root.float("XpP").unwrap_or_default(),
340        experience_total: root.int("XpTotal").unwrap_or_default(),
341        enchantment_seed: root.int("XpSeed").unwrap_or_default(),
342        score: root.int("Score").unwrap_or_default(),
343        seen_credits: nbt::boolean(root, "seenCredits"),
344        root_vehicle,
345        respawn_config: root.compound("respawn").and_then(respawn_config),
346        ender_pearls,
347        ender_items: items(root.list("EnderItems"), report),
348        stats: Vec::new(),
349        advancements: Vec::new(),
350    })
351}

353/// Reads a vanilla `ItemStackWithSlot` list.
354fn items(list: Option<&NbtList>, report: &mut PlayerReport) -> Vec<PersistentSlot> {
355    list.and_then(|list| list.compounds())
356        .unwrap_or_default()
357        .iter()
358        .filter_map(|entry| {
359            let slot = entry.byte("Slot")?;
360            let item = item_stack(entry, report)?;
361            Some(PersistentSlot { slot, item })
362        })
363        .collect()
364}
365
366fn item_stack(entry: &NbtCompound, report: &mut PlayerReport) -> Option<ItemStack> {
367    let item = nbt::without_keys(entry, &["Slot"]);
368    let bytes = nbt::compound_bytes(&item);
369    let parsed = borrow::read_compound(&mut io::Cursor::new(&bytes[..])).ok();
370    let stack = parsed.and_then(|compound| {
371        ItemStack::from_borrowed_compound(&borrow::NbtCompound::from(&compound))
372    });
373    if stack.is_none() {
374        let id = nbt::string(entry, "id").unwrap_or_else(|| "<missing id>".to_owned());
375        ImportReport::count(&mut report.unknown_items, id);
376    }
377    stack
378}

380/// `ServerPlayer.RespawnConfig`: `{dimension, pos, yaw, pitch, forced}`.
381fn respawn_config(respawn: &NbtCompound) -> Option<PlayerRespawnConfig> {
382    let dimension = nbt::identifier(&nbt::string(respawn, "dimension")?)?;
383    let [x, y, z] = *respawn.int_array("pos")? else {
384        return None;
385    };
386    Some(PlayerRespawnConfig {
387        respawn_data: RespawnData::of(
388            dimension,
389            BlockPos::new(x, y, z),
390            respawn.float("yaw").unwrap_or_default(),
391            respawn.float("pitch").unwrap_or_default(),
392        ),
393        forced: nbt::boolean(respawn, "forced"),
394    })
395}

397/// Vanilla `ServerStatsCounter`: `{"stats": {type: {value: count}}}`.
398fn convert_stats(stats: &Value, report: &mut PlayerReport) -> Result<Vec<PersistentStat>, String> {
399    let types = stats
400        .get("stats")
401        .and_then(Value::as_object)
402        .ok_or("stats file has no stats object")?;
403    let mut converted = Vec::new();
404    for (type_key, values) in types {
405        let values = values
406            .as_object()
407            .ok_or_else(|| format!("stat type {type_key} is not an object"))?;
408        let stat_type = nbt::identifier(type_key).and_then(|key| REGISTRY.stat_types.by_key(&key));
409        for (value_key, count) in values {
410            let stat = stat_type.and_then(|stat_type| {
411                let value = stat_type.value_from_key(&nbt::identifier(value_key)?)?;
412                Some(Stat::from_erased(stat_type, value))
413            });
414            let count = count
415                .as_i64()
416                .and_then(|count| i32::try_from(count).ok())
417                .ok_or_else(|| format!("stat {type_key} {value_key} is not an int"))?;
418            match stat {
419                Some(stat) => converted.push(PersistentStat { stat, count }),
420                None => {
421                    ImportReport::count(
422                        &mut report.unknown_stats,
423                        format!("{type_key} {value_key}"),
424                    );
425                }
426            }
427        }
428    }
429    Ok(converted)
430}

432/// Vanilla `PlayerAdvancements`: `{id: {criteria: {name: date}, done}, DataVersion}`.
433fn convert_advancements(
434    advancements: &Value,
435    report: &mut PlayerReport,
436) -> Result<Vec<(Identifier, AdvancementProgress)>, String> {
437    let entries = advancements
438        .as_object()
439        .ok_or("advancements file is not an object")?;
440    let mut converted = Vec::new();
441    for (key, progress) in entries {
442        if key == "DataVersion" {
443            continue;
444        }
445        let Some(id) =
446            nbt::identifier(key).filter(|id| REGISTRY.advancements.value_by_key(id).is_some())
447        else {
448            ImportReport::count(&mut report.unknown_advancements, key.clone());
449            continue;
450        };
451        let mut criteria = FxHashMap::default();
452        for (name, date) in progress
453            .get("criteria")
454            .and_then(Value::as_object)
455            .into_iter()
456            .flatten()
457        {
458            let date = date
459                .as_str()
460                .ok_or_else(|| format!("criterion {key} {name} has no date"))?;
461            let obtained = DateTime::parse_from_str(date, ADVANCEMENT_DATE_FORMAT)
462                .map_err(|error| format!("criterion {key} {name}: {error}"))?;
463            criteria.insert(
464                Cow::Owned(name.clone()),
465                CriterionProgress(Some(SystemTime::from(obtained))),
466            );
467        }
468        converted.push((id, AdvancementProgress::new(criteria)));
469    }
470    Ok(converted)
471}

473fn read_json(path: &Path) -> Result<Option<Value>, String> {
474    match fs::read(path) {
475        Ok(bytes) => serde_json::from_slice(&bytes)
476            .map(Some)
477            .map_err(|error| format!("{}: {error}", path.display())),
478        Err(error) if error.kind() == io::ErrorKind::NotFound => Ok(None),
479        Err(error) => Err(format!("{}: {error}", path.display())),
480    }
