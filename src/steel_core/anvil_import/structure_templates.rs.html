29/// Converts the payload of a template piece of vanilla type `piece_type` and returns it with the
30/// piece's bounding box, or returns `None` when the type is not a template piece.
31///
32/// The bounding box is computed again from the template, as `TemplateStructurePiece`'s load
33/// constructor does: the saved one can be moved by a placement adjustment (igloos move theirs
34/// to the terrain height while placing), and vanilla does not keep that after a reload.
35pub(super) fn template_payload(
36    piece_type: &str,
37    piece: &NbtCompound,
38) -> Option<Result<(PersistentStructurePiecePayload, PersistentBoundingBox), String>> {
39    let data = match piece_type {
40        "minecraft:shipwreck" => shipwreck(piece),
41        "minecraft:iglu" => igloo(piece),
42        "minecraft:orp" => ocean_ruin(piece),
43        "minecraft:rupo" => ruined_portal(piece),
44        "minecraft:nefos" => nether_fossil(piece),
45        "minecraft:ecp" => end_city(piece),
46        "minecraft:wmp" => woodland_mansion(piece),
47        _ => return None,
48    };
49    Some(data.and_then(|data| {
50        let bounding_box = template_bounding_box(&data)?;
51        let payload = ChunkStorage::structure_piece_payload_to_persistent(
52            &StructurePiecePayload::Template(data),
53        );
54        Ok((payload, bounding_box))
55    }))
56}

58/// `StructureTemplate.getBoundingBox(StructurePlaceSettings, BlockPos)`: the template's corners
59/// transformed by `StructureTemplate.transform` (mirror, then rotation around the pivot).
60fn template_bounding_box(data: &TemplatePieceData) -> Result<PersistentBoundingBox, String> {
61    let [size_x, size_y, size_z] = template_size(&data.template_id)?;
62    let transform = |pos: IVec3| {
63        let mirrored = match data.mirror {
64            StructureMirror::None => pos,
65            StructureMirror::LeftRight => IVec3::new(pos.x, pos.y, -pos.z),
66            StructureMirror::FrontBack => IVec3::new(-pos.x, pos.y, pos.z),
67        };
68        data.rotation.transform_pos(mirrored, data.rotation_pivot)
69    };
70    let first = transform(IVec3::ZERO);
71    let second = transform(IVec3::new(size_x - 1, size_y - 1, size_z - 1));
72    let min = first.min(second) + data.template_position;
73    let max = first.max(second) + data.template_position;
74    Ok(PersistentBoundingBox {
75        min_x: min.x,
76        min_y: min.y,
77        min_z: min.z,
78        max_x: max.x,
79        max_y: max.y,
80        max_z: max.z,
81    })
82}

84/// Settings shared by every family, from `StructurePlaceSettings`' defaults.
85struct Base {
86    template_id: Identifier,
87    template_position: IVec3,
88    rotation: Rotation,
89}
90
91impl Base {
92    /// Reads `TemplateStructurePiece(type, tag, ...)`: `Template` and `TPX`/`TPY`/`TPZ`.
93    /// `prefix` is the folder vanilla's `makeTemplateLocation` puts before a short name.
94    fn read(piece: &NbtCompound, prefix: Option<&str>, rotation_key: &str) -> Result<Self, String> {
95        let name = nbt::string(piece, "Template").ok_or("template piece without a template")?;
96        let template_id = match prefix {
97            Some(prefix) => Identifier::vanilla(format!("{prefix}/{name}")),
98            None => nbt::identifier(&name).ok_or("malformed template id")?,
99        };
100        if vanilla_template_nbt_bytes(&template_id).is_none() {
101            return Err(format!("structure template {template_id} is not bundled"));
102        }
103        Ok(Self {
104            template_id,
105            template_position: IVec3::new(
106                int(piece, "TPX")?,
107                int(piece, "TPY")?,
108                int(piece, "TPZ")?,
109            ),
110            rotation: rotation(piece, rotation_key)?,
111        })
112    }
113
114    fn into_data(
115        self,
116        mirror: StructureMirror,
117        rotation_pivot: IVec3,
118        block_ignore: StructureBlockIgnore,
119        family: Family,
120    ) -> TemplatePieceData {
121        TemplatePieceData {
122            template_id: self.template_id,
123            template_position: self.template_position,
124            rotation: self.rotation,
125            mirror,
126            rotation_pivot,
127            block_ignore,
128            late_block_ignore: family.late_block_ignore,
129            processors: family.processors,
130            liquid_settings: family.liquid_settings,
131            marker_handling: family.marker_handling,
132            placement_adjustment: family.placement_adjustment,
133            placement_clip: family.placement_clip,
134            post_process: family.post_process,
135        }
136    }
137}

139/// Family behavior that is not part of `StructurePlaceSettings`' rotation and mirror.
140struct Family {
141    late_block_ignore: StructureBlockIgnore,
142    processors: TemplateProcessorList,
143    liquid_settings: LiquidSettingsData,
144    marker_handling: TemplateMarkerHandling,
145    placement_adjustment: TemplatePlacementAdjustment,
146    placement_clip: TemplatePlacementClip,
147    post_process: TemplatePostProcess,
148}
149
150impl Family {
151    const fn new(marker_handling: TemplateMarkerHandling) -> Self {
152        Self {
153            late_block_ignore: StructureBlockIgnore::None,
154            processors: TemplateProcessorList::Empty,
155            liquid_settings: LiquidSettingsData::ApplyWaterlogging,
156            marker_handling,
157            placement_adjustment: TemplatePlacementAdjustment::None,
158            placement_clip: TemplatePlacementClip::CenterChunk,
159            post_process: TemplatePostProcess::None,
160        }
161    }
162}

164/// `ShipwreckPieces.ShipwreckPiece(StructureTemplateManager, CompoundTag)`.
165fn shipwreck(piece: &NbtCompound) -> Result<TemplatePieceData, String> {
166    let base = Base::read(piece, None, "Rot")?;
167    let family = Family {
168        placement_adjustment: TemplatePlacementAdjustment::Shipwreck {
169            is_beached: nbt::boolean(piece, "isBeached"),
170            height_adjusted: nbt::boolean(piece, "height_adjusted"),
171        },
172        ..Family::new(TemplateMarkerHandling::Shipwreck)
173    };
174    Ok(base.into_data(
175        StructureMirror::None,
176        SHIPWRECK_PIVOT,
177        StructureBlockIgnore::StructureAndAir,
178        family,
179    ))
180}

182/// `IglooPieces.IglooPiece(StructureTemplateManager, CompoundTag)` with `IglooPieces.PIVOTS` and
183/// `IglooPieces.OFFSETS`.
184fn igloo(piece: &NbtCompound) -> Result<TemplatePieceData, String> {
185    let base = Base::read(piece, None, "Rot")?;
186    let (pivot, offset, post_process) = match base.template_id.path.as_ref() {
187        "igloo/top" => (
188            IVec3::new(3, 5, 5),
189            (0, 0, 0),
190            TemplatePostProcess::IglooTop,
191        ),
192        "igloo/middle" => (IVec3::new(1, 3, 1), (2, -3, 4), TemplatePostProcess::None),
193        "igloo/bottom" => (IVec3::new(3, 6, 7), (0, -3, -2), TemplatePostProcess::None),
194        _ => return Err(format!("{} is not an igloo template", base.template_id)),
195    };
196    let family = Family {
197        liquid_settings: LiquidSettingsData::IgnoreWaterlogging,
198        placement_adjustment: TemplatePlacementAdjustment::Igloo {
199            template_offset: offset,
200        },
201        post_process,
202        ..Family::new(TemplateMarkerHandling::Igloo)
203    };
204    Ok(base.into_data(
205        StructureMirror::None,
206        pivot,
207        StructureBlockIgnore::StructureBlock,
208        family,
209    ))
210}

212/// `OceanRuinPieces.OceanRuinPiece.create(StructureTemplateManager, CompoundTag)`.
213fn ocean_ruin(piece: &NbtCompound) -> Result<TemplatePieceData, String> {
214    let base = Base::read(piece, None, "Rot")?;
215    let biome_temp = match nbt::string(piece, "BiomeType").as_deref() {
216        Some("WARM") => OceanRuinBiomeTempData::Warm,
217        Some("COLD") => OceanRuinBiomeTempData::Cold,
218        other => return Err(format!("invalid ocean ruin biome type {other:?}")),
219    };
220    let family = Family {
221        late_block_ignore: StructureBlockIgnore::StructureAndAir,
222        processors: TemplateProcessorList::OceanRuin {
223            biome_temp,
224            integrity: piece.float("Integrity").unwrap_or(0.0),
225        },
226        placement_adjustment: TemplatePlacementAdjustment::OceanRuin,
227        ..Family::new(TemplateMarkerHandling::OceanRuin {
228            is_large: nbt::boolean(piece, "IsLarge"),
229        })
230    };
231    Ok(base.into_data(
232        StructureMirror::None,
233        IVec3::ZERO,
234        StructureBlockIgnore::None,
235        family,
236    ))
237}

239/// `RuinedPortalPiece(StructurePieceSerializationContext, CompoundTag)`.
240fn ruined_portal(piece: &NbtCompound) -> Result<TemplatePieceData, String> {
241    let base = Base::read(piece, None, "Rotation")?;
242    let mirror = mirror(piece, "Mirror")?;
243    let vertical_placement = match nbt::string(piece, "VerticalPlacement").as_deref() {
244        Some("on_land_surface") => RuinedPortalPlacementData::OnLandSurface,
245        Some("partly_buried") => RuinedPortalPlacementData::PartlyBuried,
246        Some("on_ocean_floor") => RuinedPortalPlacementData::OnOceanFloor,
247        Some("in_mountain") => RuinedPortalPlacementData::InMountain,
248        Some("underground") => RuinedPortalPlacementData::Underground,
249        Some("in_nether") => RuinedPortalPlacementData::InNether,
250        other => return Err(format!("invalid ruined portal placement {other:?}")),
251    };
252    let properties = piece
253        .compound("Properties")
254        .ok_or("ruined portal without properties")?;
255    let flag = |key: &str| {
256        properties
257            .byte(key)
258            .map(|value| value != 0)
259            .ok_or_else(|| format!("ruined portal properties without {key}"))
260    };
261    let properties = RuinedPortalProperties {
262        cold: flag("cold")?,
263        mossiness: properties
264            .float("mossiness")
265            .ok_or("ruined portal properties without mossiness")?,
266        air_pocket: flag("air_pocket")?,
267        overgrown: flag("overgrown")?,
268        vines: flag("vines")?,
269        replace_with_blackstone: flag("replace_with_blackstone")?,
270    };
271    // The pivot is the template's horizontal center, as in `makeSettings(..., tag, location)`.
272    let [size_x, _, size_z] = template_size(&base.template_id)?;
273    let block_ignore = if properties.air_pocket {
274        StructureBlockIgnore::StructureBlock
275    } else {
276        StructureBlockIgnore::StructureAndAir
277    };
278    let family = Family {
279        processors: TemplateProcessorList::RuinedPortal {
280            vertical_placement,
281            properties,
282        },
283        placement_clip: TemplatePlacementClip::CenterChunkContainsTemplateCenterExpandedToTemplate,
284        post_process: TemplatePostProcess::RuinedPortal,
285        ..Family::new(TemplateMarkerHandling::Ignore)
286    };
287    Ok(base.into_data(
288        mirror,
289        IVec3::new(size_x / 2, 0, size_z / 2),
290        block_ignore,
291        family,
292    ))
293}

339/// The `size` of a bundled template, `StructureTemplate.getSize`.
340fn template_size(template_id: &Identifier) -> Result<[i32; 3], String> {
341    let compressed = vanilla_template_nbt_bytes(template_id)
342        .ok_or_else(|| format!("structure template {template_id} is not bundled"))?;
343    let mut bytes = Vec::new();
344    GzDecoder::new(compressed)
345        .read_to_end(&mut bytes)
346        .map_err(|error| format!("structure template {template_id}: {error}"))?;
347    let root = nbt::read_root(&bytes)?;
348    root.list("size")
349        .and_then(NbtList::ints)
350        .and_then(|size| size.try_into().ok())
351        .ok_or_else(|| format!("structure template {template_id} has no size"))
352}

354/// `Rotation.LEGACY_CODEC`.
355pub(super) fn rotation(piece: &NbtCompound, key: &str) -> Result<Rotation, String> {
356    match nbt::string(piece, key).as_deref() {
357        Some("NONE") => Ok(Rotation::None),
358        Some("CLOCKWISE_90") => Ok(Rotation::Clockwise90),
359        Some("CLOCKWISE_180") => Ok(Rotation::Clockwise180),
360        Some("COUNTERCLOCKWISE_90") => Ok(Rotation::CounterClockwise90),
361        other => Err(format!("invalid rotation {other:?}")),
362    }
363}

365/// `Mirror.LEGACY_CODEC`.
366fn mirror(piece: &NbtCompound, key: &str) -> Result<StructureMirror, String> {
367    match nbt::string(piece, key).as_deref() {
368        Some("NONE") => Ok(StructureMirror::None),
369        Some("LEFT_RIGHT") => Ok(StructureMirror::LeftRight),
370        Some("FRONT_BACK") => Ok(StructureMirror::FrontBack),
371        other => Err(format!("invalid mirror {other:?}")),
372    }
373}

375fn int(compound: &NbtCompound, key: &str) -> Result<i32, String> {
