1//! Conversion of vanilla world data into Steel's per-world `level.toml`.
2//!
3//! Vanilla 26.x keeps world data in several files:
4//! - `level.dat`: game time, spawn, difficulty, default game mode.
5//! - `data/minecraft/world_gen_settings.dat`: the seed.
6//! - `data/minecraft/game_rules.dat`, `weather.dat` and `world_clocks.dat`: server-wide state.
7//! - `dimensions/<namespace>/<name>/data/minecraft/world_border.dat`: one border per dimension.
8//!
9//! Steel stores this in one `level.toml` per world, so server-wide vanilla state (game rules,
10//! weather, clocks, game time) is written into every imported world.
11

30/// What the level data import wrote and what it could not carry over.
31#[derive(Debug, Default)]
32pub struct LevelReport {
33    /// `level.toml` files written.
34    pub written: Vec<PathBuf>,
35    /// World seed.
36    pub seed: i64,
37    /// Vanilla default game mode id (`GameType` in `level.dat`).
38    pub game_type: i32,
39    /// Whether the vanilla world is hardcore.
40    pub hardcore: bool,
41    /// Game rules Steel does not know, with their vanilla value.
42    pub unknown_game_rules: Vec<String>,
43    /// Game rules whose saved value Steel rejects; they keep Steel's default.
44    pub invalid_game_rules: Vec<String>,
45    /// World clocks Steel does not know.
46    pub unknown_clocks: Vec<String>,
47}
48
49impl fmt::Display for LevelReport {
50    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
51        for path in &self.written {
52            writeln!(f, "Wrote {}", path.display())?;
53        }
54        for (title, entries) in [
55            ("Unknown game rules dropped", &self.unknown_game_rules),
56            (
57                "Game rules with invalid values reset to the default",
58                &self.invalid_game_rules,
59            ),
60            ("Unknown world clocks dropped", &self.unknown_clocks),
61        ] {
62            if !entries.is_empty() {
63                writeln!(f, "{title}:")?;
64                for entry in entries {
65                    writeln!(f, "  {entry}")?;
66                }
67            }
68        }
69        Ok(())
70    }
71}

76    game_time: i64,
77    difficulty: Difficulty,
78    difficulty_locked: bool,

86/// Writes `level.toml` for every imported dimension.
87///
88/// Fails without writing anything if one of the worlds already has a `level.toml`.
89pub fn import_level_data(
90    world: &Path,
91    dimensions: &[DimensionImport],
92) -> Result<LevelReport, AnvilImportError> {
93    let targets = dimensions
94        .iter()
95        .map(|dimension| {
96            let world_dir = dimension
97                .target
98                .parent()
99                .unwrap_or(&dimension.target)
100                .to_path_buf();
101            let path = world_dir.join("level.toml");
102            if path.exists() {
103                return Err(AnvilImportError::LevelDataExists { path });
104            }
105            Ok(path)
106        })
107        .collect::<Result<Vec<_>, _>>()?;
108
109    let mut report = LevelReport::default();
110    let shared = read_shared_state(world, &mut report)?;
111
112    for (dimension, path) in dimensions.iter().zip(targets) {
113        let border_path = dimension
114            .source
115            .join("data")
116            .join("minecraft")
117            .join("world_border.dat");
118        let world_border = match read_saved_data(&border_path)? {
119            Some(data) => world_border(&data),
120            None => WorldBorderData::default(),
121        };
122
123        let mut level = LevelData::new_with_seed_and_difficulty(shared.seed, shared.difficulty);
124        level.set_saved_game_time(shared.game_time);
125        level.world_clocks = shared.world_clocks.clone();
126        level.weather = shared.weather.clone();
127        level.world_border = world_border;
128        level.difficulty_locked = shared.difficulty_locked;
129        level.game_rules_values = shared.game_rules.clone();
130        level.initialized = shared.initialized;
131        if *shared.respawn.dimension() == dimension.key {
132            level.set_spawn_pos(shared.respawn.pos());
133            level.spawn.angle = shared.respawn.yaw;
134        }
135        // Steel keeps the global respawn on the domain default world.
136        if dimension.key == OVERWORLD.key {
137            level.set_respawn_data(shared.respawn.clone());
138        }
139        level.save_game_rules();
140
141        let content = toml::to_string_pretty(&level).map_err(|error| AnvilImportError::Io {
142            path: path.clone(),
143            source: io::Error::new(io::ErrorKind::InvalidData, error),
144        })?;
145        if let Some(parent) = path.parent() {
146            fs::create_dir_all(parent).map_err(|source| AnvilImportError::Io {
147                path: parent.to_path_buf(),
148                source,
149            })?;
150        }
151        fs::write(&path, content).map_err(|source| AnvilImportError::Io {
152            path: path.clone(),
153            source,
154        })?;
155        report.written.push(path);
156    }
157    Ok(report)
158}

160fn read_shared_state(
161    world: &Path,
162    report: &mut LevelReport,
163) -> Result<SharedLevelState, AnvilImportError> {
164    let level_path = world.join("level.dat");
165    let invalid = |reason: String| AnvilImportError::InvalidFile {
166        path: level_path.clone(),
167        reason,
168    };
169    let root = read_gzip_root(&level_path)?;
170    let data = root
171        .compound("Data")
172        .ok_or_else(|| invalid("missing Data".to_owned()))?;
173    chunk::check_data_version(data).map_err(invalid)?;
174
175    report.seed = read_world_seed(world)?;
176    report.game_type = data.int("GameType").unwrap_or(0);
177
178    let (difficulty, difficulty_locked) = match data.compound("difficulty_settings") {
179        Some(settings) => {
180            report.hardcore = nbt::boolean(settings, "hardcore");
181            let name = nbt::string(settings, "difficulty").unwrap_or_default();
182            let difficulty = difficulty_by_name(&name)
183                .ok_or_else(|| invalid(format!("unknown difficulty {name:?}")))?;
184            (difficulty, nbt::boolean(settings, "locked"))
185        }
186        None => (Difficulty::default(), false),
187    };
188
189    let data_dir = world.join("data").join("minecraft");
190    let game_rules = match read_saved_data(&data_dir.join("game_rules.dat"))? {
191        Some(rules) => game_rules(&rules, report),
192        None => GameRuleValues::new(&REGISTRY.game_rules),
193    };
194    let weather = match read_saved_data(&data_dir.join("weather.dat"))? {
195        Some(weather) => weather_state(&weather),
196        None => WeatherState::default(),
197    };
198    let clocks_path = data_dir.join("world_clocks.dat");
199    let world_clocks = match read_saved_data(&clocks_path)? {
200        Some(clocks) => {
201            world_clocks(&clocks, report).map_err(|reason| AnvilImportError::InvalidFile {
202                path: clocks_path.clone(),
203                reason,
204            })?
205        }
206        None => WorldClockManager::new(),
207    };
208
209    Ok(SharedLevelState {
210        seed: report.seed,
211        game_time: data.long("Time").unwrap_or(0),
212        difficulty,
213        difficulty_locked,
214        initialized: nbt::boolean(data, "initialized"),
215        respawn: respawn(data.compound("spawn")),
216        game_rules,
217        weather,
218        world_clocks,
219    })
220}

222/// Vanilla `Difficulty` is saved by its serialized name.
223fn difficulty_by_name(name: &str) -> Option<Difficulty> {
224    match name {
225        "peaceful" => Some(Difficulty::Peaceful),
226        "easy" => Some(Difficulty::Easy),
227        "normal" => Some(Difficulty::Normal),
228        "hard" => Some(Difficulty::Hard),
229        _ => None,
230    }
231}

233/// `LevelData.RespawnData`: `{dimension, pos: [x, y, z], yaw, pitch}`. Vanilla falls back to the
234/// overworld origin when it is missing.
235fn respawn(spawn: Option<&NbtCompound>) -> RespawnData {
236    let default = || RespawnData::of(OVERWORLD.key.clone(), BlockPos::new(0, 0, 0), 0.0, 0.0);
237    let Some(spawn) = spawn else {
238        return default();
239    };
240    let dimension = nbt::string(spawn, "dimension")
241        .as_deref()
242        .and_then(nbt::identifier);
243    let pos = spawn.int_array("pos").and_then(|pos| match *pos {
244        [x, y, z] => Some(BlockPos::new(x, y, z)),
245        _ => None,
246    });
247    match (dimension, pos) {
248        (Some(dimension), Some(pos)) => RespawnData::of(
249            dimension,
250            pos,
251            spawn.float("yaw").unwrap_or(0.0),
252            spawn.float("pitch").unwrap_or(0.0),
253        ),
254        _ => default(),
255    }
256}

258/// `game_rules.dat` stores every rule by id: booleans as bytes, integers as ints.
259fn game_rules(rules: &NbtCompound, report: &mut LevelReport) -> GameRuleValues {
260    let mut values = GameRuleValues::new(&REGISTRY.game_rules);
261    for (name, tag) in rules.iter() {
262        let name = name.to_str();
263        let rule = nbt::identifier(&name).and_then(|key| REGISTRY.game_rules.by_key(&key));
264        let Some(rule) = rule else {
265            report
266                .unknown_game_rules
267                .push(format!("{name} = {}", tag_value(tag)));
268            continue;
269        };
270        let default = rule.serialize_erased_value(&rule.default_erased_value());
271        let value = match (default, tag) {
272            (Value::Bool(_), NbtTag::Byte(value)) => Some(Value::Bool(*value != 0)),
273            (Value::Number(_), NbtTag::Int(value)) => Some(Value::from(*value)),
274            _ => None,
275        };
276        let applied = value.is_some_and(|value| {
277            values.set_serialized_by_name(&name, &value, &REGISTRY.game_rules)
278        });
279        if !applied {
280            report
281                .invalid_game_rules
282                .push(format!("{name} = {}", tag_value(tag)));
283        }
284    }
285    values
286}
287
288fn tag_value(tag: &NbtTag) -> String {
289    match tag {
290        NbtTag::Byte(value) => value.to_string(),
291        NbtTag::Int(value) => value.to_string(),
292        NbtTag::String(value) => format!("{:?}", value.to_str()),
293        _ => "<unsupported tag>".to_owned(),
294    }
295}

297/// `WeatherData`: the same fields as Steel's weather state.
298fn weather_state(weather: &NbtCompound) -> WeatherState {
299    WeatherState {
300        raining: nbt::boolean(weather, "raining"),
301        rain_time: weather.int("rain_time").unwrap_or(0),
302        thundering: nbt::boolean(weather, "thundering"),
303        thunder_time: weather.int("thunder_time").unwrap_or(0),
304        clear_weather_time: weather.int("clear_weather_time").unwrap_or(0),
305    }

308/// `WorldBorder.Settings`: the same fields as Steel's border data; missing fields keep the
309/// vanilla defaults.
310fn world_border(border: &NbtCompound) -> WorldBorderData {
311    let default = WorldBorderData::default();
312    WorldBorderData {
313        center_x: border.double("center_x").unwrap_or(default.center_x),
314        center_z: border.double("center_z").unwrap_or(default.center_z),
315        damage_per_block: border
316            .double("damage_per_block")
317            .unwrap_or(default.damage_per_block),
318        safe_zone: border.double("safe_zone").unwrap_or(default.safe_zone),
319        warning_blocks: border
320            .int("warning_blocks")
321            .unwrap_or(default.warning_blocks),
322        warning_time: border.int("warning_time").unwrap_or(default.warning_time),
323        size: border.double("size").unwrap_or(default.size),
324        lerp_time: border.long("lerp_time").unwrap_or(default.lerp_time),
325        lerp_target: border.double("lerp_target").unwrap_or(default.lerp_target),
326    }
327}

329/// `world_clocks.dat` maps clock ids to `ClockState {total_ticks, partial_tick, rate, paused}`,
330/// the same fields Steel persists, with the same defaults for missing ones.
331fn world_clocks(
332    clocks: &NbtCompound,
333    report: &mut LevelReport,
334) -> Result<WorldClockManager, String> {
335    let mut states = Map::new();
336    for (key, tag) in clocks.iter() {
337        let key = key.to_str();
338        let known =
339            nbt::identifier(&key).is_some_and(|id| REGISTRY.world_clocks.by_key(&id).is_some());
340        let NbtTag::Compound(state) = tag else {
341            return Err(format!("clock {key} is not a compound"));
342        };
343        if !known {
344            report.unknown_clocks.push(key.into_owned());
345            continue;
346        }
347        let mut fields = Map::new();
348        if let Some(total_ticks) = state.long("total_ticks") {
349            fields.insert("total_ticks".to_owned(), Value::from(total_ticks));
350        }
351        if let Some(partial_tick) = state.float("partial_tick") {
352            fields.insert("partial_tick".to_owned(), Value::from(partial_tick));
353        }
354        if let Some(rate) = state.float("rate") {
355            fields.insert("rate".to_owned(), Value::from(rate));
356        }
357        if state.contains("paused") {
358            fields.insert(
359                "paused".to_owned(),
360                Value::Bool(nbt::boolean(state, "paused")),
361            );
362        }
363        states.insert(key.into_owned(), Value::Object(fields));
364    }
365    let mut manager: WorldClockManager =
366        serde_json::from_value(Value::Object(states)).map_err(|error| error.to_string())?;
367    manager
368        .initialize_registered_clocks()
369        .map_err(|error| error.to_string())?;
370    Ok(manager)
371}

373/// Reads the `data` compound of a vanilla `SavedData` file, or `None` if the file is missing.
374fn read_saved_data(path: &Path) -> Result<Option<NbtCompound>, AnvilImportError> {
375    if !path.exists() {
376        return Ok(None);
377    }
378    let invalid = |reason: String| AnvilImportError::InvalidFile {
379        path: path.to_path_buf(),
380        reason,
381    };
382    let root = read_gzip_root(path)?;
383    chunk::check_data_version(&root).map_err(invalid)?;
384    root.compound("data")
385        .cloned()
386        .map(Some)
387        .ok_or_else(|| invalid("missing data".to_owned()))
388}

390/// Reads a gzip-compressed NBT file.
391pub(super) fn read_gzip_root(path: &Path) -> Result<NbtCompound, AnvilImportError> {
392    let invalid = |reason: String| AnvilImportError::InvalidFile {
393        path: path.to_path_buf(),
394        reason,
395    };
396    let compressed = fs::read(path).map_err(|source| AnvilImportError::Io {
397        path: path.to_path_buf(),
398        source,
399    })?;
400    let mut bytes = Vec::new();
401    GzDecoder::new(&compressed[..])
402        .read_to_end(&mut bytes)
403        .map_err(|error| invalid(error.to_string()))?;
404    nbt::read_root(&bytes).map_err(invalid)
