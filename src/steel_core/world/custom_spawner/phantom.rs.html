20/// Vanilla `dayLength * 3`: phantoms need three days without rest.
21const MIN_TICKS_SINCE_REST: i32 = 72_000;
22/// Sky darkening from which the overworld counts as night for phantoms.
23const MIN_SKY_DARKEN: u8 = 5;
24
25/// Vanilla `PhantomSpawner`.
26#[derive(Default)]
27pub(crate) struct PhantomSpawner {
28    next_tick: i32,
29}
30
31impl CustomSpawner for PhantomSpawner {
32    fn tick(&mut self, world: &Arc<World>, spawn_enemies: bool) {
33        if !spawn_enemies || !world.get_game_rule(&SPAWN_PHANTOMS) {
34            return;
35        }
36        let mut random = rand::rng();
37        self.next_tick -= 1;
38        if self.next_tick > 0 {
39            return;
40        }
41        self.next_tick += (60 + random.random_range(0..60)) * 20;
42        if world.sky_darkening() < MIN_SKY_DARKEN && world.dimension_type.has_skylight {
43            return;
44        }
45
46        let mut players = Vec::new();
47        world.players.iter_players(|_, player| {
48            players.push(Arc::clone(player));
49            true
50        });
51        for player in players {
52            if let Some((spawn_pos, difficulty)) = phantom_spawn_pos(world, &player, &mut random) {
53                spawn_phantom_group(world, spawn_pos, difficulty, &mut random);
54            }
55        }
56    }
57}

59/// The per-player half of vanilla `PhantomSpawner.tick`, up to the spawn
60/// block check: the position a group appears at, with the regional difficulty
61/// at the player.
62pub(super) fn phantom_spawn_pos(
63    world: &World,
64    player: &Player,
65    random: &mut impl Rng,
66) -> Option<(BlockPos, DifficultyInstance)> {
67    if player.is_spectator() {
68        return None;
69    }
70    let player_pos = player.block_position();
71    if world.dimension_type.has_skylight
72        && (player_pos.y() < world.sea_level || !world.can_see_sky(player_pos))
73    {
74        return None;
75    }
76    let difficulty = world.current_difficulty_at(player_pos);
77    if !difficulty.is_harder_than(random.random::<f32>() * 3.0) {
78        return None;
79    }
80    let ticks_since_rest = player.custom_stat_value(&TIME_SINCE_REST).max(1);
81    if random.random_range(0..ticks_since_rest) < MIN_TICKS_SINCE_REST {
82        return None;
83    }
84    let above = 20 + random.random_range(0..15);
85    let east = -10 + random.random_range(0..21);
86    let south = -10 + random.random_range(0..21);
87    let spawn_pos = player_pos.offset(east, above, south);
88    is_valid_empty_spawn_block(world, spawn_pos, &PHANTOM).then_some((spawn_pos, difficulty))
89}

91/// Vanilla's group loop: `1 + nextInt(difficulty id + 1)` phantoms at one block.
92fn spawn_phantom_group(
93    world: &Arc<World>,
94    spawn_pos: BlockPos,
95    difficulty: DifficultyInstance,
96    random: &mut impl Rng,
97) {
98    let mut group_data: Option<SpawnGroupData> = None;
99    let group_size = 1 + random.random_range(0..=i32::from(difficulty.difficulty() as u8));
100    let (x, y, z) = spawn_pos.get_bottom_center();
101    let position = DVec3::new(x, y, z);
102    for _ in 0..group_size {
103        // Vanilla `EntityType.create` returns null for a type it cannot build;
104        // Steel reports that as an error and the phantom is skipped the same way.
105        let Ok(phantom) = create_entity_instance(world, &PHANTOM, position) else {
106            continue;
107        };
108        phantom.snap_to(position, 0.0, 0.0);
109        if let Some(mob) = phantom.as_mob() {
110            group_data = mob.finalize_spawn(world, EntitySpawnReason::Natural, group_data.take());
111        }
112        if let Err(error) = world.try_add_fresh_entity_with_passengers(phantom) {
113            log::debug!("Phantom spawn failed: {error:?}");
114        }
115    }
116}
