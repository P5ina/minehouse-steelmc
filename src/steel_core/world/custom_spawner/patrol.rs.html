20/// Vanilla `nextTick += 12000 + random.nextInt(1200)`.
21const BASE_TICK_DELAY: i32 = 12_000;
22const TICK_DELAY_SPREAD: i32 = 1_200;
23/// Vanilla `random.nextInt(5) == 0`: one attempt in five spawns.
24const SPAWN_ATTEMPT_ROLL: i32 = 5;
25/// No patrol spawns within this many sections of a village.
26const VILLAGE_SECTION_DISTANCE: i32 = 2;
27/// The patrol lands `24 + nextInt(24)` blocks away on each axis.
28const MIN_PLAYER_DISTANCE: i32 = 24;
29const PLAYER_DISTANCE_SPREAD: i32 = 24;
30/// Vanilla `delta`: the area around the spawn position that must be loaded.
31const LOADED_AREA_RADIUS: i32 = 10;
32/// Members spread by `nextInt(5) - nextInt(5)` blocks from the previous one.
33const MEMBER_SPREAD: i32 = 5;
34
35/// Vanilla `PatrolSpawner`.
36#[derive(Default)]
37pub(crate) struct PatrolSpawner {
38    next_tick: i32,
39}
40
41impl CustomSpawner for PatrolSpawner {
42    fn tick(&mut self, world: &Arc<World>, spawn_enemies: bool) {
43        if !spawn_enemies || !world.get_game_rule(&SPAWN_PATROLS) {
44            return;
45        }
46        let mut random = rand::rng();
47        self.next_tick -= 1;
48        if self.next_tick > 0 {
49            return;
50        }
51        self.next_tick += BASE_TICK_DELAY + random.random_range(0..TICK_DELAY_SPREAD);
52        if !world.is_bright_outside() || random.random_range(0..SPAWN_ATTEMPT_ROLL) != 0 {
53            return;
54        }
55        if let Some(spawn_pos) = patrol_spawn_pos(world, &mut random) {
56            spawn_patrol(world, spawn_pos, &mut random);
57        }
58    }
59}

61/// The player half of vanilla `PatrolSpawner.tick`: a random player away from
62/// villages, and the position 24 to 47 blocks off them on each axis where the
63/// patrol may spawn.
64pub(super) fn patrol_spawn_pos(world: &World, random: &mut impl Rng) -> Option<BlockPos> {
65    let mut players = Vec::new();
66    world.players.iter_players(|_, player| {
67        players.push(Arc::clone(player));
68        true
69    });
70    if players.is_empty() {
71        return None;
72    }
73    let player = &players[random.random_range(0..players.len())];
74    if player.is_spectator() {
75        return None;
76    }
77    let player_pos = player.block_position();
78    if world.is_close_to_village(player_pos, VILLAGE_SECTION_DISTANCE) {
79        return None;
80    }
81    let x = patrol_offset(random);
82    let z = patrol_offset(random);
83    let spawn_pos = player_pos.offset(x, 0, z);
84    (has_chunks_at(
85        world,
86        spawn_pos.x() - LOADED_AREA_RADIUS,
87        spawn_pos.z() - LOADED_AREA_RADIUS,
88        spawn_pos.x() + LOADED_AREA_RADIUS,
89        spawn_pos.z() + LOADED_AREA_RADIUS,
90    ) && world.can_pillager_patrol_spawn(spawn_pos))
91    .then_some(spawn_pos)
92}

94/// Vanilla `(24 + nextInt(24)) * (nextBoolean() ? -1 : 1)`.
95fn patrol_offset(random: &mut impl Rng) -> i32 {
96    let distance = MIN_PLAYER_DISTANCE + random.random_range(0..PLAYER_DISTANCE_SPREAD);
97    if random.random::<bool>() {
98        -distance
99    } else {
100        distance
101    }
102}

104/// Vanilla `LevelReader.hasChunksAt(x0, z0, x1, z1)`: every chunk in the block
105/// rectangle is loaded.
106fn has_chunks_at(world: &World, min_x: i32, min_z: i32, max_x: i32, max_z: i32) -> bool {
107    let min = ChunkPos::from_block_pos(BlockPos::new(min_x, 0, min_z));
108    let max = ChunkPos::from_block_pos(BlockPos::new(max_x, 0, max_z));
109    (min.0.x..=max.0.x).all(|chunk_x| {
110        (min.0.y..=max.0.y).all(|chunk_z| {
111            world.is_full_chunk_loaded_at(BlockPos::new(chunk_x << 4, 0, chunk_z << 4))
112        })
113    })
114}

116/// Vanilla's group loop: `ceil(effective difficulty) + 1` pillagers, the first
117/// one leading. The patrol stops if the leader cannot spawn.
118pub(super) fn spawn_patrol(world: &Arc<World>, mut spawn_pos: BlockPos, random: &mut impl Rng) {
119    let effective_difficulty = world
120        .current_difficulty_at(spawn_pos)
121        .effective_difficulty();
122    // Effective difficulty stays within 0..=6.75, so the cast cannot truncate.
123    #[expect(
124        clippy::cast_possible_truncation,
125        reason = "vanilla (int) Math.ceil of a small float"
126    )]
127    let group_size = effective_difficulty.ceil() as i32 + 1;
128    for member in 0..group_size {
129        spawn_pos = BlockPos::new(
130            spawn_pos.x(),
131            world
132                .heightmap_pos(HeightmapType::MotionBlockingNoLeaves, spawn_pos)
133                .y(),
134            spawn_pos.z(),
135        );
136        let is_leader = member == 0;
137        if !spawn_patrol_member(world, spawn_pos, is_leader) && is_leader {
138            break;
139        }
140        spawn_pos = BlockPos::new(
141            spawn_pos.x() + random.random_range(0..MEMBER_SPREAD)
142                - random.random_range(0..MEMBER_SPREAD),
143            spawn_pos.y(),
144            spawn_pos.z() + random.random_range(0..MEMBER_SPREAD)
145                - random.random_range(0..MEMBER_SPREAD),
146        );
147    }
148}

150/// Vanilla `PatrolSpawner.spawnPatrolMember`.
151fn spawn_patrol_member(world: &Arc<World>, pos: BlockPos, is_leader: bool) -> bool {
152    if !is_valid_empty_spawn_block(world, pos, &PILLAGER)
153        || !<PillagerEntity as PatrollingMonster>::check_patrolling_monster_spawn_rules(
154            &PILLAGER,
155            world.as_ref(),
156            EntitySpawnReason::Patrol,
157            pos,
158        )
159    {
160        return false;
161    }
162    let position = DVec3::new(f64::from(pos.x()), f64::from(pos.y()), f64::from(pos.z()));
163    // Vanilla `EntityType.create` returns null for a type it cannot build;
164    // Steel reports that as an error and the member counts as not spawned.
165    // Vanilla creates the mob at the origin and picks the leader's patrol
166    // target before `setPos`, so the target lies within 500 blocks of 0, 0.
167    let Ok(pillager) = create_entity_instance(world, &PILLAGER, DVec3::ZERO) else {
168        return false;
169    };
170    if is_leader && let Some(patrolling) = pillager.as_patrolling_monster() {
171        patrolling.set_patrol_leader(true);
172        patrolling.find_patrol_target();
173    }
174    pillager.snap_to(position, 0.0, 0.0);
175    if let Some(mob) = pillager.as_mob() {
176        mob.finalize_spawn(world, EntitySpawnReason::Patrol, None);
177    }
178    if let Err(error) = world.try_add_fresh_entity_with_passengers(pillager) {
179        log::debug!("Patrol member spawn failed: {error:?}");
180    }
181    true
