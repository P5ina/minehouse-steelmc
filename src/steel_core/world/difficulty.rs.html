3use steel_registry::vanilla_world_clocks;
4use steel_utils::types::Difficulty;
5use steel_utils::{BlockPos, ChunkPos};
6
7use super::World;

9const DIFFICULTY_TIME_GLOBAL_OFFSET: f32 = -72000.0;
10const MAX_DIFFICULTY_TIME_GLOBAL: f32 = 1_440_000.0;
11const MAX_DIFFICULTY_TIME_LOCAL: f32 = 3_600_000.0;
12
13/// Vanilla `DifficultyInstance`: the world difficulty scaled by world age, how long
14/// players stayed in the chunk, and the moon phase.
15#[derive(Debug, Clone, Copy, PartialEq)]
16pub struct DifficultyInstance {
17    base: Difficulty,
18    effective_difficulty: f32,
19}
20
21impl DifficultyInstance {
22    /// Computes the regional difficulty from its vanilla inputs.
23    #[must_use]
24    pub fn new(
25        base: Difficulty,
26        total_game_time: i64,
27        local_game_time: i64,
28        moon_brightness: f32,
29    ) -> Self {
30        Self {
31            base,
32            effective_difficulty: Self::calculate_difficulty(
33                base,
34                total_game_time,
35                local_game_time,
36                moon_brightness,
37            ),
38        }
39    }
40
41    /// Returns the world difficulty this instance was computed from.
42    #[must_use]
43    pub const fn difficulty(self) -> Difficulty {
44        self.base
45    }
46
47    /// Returns the regional ("effective") difficulty, from 0 to 6.75.
48    #[must_use]
49    pub const fn effective_difficulty(self) -> f32 {
50        self.effective_difficulty
51    }
52
53    /// Vanilla `isHard`: the regional difficulty reaches the `HARD` ordinal.
54    #[must_use]
55    pub fn is_hard(self) -> bool {
56        self.effective_difficulty >= f32::from(Difficulty::Hard as u8)
57    }
58
59    /// Vanilla `isHarderThan`.
60    #[must_use]
61    pub fn is_harder_than(self, required_difficulty: f32) -> bool {
62        self.effective_difficulty > required_difficulty
63    }
64
65    /// Vanilla `getSpecialMultiplier` ("clamped regional difficulty"), from 0 to 1.
66    #[must_use]
67    pub fn special_multiplier(self) -> f32 {
68        if self.effective_difficulty < 2.0 {
69            0.0
70        } else if self.effective_difficulty > 4.0 {
71            1.0
72        } else {
73            (self.effective_difficulty - 2.0) / 2.0
74        }
75    }
76
77    fn calculate_difficulty(
78        base: Difficulty,
79        total_game_time: i64,
80        local_game_time: i64,
81        moon_brightness: f32,
82    ) -> f32 {
83        if base == Difficulty::Peaceful {
84            return 0.0;
85        }
86        let is_hard = base == Difficulty::Hard;
87        let mut scale = 0.75_f32;
88        let global_scale = ((total_game_time as f32 + DIFFICULTY_TIME_GLOBAL_OFFSET)
89            / MAX_DIFFICULTY_TIME_GLOBAL)
90            .clamp(0.0, 1.0)
91            * 0.25;
92        scale += global_scale;
93        let mut local_scale = 0.0_f32;
94        local_scale += (local_game_time as f32 / MAX_DIFFICULTY_TIME_LOCAL).clamp(0.0, 1.0)
95            * if is_hard { 1.0 } else { 0.75 };
96        local_scale += (moon_brightness * 0.25).clamp(0.0, global_scale);
97        if base == Difficulty::Easy {
98            local_scale *= 0.5;
99        }
100        scale += local_scale;
101        f32::from(base as u8) * scale
102    }
103}
104
105impl World {
106    /// Vanilla `ServerLevel.getCurrentDifficultyAt`. Unloaded chunks count as never
107    /// inhabited, under a new moon.
108    #[must_use]
109    pub fn current_difficulty_at(&self, pos: BlockPos) -> DifficultyInstance {
110        let chunk_pos = ChunkPos::from_block_pos(pos);
111        let (local_time, moon_brightness) = self
112            .chunk_map
113            .with_full_chunk(chunk_pos, |chunk| chunk.common().inhabited_time())
114            .map_or((0, 0.0), |inhabited| (inhabited, self.moon_brightness()));
115        let overworld_time = self
116            .clock_total_ticks(&vanilla_world_clocks::OVERWORLD)
117            .unwrap_or(0);
118        DifficultyInstance::new(
119            self.difficulty(),
120            overworld_time,
121            local_time,
122            moon_brightness,
123        )
124    }
125}

127#[cfg(test)]
128mod tests {
129    use steel_utils::types::Difficulty;
130
131    use super::DifficultyInstance;
132
133    #[test]
134    fn peaceful_is_always_zero() {
135        let instance = DifficultyInstance::new(Difficulty::Peaceful, 10_000_000, 10_000_000, 1.0);
136        assert!(instance.effective_difficulty().abs() < f32::EPSILON);
137        assert!(instance.special_multiplier().abs() < f32::EPSILON);
138    }
139
140    #[test]
141    fn fresh_world_uses_the_base_scale() {
142        // 0.75 × difficulty id with no world age, inhabited time, or moon.
143        let hard = DifficultyInstance::new(Difficulty::Hard, 0, 0, 1.0);
144        assert!((hard.effective_difficulty() - 2.25).abs() < 1e-6);
145        assert!((hard.special_multiplier() - 0.125).abs() < 1e-6);
146        let normal = DifficultyInstance::new(Difficulty::Normal, 0, 0, 1.0);
147        assert!((normal.effective_difficulty() - 1.5).abs() < 1e-6);
148        assert!(normal.special_multiplier().abs() < f32::EPSILON);
149    }
150
151    #[test]
152    fn old_inhabited_hard_chunk_under_full_moon_reaches_the_cap() {
153        let instance = DifficultyInstance::new(Difficulty::Hard, 2_000_000, 4_000_000, 1.0);
154        assert!((instance.effective_difficulty() - 6.75).abs() < 1e-5);
155        assert!(instance.is_hard());
156        assert!((instance.special_multiplier() - 1.0).abs() < f32::EPSILON);
157    }
158
159    #[test]
160    fn easy_halves_the_local_part() {
161        // global 0.25, local (0.75 + 0.25) * 0.5 = 0.5 → 1 × 1.5.
162        let instance = DifficultyInstance::new(Difficulty::Easy, 2_000_000, 4_000_000, 1.0);
163        assert!((instance.effective_difficulty() - 1.5).abs() < 1e-5);
164    }
165}
