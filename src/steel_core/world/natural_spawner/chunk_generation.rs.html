23/// Attempts per mob to find a free column near the group origin.
24const ATTEMPTS_PER_MOB: i32 = 4;

26/// Vanilla `WorldgenRandom.setDecorationSeed` over a `LegacyRandomSource`.
27pub(crate) fn set_decoration_seed(
28    random: &mut LegacyRandom,
29    seed: i64,
30    block_x: i32,
31    block_z: i32,
32) {
33    random.set_seed(seed);
34    let x_scale = random.next_i64() | 1;
35    let z_scale = random.next_i64() | 1;
36    random.set_seed(
37        i64::from(block_x)
38            .wrapping_mul(x_scale)
39            .wrapping_add(i64::from(block_z).wrapping_mul(z_scale))
40            ^ seed,
41    );
42}

44/// Vanilla `NaturalSpawner.spawnMobsForChunkGeneration`.
45///
46/// Spawn predicates read the region; `finalizeSpawn` runs against the live
47/// world, whose biome lookup falls back to the generator for chunks that are
48/// not loaded yet.
49pub(crate) fn spawn_mobs_for_chunk_generation(
50    region: &WorldGenRegion<'_>,
51    world: &Arc<World>,
52    source_pos: BlockPos,
53    chunk_pos: ChunkPos,
54    random: &mut LegacyRandom,
55) {
56    if !world.get_game_rule(&SPAWN_MOBS) {
57        return;
58    }
59    let mobs = world.biome_mobs_at(MobCategory::Creature, source_pos);
60    if mobs.is_empty() {
61        return;
62    }
63    let creature_probability = world
64        .biome_at(source_pos)
65        .map_or(0.0, |biome| biome.creature_spawn_probability);
66    let xo = chunk_pos.0.x * 16;
67    let zo = chunk_pos.0.y * 16;
68
69    while random.next_f32() < creature_probability {
70        let Some(spawner_data) = mobs.get_random_java(random) else {
71            continue;
72        };
73        let count = spawner_data.sample_count_java(random);
74        let mut group_data = None;
75        let start_x = xo + random.next_i32_bounded(16);
76        let start_z = zo + random.next_i32_bounded(16);
77        let (mut x, mut z) = (start_x, start_z);
78
79        for _ in 0..count {
80            let mut success = false;
81            let mut attempts = 0;
82            while !success && attempts < ATTEMPTS_PER_MOB {
83                attempts += 1;
84                if let Some(entity) =
85                    try_spawn_at(region, world, &spawner_data, (xo, zo), (x, z), random)
86                    && let Some(mob) = entity.as_mob()
87                {
88                    group_data = mob.finalize_spawn(
89                        world,
90                        EntitySpawnReason::ChunkGeneration,
91                        group_data.take(),
92                    );
93                    add_fresh_entity_with_passengers(region, &entity);
94                    success = true;
95                }
96
97                x += random.next_i32_bounded(5) - random.next_i32_bounded(5);
98                z += random.next_i32_bounded(5) - random.next_i32_bounded(5);
99                while x < xo || x >= xo + 16 || z < zo || z >= zo + 16 {
100                    x = start_x + random.next_i32_bounded(5) - random.next_i32_bounded(5);
101                    z = start_z + random.next_i32_bounded(5) - random.next_i32_bounded(5);
102                }
103            }
104        }
105    }
106}

108/// One placement attempt of vanilla's inner loop: returns the created mob when
109/// the column, the type's rules and the mob's own checks all accept it.
110fn try_spawn_at(
111    region: &WorldGenRegion<'_>,
112    world: &Arc<World>,
113    spawner_data: &SpawnerEntry,
114    (xo, zo): (i32, i32),
115    (x, z): (i32, i32),
116    random: &mut LegacyRandom,
117) -> Option<SharedEntity> {
118    let entity_type = spawner_data.entity_type;
119    let pos = top_non_colliding_pos(region, entity_type, x, z);
120    if !entity_type.summonable || !SpawnPlacements::is_spawn_position_ok(entity_type, region, pos) {
121        return None;
122    }
123    let width = f64::from(entity_type.dimensions.width);
124    let fx = clamp(
125        f64::from(x),
126        f64::from(xo) + width,
127        f64::from(xo) + 16.0 - width,
128    );
129    let fz = clamp(
130        f64::from(z),
131        f64::from(zo) + width,
132        f64::from(zo) + 16.0 - width,
133    );
134    let y = f64::from(pos.y());
135    // Vanilla passes the region's own random here, which the creature spawn
136    // rules never draw from; it stays separate from the seeded group random.
137    if level_has_block_collision(region, spawn_aabb(entity_type, fx, y, fz))
138        || !SpawnPlacements::check_spawn_rules(
139            entity_type,
140            region,
141            EntitySpawnReason::ChunkGeneration,
142            BlockPos::new(fx.floor() as i32, pos.y(), fz.floor() as i32),
143            &mut rand::rng(),
144        )
145    {
146        return None;
147    }
148
149    let position = DVec3::new(fx, y, fz);
150    let entity = match create_entity_instance(world, entity_type, position) {
151        Ok(entity) => entity,
152        Err(error) => {
153            log::warn!("Failed to create mob {}: {error:?}", entity_type.key);
154            return None;
155        }
156    };
157    entity.snap_to(position, random.next_f32() * DEGREE_360, 0.0);
158    let mob = entity.as_mob()?;
159    (mob.check_spawn_rules(region, EntitySpawnReason::ChunkGeneration)
160        && mob.check_spawn_obstruction(region))
161    .then_some(entity)
162}

164/// Vanilla `Level.addFreshEntityWithPassengers` on a worldgen region.
165fn add_fresh_entity_with_passengers(region: &WorldGenRegion<'_>, entity: &SharedEntity) {
166    let passengers = entity.passengers();
167    let _ = region.add_fresh_entity(Arc::clone(entity));
168    for passenger in passengers {
169        add_fresh_entity_with_passengers(region, &passenger);
170    }
171}

173/// Vanilla `NaturalSpawner.getTopNonCollidingPos`.
174fn top_non_colliding_pos(
175    region: &WorldGenRegion<'_>,
176    entity_type: EntityTypeRef,
177    x: i32,
178    z: i32,
179) -> BlockPos {
180    let level_height =
181        LevelReader::height_at(region, SpawnPlacements::heightmap_type(entity_type), x, z);
182    let mut pos = BlockPos::new(x, level_height, z);
183    if ServerLevelAccessor::dimension_type(region).has_ceiling {
184        loop {
185            pos = pos.below();
186            if region.get_block_state(pos).is_air() {
187                break;
188            }
189        }
190        loop {
191            pos = pos.below();
192            if !(region.get_block_state(pos).is_air() && pos.y() > LevelReader::min_y(region)) {
193                break;
194            }
195        }
196    }
197    SpawnPlacements::placement_type(entity_type).adjust_spawn_position(region, pos)
198}
