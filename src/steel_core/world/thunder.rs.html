21/// Vanilla `random.nextInt(100000) == 0` strike chance per chunk tick.
22const THUNDER_CHANCE: u32 = 100_000;
23/// Vanilla `findLightningRod` search radius.
24const LIGHTNING_ROD_RANGE: i32 = 128;
25/// Vanilla `findLightningTargetAround` horizontal inflate of the search box.
26const LIVING_TARGET_INFLATE: f64 = 3.0;
27/// Vanilla `difficulty.getEffectiveDifficulty() * 0.01` trap chance factor.
28const TRAP_CHANCE_PER_DIFFICULTY: f64 = 0.01;
29
30impl World {
31    /// Vanilla `ServerLevel.tickThunder`, run for each spawning chunk in
32    /// entity-ticking range.
33    pub(crate) fn tick_thunder(self: &Arc<Self>, chunk_pos: ChunkPos) {
34        if !self.is_raining() || !self.is_thundering() || rand::random_range(0..THUNDER_CHANCE) != 0
35        {
36            return;
37        }
38        let mut random_positions = BlockRandomPositionGenerator::from_runtime_rng();
39        let origin = random_positions.next_pos(chunk_pos.0.x << 4, 0, chunk_pos.0.y << 4);
40        let Some(pos) = self.find_lightning_target_around(origin) else {
41            return;
42        };
43        if !self.is_raining_at(pos) {
44            return;
45        }
46
47        let difficulty = self.current_difficulty_at(pos);
48        let is_trap = self.get_game_rule(&SPAWN_MOBS)
49            && rand::random::<f64>()
50                < f64::from(difficulty.effective_difficulty()) * TRAP_CHANCE_PER_DIFFICULTY
51            && !self
52                .get_block_state(pos.below())
53                .get_block()
54                .has_tag(&BlockTag::LIGHTNING_RODS);
55        if is_trap {
56            self.spawn_skeleton_horse_trap(pos);
57        }
58
59        let strike = DVec3::new(
60            f64::from(pos.x()) + 0.5,
61            f64::from(pos.y()),
62            f64::from(pos.z()) + 0.5,
63        );
64        let Ok(bolt) = create_entity_instance(self, &vanilla_entities::LIGHTNING_BOLT, strike)
65        else {
66            return;
67        };
68        bolt.snap_to(strike, 0.0, 0.0);
69        if let Some(bolt) = bolt.downcast_ref::<LightningBoltEntity>() {
70            bolt.set_visual_only(is_trap);
71        }
72        if let Err(error) = self.try_add_entity(bolt) {
73            log::debug!("world rejected a storm lightning bolt: {error}");
74        }
75    }

77    /// The trap half of vanilla `tickThunder`. Vanilla does not run
78    /// `finalizeSpawn` here and places the horse at the block corner.
79    fn spawn_skeleton_horse_trap(self: &Arc<Self>, pos: BlockPos) {
80        let corner = DVec3::new(f64::from(pos.x()), f64::from(pos.y()), f64::from(pos.z()));
81        let Ok(horse) = create_entity_instance(self, &vanilla_entities::SKELETON_HORSE, corner)
82        else {
83            return;
84        };
85        if let Some(skeleton_horse) = horse.downcast_ref::<SkeletonHorseEntity>() {
86            skeleton_horse.set_trap(true);
87            skeleton_horse.set_age(0);
88        }
89        if let Err(error) = self.try_add_entity(horse) {
90            log::debug!("world rejected a skeleton horse trap: {error}");
91        }
92    }

94    /// Vanilla `ServerLevel.findLightningTargetAround`. `None` when the
95    /// column is not loaded.
96    fn find_lightning_target_around(&self, pos: BlockPos) -> Option<BlockPos> {
97        let height = self.height_at(HeightmapType::MotionBlocking, pos.x(), pos.z())?;
98        let mut center = pos.at_y(height);
99        if let Some(rod) = self.find_lightning_rod(center) {
100            return Some(rod);
101        }
102
103        let search = WorldAabb::new(
104            f64::from(center.x()),
105            f64::from(center.y()),
106            f64::from(center.z()),
107            f64::from(center.x()) + 1.0,
108            f64::from(self.get_max_y() + 1) + 1.0,
109            f64::from(center.z()) + 1.0,
110        )
111        .inflate(LIVING_TARGET_INFLATE);
112        let entities = self.get_entities_in_aabb_matching(&search, |entity| {
113            entity.as_living_entity().is_some()
114                && entity.is_alive()
115                && self.can_see_sky(entity.block_position())
116        });
117        if let Some(target) = entities.choose(&mut rand::rng()) {
118            return Some(target.block_position());
119        }
120
121        if center.y() == self.get_min_y() - 1 {
122            center = center.above_n(2);
123        }
124        Some(center)
125    }

127    /// Vanilla `ServerLevel.findLightningRod`: the closest lightning rod on
128    /// the world surface within 128 blocks, struck from the block above.
129    fn find_lightning_rod(&self, center: BlockPos) -> Option<BlockPos> {
130        let lightning_rod_type = vanilla_poi_types::LIGHTNING_ROD.try_id()?;
131        let radius_sqr = i64::from(LIGHTNING_ROD_RANGE) * i64::from(LIGHTNING_ROD_RANGE);
132        let candidates = self.poi_storage.lock().get_in_horizontal_square(
133            &|type_id| type_id == lightning_rod_type,
134            center,
135            LIGHTNING_ROD_RANGE,
136            OccupationStatus::Any,
137        );
138        candidates
139            .into_iter()
140            .map(|(pos, _)| pos)
141            .filter(|pos| dist_sqr(*pos, center) <= radius_sqr)
142            .filter(|pos| {
143                self.height_at(HeightmapType::WorldSurface, pos.x(), pos.z())
144                    .is_some_and(|height| pos.y() == height - 1)
145            })
146            .min_by_key(|pos| dist_sqr(*pos, center))
147            .map(|pos| pos.above())
148    }
149}
150
151/// Vanilla `Vec3i.distSqr` between two block positions.
152fn dist_sqr(a: BlockPos, b: BlockPos) -> i64 {
153    let dx = i64::from(a.x() - b.x());
154    let dy = i64::from(a.y() - b.y());
155    let dz = i64::from(a.z() - b.z());
156    dx * dx + dy * dy + dz * dz
157}

159#[cfg(test)]
160mod tests {
161    use steel_registry::{init_vanilla_registry, vanilla_blocks};
162    use steel_utils::types::UpdateFlags;
163    use steel_utils::{BlockPos, ChunkPos};
164
165    use crate::behavior::init_behaviors;
166    use crate::test_support::{fresh_test_world, insert_ready_full_chunk};
167
168    #[test]
169    fn storm_lightning_targets_the_surface_rod() {
170        init_vanilla_registry();
171        init_behaviors();
172        let fixture = fresh_test_world("thunder_rod");
173        let world = &fixture.world;
174        insert_ready_full_chunk(world, ChunkPos::new(0, 0));
175        let rod = BlockPos::new(3, 70, 12);
176        world.set_block(
177            rod,
178            vanilla_blocks::LIGHTNING_ROD.default_state(),
179            UpdateFlags::UPDATE_ALL,
180        );
181        // Ground under the strike column keeps the rod within 128 blocks.
182        world.set_block(
183            BlockPos::new(10, 69, 4),
184            vanilla_blocks::STONE.default_state(),
185            UpdateFlags::UPDATE_ALL,
186        );
187
188        assert_eq!(
189            world.find_lightning_target_around(BlockPos::new(10, 0, 4)),
190            Some(rod.above())
191        );
192    }
193
194    #[test]
195    fn buried_rod_is_ignored() {
196        init_vanilla_registry();
197        init_behaviors();
198        let fixture = fresh_test_world("thunder_buried_rod");
199        let world = &fixture.world;
200        insert_ready_full_chunk(world, ChunkPos::new(0, 0));

202        world.set_block(

209            vanilla_blocks::STONE.default_state(),

213        assert_eq!(world.find_lightning_rod(BlockPos::new(10, 70, 4)), None);
