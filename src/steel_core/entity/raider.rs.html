1//! Raid members (Vanilla `Raider`) and the ominous banner they carry.
2//!
3//! Raids themselves come later; until then a raider never has a current raid,
4//! so the raid-only behavior (joining raids, banner pickup, celebrating, raid
5//! goals) stays inactive exactly as it is for a raider outside a raid.

27const TAG_WAVE: &str = "Wave";
28const TAG_CAN_JOIN_RAID: &str = "CanJoinRaid";
29/// Vanilla `Raider.updateNoActionTime`: raiders idle twice as fast as other mobs.
30const RAIDER_NO_ACTION_TIME_INCREMENT: i32 = 2;
31
32/// Vanilla `Raid.getOminousBannerInstance`: the white banner raid captains wear.
33#[must_use]
34pub fn ominous_banner_instance() -> ItemStack {
35    let layer = |pattern, color| BannerPatternLayer::new(RegistryHolder::reference(pattern), color);
36    let patterns = BannerPatternLayers::new(vec![
37        layer(&vanilla_banner_patterns::RHOMBUS, DyeColor::Cyan),
38        layer(&vanilla_banner_patterns::STRIPE_BOTTOM, DyeColor::LightGray),
39        layer(&vanilla_banner_patterns::STRIPE_CENTER, DyeColor::Gray),
40        layer(&vanilla_banner_patterns::BORDER, DyeColor::LightGray),
41        layer(&vanilla_banner_patterns::STRIPE_MIDDLE, DyeColor::Black),
42        layer(
43            &vanilla_banner_patterns::HALF_HORIZONTAL,
44            DyeColor::LightGray,
45        ),
46        layer(&vanilla_banner_patterns::CIRCLE, DyeColor::LightGray),
47        layer(&vanilla_banner_patterns::BORDER, DyeColor::Black),
48    ]);
49    let mut banner = ItemStack::new(&vanilla_items::WHITE_BANNER);
50    banner.set(BANNER_PATTERNS, patterns);
51    banner.set(
52        TOOLTIP_DISPLAY,
53        TooltipDisplay::DEFAULT.with_hidden(BANNER_PATTERNS, true),
54    );
55    banner.set(
56        ITEM_NAME,
57        TextComponent::translated(translations::BLOCK_MINECRAFT_OMINOUS_BANNER.msg()),
58    );
59    banner.set(RARITY, Rarity::Uncommon);
60    banner
61}

63/// Raid fields of a raider.
64pub struct RaiderState {
65    wave: AtomicI32,
66    can_join_raid: AtomicBool,
67    ticks_outside_raid: AtomicI32,
68}
69
70impl RaiderState {
71    #[must_use]
72    pub(crate) const fn new() -> Self {
73        Self {
74            wave: AtomicI32::new(0),
75            can_join_raid: AtomicBool::new(false),
76            ticks_outside_raid: AtomicI32::new(0),
77        }
78    }
79}
80
81impl Default for RaiderState {
82    fn default() -> Self {
83        Self::new()
84    }
85}

87/// Vanilla `Raider`.
88// TODO: Raids: current raid, `getRaidAt` joining in `aiStep`, leader/hero bookkeeping in
89// `die`, banner pickup, boss bar on hurt, and the `ObtainRaidLeaderBannerGoal`,
90// `PathfindToRaidGoal`, `RaiderMoveThroughVillageGoal` and `RaiderCelebration` goals.
91pub trait Raider: PatrollingMonster {
92    fn raider_state(&self) -> &RaiderState;
93
94    /// Vanilla `IS_CELEBRATING` synced flag.
95    fn is_celebrating(&self) -> bool;
96
97    fn set_celebrating(&self, celebrating: bool);
98
99    /// Vanilla `getCelebrateSound`.
100    fn celebrate_sound(&self) -> SoundEventRef;
101
102    fn can_join_raid(&self) -> bool {
103        self.raider_state().can_join_raid.load(Ordering::Relaxed)
104    }
105
106    fn set_can_join_raid(&self, can_join_raid: bool) {
107        self.raider_state()
108            .can_join_raid
109            .store(can_join_raid, Ordering::Relaxed);
110    }
111
112    fn wave(&self) -> i32 {
113        self.raider_state().wave.load(Ordering::Relaxed)
114    }
115
116    fn set_wave(&self, wave: i32) {
117        self.raider_state().wave.store(wave, Ordering::Relaxed);
118    }
119
120    fn ticks_outside_raid(&self) -> i32 {
121        self.raider_state()
122            .ticks_outside_raid
123            .load(Ordering::Relaxed)
124    }
125
126    fn set_ticks_outside_raid(&self, ticks: i32) {
127        self.raider_state()
128            .ticks_outside_raid
129            .store(ticks, Ordering::Relaxed);
130    }
131
132    /// Vanilla `hasActiveRaid`.
133    fn has_active_raid(&self) -> bool {
134        // TODO: `getCurrentRaid() != null && getCurrentRaid().isActive()` once raids exist.
135        false
136    }
137
138    /// Vanilla `isCaptain`: a patrol leader wearing the ominous banner.
139    fn is_captain(&self) -> bool {
140        let has_captain_banner = {
141            let equipment = self.living_base().equipment().lock();
142            let banner = equipment.get_ref(EquipmentSlot::Head);
143            !banner.is_empty() && ItemStack::matches(banner, &ominous_banner_instance())
144        };
145        has_captain_banner && self.is_patrol_leader()
146    }
147
148    /// Vanilla `Raider.canJoinPatrol`.
149    fn can_join_patrol_raider(&self) -> bool {
150        !self.has_active_raid()
151    }
152
153    /// Vanilla `Raider.updateNoActionTime`.
154    fn update_no_action_time_raider(&self) {
155        self.set_no_action_time(self.no_action_time() + RAIDER_NO_ACTION_TIME_INCREMENT);
156    }
157
158    /// Vanilla `Raider.removeWhenFarAway`: no current raid, so the patrol rule applies.
159    fn remove_when_far_away_raider(&self, dist_sqr: f64) -> bool {
160        self.remove_when_far_away_patrolling(dist_sqr)
161    }
162
163    /// The raider and patrol part of Vanilla `finalizeSpawn`, run before the mob's own.
164    fn finalize_spawn_raider(&self, spawn_reason: EntitySpawnReason) {
165        self.set_can_join_raid(
166            self.entity_type() != &vanilla_entities::WITCH
167                || spawn_reason != EntitySpawnReason::Natural,
168        );
169        self.finalize_spawn_patrolling(spawn_reason);
170    }
171
172    fn save_raider(&self, nbt: &mut NbtCompound) {
173        self.save_patrolling(nbt);
174        nbt.insert(TAG_WAVE, self.wave());
175        nbt.insert(TAG_CAN_JOIN_RAID, i8::from(self.can_join_raid()));
176    }
177
178    fn load_raider(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
179        self.load_patrolling(nbt);
180        self.set_wave(nbt.int(TAG_WAVE).unwrap_or(0));
181        self.set_can_join_raid(nbt.byte(TAG_CAN_JOIN_RAID).is_some_and(|b| b != 0));
182    }
183}
