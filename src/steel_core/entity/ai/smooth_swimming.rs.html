2//! by dolphins, axolotls, nautiluses and tadpoles.

15const FULL_SPEED_TURN_THRESHOLD: f32 = 10.0;
16const STOP_TURN_THRESHOLD: f32 = 60.0;
17const MIN_DISTANCE_SQR: f64 = 2.500_000_3E-7_f32 as f64;
18const ANGLE_EPSILON: f64 = 1.0E-5_f32 as f64;
19const X_ROT_STEP: f32 = 5.0;
20const HEAD_TILT_X: f32 = 10.0;
21const HEAD_TILT_Y: f32 = 20.0;
22const BODY_CATCH_UP: f32 = 4.0;
23const IN_WATER_LIFT: f64 = 0.005;
24
25/// Vanilla `(float)(Mth.atan2(a, b) * 180.0F / (float)Math.PI)`.
26fn atan2_degrees(a: f64, b: f64) -> f32 {
27    (a.atan2(b) * 180.0 / f64::from(PI)) as f32
28}
29
30/// Vanilla `x * (float)(Math.PI / 180.0)` on a float angle.
31fn to_radians(degrees: f32) -> f32 {
32    degrees * (PI_F64 / 180.0) as f32
33}

35/// Settings of a vanilla `SmoothSwimmingMoveControl`.
36#[derive(Debug, Clone, Copy)]
37pub(crate) struct SmoothSwimmingMoveControl {
38    /// Largest pitch, in degrees, the mob swims at.
39    pub max_turn_x: i32,
40    /// Largest yaw change per tick, in degrees.
41    pub max_turn_y: i32,
42    /// Speed factor while in water.
43    pub in_water_speed_modifier: f32,
44    /// Speed factor while out of water.
45    pub outside_water_speed_modifier: f32,
46    /// Whether the mob drifts up a little each tick in water.
47    pub apply_gravity: bool,
48}
49
50impl SmoothSwimmingMoveControl {
51    /// Vanilla `getTurningSpeedFactor`: slows down on sharp turns on land.
52    fn turning_speed_factor(left_to_turn: f32) -> f32 {
53        1.0 - ((left_to_turn - FULL_SPEED_TURN_THRESHOLD)
54            / (STOP_TURN_THRESHOLD - FULL_SPEED_TURN_THRESHOLD))
55            .clamp(0.0, 1.0)
56    }
57
58    /// Vanilla `SmoothSwimmingMoveControl.tick`. It keeps `MOVE_TO` until
59    /// navigation finishes, unlike the base move control.
60    pub(crate) fn tick<M: Mob + ?Sized>(self, mob: &M) {
61        if self.apply_gravity && mob.is_in_water() {
62            mob.set_velocity(mob.velocity() + DVec3::new(0.0, IN_WATER_LIFT, 0.0));
63        }
64
65        let move_control = mob.mob_base().controls().lock().move_control;
66        let navigating = !mob.mob_base().navigation().lock().is_done();
67        if !matches!(move_control.operation(), MoveControlOperation::MoveTo) || !navigating {
68            mob.set_mob_speed(0.0);
69            mob.set_travel_input(LivingTravelInput::new(0.0, 0.0, 0.0));
70            return;
71        }
72
73        let position = mob.position();
74        let wanted = move_control.wanted_position();
75        let xd = wanted.x - position.x;
76        let yd = wanted.y - position.y;
77        let zd = wanted.z - position.z;
78        let dd = xd * xd + yd * yd + zd * zd;
79        if dd < MIN_DISTANCE_SQR {
80            let input = mob.travel_input();
81            mob.set_travel_input(LivingTravelInput::new(
82                input.sideways(),
83                input.vertical(),
84                0.0,
85            ));
86            return;
87        }
88
89        let y_rot_d = atan2_degrees(zd, xd) - 90.0;
90        let (y_rot, x_rot) = mob.rotation();
91        let y_rot = rotlerp(y_rot, y_rot_d, self.max_turn_y as f32);
92        mob.set_rotation((y_rot, x_rot));
93        mob.set_y_body_rot(y_rot);
94        mob.set_y_head_rot(y_rot);
95
96        let movement_speed = mob
97            .attributes()
98            .lock()
99            .required_value(vanilla_attributes::MOVEMENT_SPEED);
100        let speed = (move_control.speed_modifier() * movement_speed) as f32;
101        if mob.is_in_water() {
102            mob.set_mob_speed(speed * self.in_water_speed_modifier);
103            let horizontal = (xd * xd + zd * zd).sqrt();
104            let mut x_rot = x_rot;
105            if yd.abs() > ANGLE_EPSILON || horizontal.abs() > ANGLE_EPSILON {
106                let max_turn_x = self.max_turn_x as f32;
107                let x_rot_d =
108                    wrap_degrees(-atan2_degrees(yd, horizontal)).clamp(-max_turn_x, max_turn_x);
109                x_rot = rotate_towards(x_rot, x_rot_d, X_ROT_STEP);
110                mob.set_rotation((y_rot, x_rot));
111            }
112
113            let cos = to_radians(x_rot).cos();
114            let sin = to_radians(x_rot).sin();
115            let input = mob.travel_input();
116            mob.set_travel_input(LivingTravelInput::new(
117                input.sideways(),
118                -sin * speed,
119                cos * speed,
120            ));
121        } else {
122            let left_to_turn = wrap_degrees(y_rot - y_rot_d).abs();
123            mob.set_mob_speed(
124                speed
125                    * self.outside_water_speed_modifier
126                    * Self::turning_speed_factor(left_to_turn),
127            );
128        }
129    }
130}

132/// Vanilla `LookControl.getYRotD`.
133fn look_y_rot_d<M: Mob + ?Sized>(mob: &M, wanted: DVec3) -> Option<f32> {
134    let position = mob.position();
135    let xd = wanted.x - position.x;
136    let zd = wanted.z - position.z;
137    (zd.abs() > ANGLE_EPSILON || xd.abs() > ANGLE_EPSILON).then(|| atan2_degrees(zd, xd) - 90.0)
138}
139
140/// Vanilla `LookControl.getXRotD`.
141fn look_x_rot_d<M: Mob + ?Sized>(mob: &M, wanted: DVec3) -> Option<f32> {
142    let position = mob.position();
143    let xd = wanted.x - position.x;
144    let yd = wanted.y - mob.get_eye_y();
145    let zd = wanted.z - position.z;
146    let sd = (xd * xd + zd * zd).sqrt();
147    (yd.abs() > ANGLE_EPSILON || sd.abs() > ANGLE_EPSILON).then(|| -atan2_degrees(yd, sd))
148}

150/// Vanilla `SmoothSwimmingLookControl.tick`: the head leads the body, tilted a
151/// little, and the body follows once the head turns past
152/// `max_y_rot_from_center`.
153pub(crate) fn tick_smooth_swimming_look_control<M: Mob + ?Sized>(
154    mob: &M,
155    max_y_rot_from_center: i32,
156) {
157    let (looking, look_control) = {
158        let mut controls = mob.mob_base().controls().lock();
159        let looking = controls.look_control.tick_cooldown();
160        (looking, controls.look_control)
161    };
162
163    if looking {
164        let wanted = look_control.wanted_position();
165        if let Some(y_rot_d) = look_y_rot_d(mob, wanted) {
166            mob.set_y_head_rot(rotate_towards(
167                mob.y_head_rot(),
168                y_rot_d + HEAD_TILT_Y,
169                look_control.y_max_rot_speed(),
170            ));
171        }
172        if let Some(x_rot_d) = look_x_rot_d(mob, wanted) {
173            let (y_rot, x_rot) = mob.rotation();
174            mob.set_rotation((
175                y_rot,
176                rotate_towards(x_rot, x_rot_d + HEAD_TILT_X, look_control.x_max_rot_angle()),
177            ));
178        }
179    } else {
180        if mob.mob_base().navigation().lock().is_done() {
181            let (y_rot, x_rot) = mob.rotation();
182            mob.set_rotation((y_rot, rotate_towards(x_rot, 0.0, X_ROT_STEP)));
183        }
184        mob.set_y_head_rot(rotate_towards(
185            mob.y_head_rot(),
186            mob.y_body_rot(),
187            look_control.y_max_rot_speed(),
188        ));
189    }
190
191    let head_diff_body = wrap_degrees(mob.y_head_rot() - mob.y_body_rot());
192    let max_from_center = max_y_rot_from_center as f32;
193    if head_diff_body < -max_from_center {
194        mob.set_y_body_rot(mob.y_body_rot() - BODY_CATCH_UP);
195    } else if head_diff_body > max_from_center {
196        mob.set_y_body_rot(mob.y_body_rot() + BODY_CATCH_UP);
197    }
198}
