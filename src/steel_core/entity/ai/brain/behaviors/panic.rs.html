18const PANIC_ENTRY: [MemoryRequirement; 2] = [
19    MemoryRequirement::new(&IS_PANICKING, MemoryStatus::Registered),
20    MemoryRequirement::new(&HURT_BY, MemoryStatus::Registered),
21];

23type PanicCauses<E> = Box<dyn Fn(&E) -> Identifier + Send>;
24type PositionGetter<E> = Box<dyn Fn(&E) -> Option<DVec3> + Send>;

26/// Runs from danger after panic-causing damage (vanilla `AnimalPanic`).
27pub struct AnimalPanic<E: ?Sized> {
28    speed_multiplier: f32,
29    panic_causing_damage_types: PanicCauses<E>,
30    position_getter: PositionGetter<E>,
31}
32
33impl<E: PathfinderMob + ?Sized> AnimalPanic<E> {
34    /// Panics on `#panic_causes` and flees to a random land position.
35    #[must_use]
36    pub fn new(speed_multiplier: f32) -> Self {
37        Self::with_damage_types(speed_multiplier, |_| DamageTypeTag::PANIC_CAUSES)
38    }
39
40    /// Panics on the damage tag chosen per body and flees to a random land
41    /// position.
42    #[must_use]
43    pub fn with_damage_types(
44        speed_multiplier: f32,
45        panic_causing_damage_types: impl Fn(&E) -> Identifier + Send + 'static,
46    ) -> Self {
47        Self::with_position_getter(speed_multiplier, panic_causing_damage_types, |body| {
48            land_random_pos(body, 5, 4)
49        })
50    }
51
52    #[must_use]
53    pub fn with_position_getter(
54        speed_multiplier: f32,
55        panic_causing_damage_types: impl Fn(&E) -> Identifier + Send + 'static,
56        position_getter: impl Fn(&E) -> Option<DVec3> + Send + 'static,
57    ) -> Self {
58        Self {
59            speed_multiplier,
60            panic_causing_damage_types: Box::new(panic_causing_damage_types),
61            position_getter: Box::new(position_getter),
62        }
63    }
64
65    fn panic_pos(&self, world: &World, body: &E) -> Option<DVec3> {
66        if body.is_on_fire()
67            && let Some(water) = look_for_water(world, body)
68        {
69            return Some(DVec3::new(
70                f64::from(water.x()) + 0.5,
71                f64::from(water.y()),
72                f64::from(water.z()) + 0.5,
73            ));
74        }
75        (self.position_getter)(body)
76    }
77}

79impl<E: PathfinderMob + ?Sized> Behavior<E> for AnimalPanic<E> {
80    fn entry_condition(&self) -> &[MemoryRequirement] {
81        &PANIC_ENTRY
82    }
83
84    fn duration(&self) -> (i32, i32) {
85        (100, 120)
86    }
87
88    fn check_extra_start_conditions(&mut self, _world: &Arc<World>, body: &E) -> bool {
89        let Some(memories) = body.brain_memories() else {
90            return false;
91        };
92        let tag = (self.panic_causing_damage_types)(body);
93        let memories = memories.lock();
94        memories
95            .get_ref(&HURT_BY)
96            .is_some_and(|source| source.is(&tag))
97            || memories.has_value(&IS_PANICKING)
98    }
99
100    fn can_still_use(&mut self, _world: &Arc<World>, _body: &E, _time: i64) -> bool {
101        true
102    }
103
104    fn start(&mut self, _world: &Arc<World>, body: &E, _time: i64) {
105        if let Some(memories) = body.brain_memories() {
106            let mut memories = memories.lock();
107            memories.set(&IS_PANICKING, Some(true));
108            memories.erase(&WALK_TARGET);
109        }
110        body.mob_base().navigation().lock().stop();
111    }
112
113    fn stop(&mut self, _world: &Arc<World>, body: &E, _time: i64) {
114        if let Some(memories) = body.brain_memories() {
115            memories.lock().erase(&IS_PANICKING);
116        }
117    }
118
119    fn tick(&mut self, world: &Arc<World>, body: &E, _time: i64) {
120        if !body.mob_base().navigation().lock().is_done() {
121            return;
122        }
123        if let Some(position) = self.panic_pos(world, body)
124            && let Some(memories) = body.brain_memories()
125        {
126            memories.lock().set(
127                &WALK_TARGET,
128                Some(WalkTarget::to_point(position, self.speed_multiplier, 0)),
129            );
130        }
131    }
132}

134/// Vanilla `AnimalPanic.lookForWater`: the nearest water within 5 blocks
135/// horizontally and 1 vertically, in Manhattan order; two-block-wide mobs
136/// need a 2x2 pool.
137fn look_for_water(world: &World, body: &(impl LivingEntity + ?Sized)) -> Option<BlockPos> {
138    let mob_position = body.block_position();
139    let state = world.get_block_state(mob_position);
140    if !BLOCK_BEHAVIORS
141        .get_behavior(state.get_block())
142        .get_collision_shape(state, world, mob_position, BlockCollisionContext::empty())
143        .is_empty()
144    {
145        return None;
146    }
147
148    let is_water = |pos: BlockPos| world.get_block_state(pos).get_fluid_state().is_water();
149    #[expect(
150        clippy::cast_possible_truncation,
151        reason = "vanilla Mth.ceil of the bounding box width"
152    )]
153    let wide = body.bounding_box().width().ceil() as i32 == 2;
154    mob_position.within_manhattan(5, 1, 5).find(|&pos| {
155        is_water(pos)
156            && (!wide
157                || is_water(pos.south()) && is_water(pos.east()) && is_water(pos.south().east()))
158    })
159}
