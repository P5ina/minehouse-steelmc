15const BREED_RANGE: f64 = 3.0;
16const MIN_DURATION: i32 = 60;
17const MAX_DURATION: i32 = 110;

19const MAKE_LOVE_ENTRY: [MemoryRequirement; 5] = [
20    MemoryRequirement::new(&NEAREST_VISIBLE_LIVING_ENTITIES, MemoryStatus::ValuePresent),
21    MemoryRequirement::new(&BREED_TARGET, MemoryStatus::ValueAbsent),
22    MemoryRequirement::new(&WALK_TARGET, MemoryStatus::Registered),
23    MemoryRequirement::new(&LOOK_TARGET, MemoryStatus::Registered),
24    MemoryRequirement::new(&IS_PANICKING, MemoryStatus::ValueAbsent),

27/// Walks to a visible partner in love and breeds (vanilla `AnimalMakeLove`).
28pub struct AnimalMakeLove {
29    partner_type: EntityTypeRef,
30    speed_modifier: f32,
31    close_enough_distance: i32,
32    spawn_child_at_time: i64,
33}
34
35impl AnimalMakeLove {
36    #[must_use]
37    pub const fn new(partner_type: EntityTypeRef) -> Self {
38        Self::with_speed(partner_type, 1.0, 2)
39    }
40
41    #[must_use]
42    pub const fn with_speed(
43        partner_type: EntityTypeRef,
44        speed_modifier: f32,
45        close_enough_distance: i32,
46    ) -> Self {
47        Self {
48            partner_type,
49            speed_modifier,
50            close_enough_distance,
51            spawn_child_at_time: 0,
52        }
53    }
54
55    fn find_valid_breed_partner(&self, body: &(impl Animal + ?Sized)) -> Option<SharedEntity> {
56        // The memories lock must be released before searching: visibility
57        // checks read the body's attack target.
58        let visible = body
59            .brain_memories()?
60            .lock()
61            .get(&NEAREST_VISIBLE_LIVING_ENTITIES)?;
62        visible.find_closest(|entity| {
63            entity.entity_type() == self.partner_type
64                && entity
65                    .as_animal()
66                    .is_some_and(|animal| body.can_mate(animal) && !entity_is_panicking(entity))
67        })
68    }
69
70    fn breed_target(&self, body: &(impl LivingEntity + ?Sized)) -> Option<SharedEntity> {
71        body.brain_memories()?
72            .lock()
73            .get(&BREED_TARGET)
74            .filter(|partner| partner.entity_type() == self.partner_type)
75    }
76}
77
78fn entity_is_panicking(entity: &SharedEntity) -> bool {
79    entity
80        .as_pathfinder_mob()
81        .is_some_and(PathfinderMob::is_panicking)
82}
83
84impl<E: Animal + PathfinderMob> Behavior<E> for AnimalMakeLove {
85    fn entry_condition(&self) -> &[MemoryRequirement] {
86        &MAKE_LOVE_ENTRY
87    }
88
89    fn duration(&self) -> (i32, i32) {
90        (MAX_DURATION, MAX_DURATION)
91    }
92
93    fn check_extra_start_conditions(&mut self, _world: &Arc<World>, body: &E) -> bool {
94        body.is_in_love() && self.find_valid_breed_partner(body).is_some()
95    }
96
97    fn start(&mut self, world: &Arc<World>, body: &E, time: i64) {
98        let Some(partner) = self.find_valid_breed_partner(body) else {
99            return;
100        };
101        let Some(body_entity) = world.get_entity_by_id(body.id()) else {
102            return;
103        };
104        if let Some(memories) = body.brain_memories() {
105            memories
106                .lock()
107                .set(&BREED_TARGET, Some(Arc::clone(&partner)));
108        }
109        if let Some(memories) = partner
110            .as_living_entity()
111            .and_then(LivingEntity::brain_memories)
112        {
113            memories
114                .lock()
115                .set(&BREED_TARGET, Some(Arc::clone(&body_entity)));
116        }
117        lock_gaze_and_walk_to_each_other(
118            &body_entity,
119            &partner,
120            self.speed_modifier,
121            self.close_enough_distance,
122        );
123        let duration = MIN_DURATION + rand::random_range(0..MAX_DURATION - MIN_DURATION);
124        self.spawn_child_at_time = time + i64::from(duration);
125    }
126
127    fn can_still_use(&mut self, _world: &Arc<World>, body: &E, time: i64) -> bool {
128        let Some(partner) = self.breed_target(body) else {
129            return false;
130        };
131        partner.is_alive()
132            && partner
133                .as_animal()
134                .is_some_and(|animal| body.can_mate(animal))
135            && entity_is_visible(body, &partner)
136            && time <= self.spawn_child_at_time
137            && !body.is_panicking()
138            && !entity_is_panicking(&partner)
139    }
140
141    fn tick(&mut self, world: &Arc<World>, body: &E, time: i64) {
142        let Some(partner) = self.breed_target(body) else {
143            return;
144        };
145        let Some(body_entity) = world.get_entity_by_id(body.id()) else {
146            return;
147        };
148        lock_gaze_and_walk_to_each_other(
149            &body_entity,
150            &partner,
151            self.speed_modifier,
152            self.close_enough_distance,
153        );
154        if body.position().distance_squared(partner.position()) >= BREED_RANGE * BREED_RANGE
155            || time < self.spawn_child_at_time
156        {
157            return;
158        }
159        let Some(partner_animal) = partner.as_animal() else {
160            return;
161        };
162        body.spawn_child_from_breeding(world, partner_animal);
163        if let Some(memories) = body.brain_memories() {
164            memories.lock().erase(&BREED_TARGET);
165        }
166        if let Some(memories) = partner_animal.brain_memories() {
167            memories.lock().erase(&BREED_TARGET);
168        }
169    }
170
171    fn stop(&mut self, _world: &Arc<World>, body: &E, _time: i64) {
172        if let Some(memories) = body.brain_memories() {
173            let mut memories = memories.lock();
174            memories.erase(&BREED_TARGET);
175            memories.erase(&WALK_TARGET);
176            memories.erase(&LOOK_TARGET);
177        }
178        self.spawn_child_at_time = 0;
179    }
180}
