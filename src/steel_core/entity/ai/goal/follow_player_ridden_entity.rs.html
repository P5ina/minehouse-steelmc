12const SEARCH_INFLATE: f64 = 5.0;
13const GO_TO_ENTITY_SPEED: f32 = 0.015;
14const GO_IN_ENTITY_DIRECTION_SPEED: f32 = 0.01;
15const RECALC_PATH_TICKS: i32 = 10;
16const CATCH_UP_DISTANCE: f32 = 4.0;
17const FALL_BEHIND_DISTANCE: f32 = 12.0;
18const LEAD_BLOCKS: i32 = 10;
19const MOVED_EPSILON: f64 = 1.0E-5_f32 as f64;
20
21/// Vanilla `Entity.hasMovedHorizontallyRecently`.
22fn has_moved_horizontally_recently(entity: &dyn Entity) -> bool {
23    let speed = entity.known_speed();
24    (speed.x * speed.x + speed.z * speed.z).sqrt().abs() > MOVED_EPSILON
25}

27/// Vanilla `Entity.getDirection` (and `getMotionDirection`).
28fn facing(entity: &dyn Entity) -> Direction {
29    Direction::from_yaw(entity.rotation().0)
30}

32#[derive(Clone, Copy, PartialEq, Eq)]
33enum FollowEntityGoal {
34    GoToEntity,
35    GoInEntityDirection,
36}
37
38/// Vanilla `FollowPlayerRiddenEntityGoal`: swims along with a vehicle a player
39/// is steering. `vehicle` stands in for vanilla's entity class filter.
40pub(crate) struct FollowPlayerRiddenEntityGoal {
41    time_to_recalc_path: i32,
42    vehicle: fn(&dyn Entity) -> bool,
43    following: Option<SharedEntity>,
44    current_goal: FollowEntityGoal,
45}
46
47impl FollowPlayerRiddenEntityGoal {
48    #[must_use]
49    pub(crate) const fn new(vehicle: fn(&dyn Entity) -> bool) -> Self {
50        Self {
51            time_to_recalc_path: 0,
52            vehicle,
53            following: None,
54            current_goal: FollowEntityGoal::GoToEntity,
55        }
56    }
57
58    fn nearby_controllers(&self, mob: &dyn PathfinderMob) -> Vec<SharedEntity> {
59        let Some(world) = mob.level() else {
60            return Vec::new();
61        };
62        let area = mob.bounding_box().inflate(SEARCH_INFLATE);
63        world
64            .get_entities_in_aabb_matching(&area, |entity| (self.vehicle)(entity))
65            .into_iter()
66            .filter_map(|vehicle| vehicle.controlling_passenger())
67            .filter(|controller| controller.as_player().is_some())
68            .collect()
69    }
70}
71
72impl Goal for FollowPlayerRiddenEntityGoal {
73    fn controls(&self) -> GoalControls {
74        GoalControls::EMPTY
75    }
76
77    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
78        if self
79            .following
80            .as_ref()
81            .is_some_and(|following| has_moved_horizontally_recently(following.as_ref()))
82        {
83            return true;
84        }
85        self.nearby_controllers(mob)
86            .iter()
87            .any(|controller| has_moved_horizontally_recently(controller.as_ref()))
88    }
89
90    fn is_interruptable(&self) -> bool {
91        true
92    }
93
94    fn can_continue_to_use(&mut self, _mob: &dyn PathfinderMob) -> bool {
95        self.following.as_ref().is_some_and(|following| {
96            following.is_passenger() && has_moved_horizontally_recently(following.as_ref())
97        })
98    }
99
100    fn start(&mut self, mob: &dyn PathfinderMob) {
101        if let Some(controller) = self.nearby_controllers(mob).into_iter().next() {
102            self.following = Some(controller);
103        }
104        self.time_to_recalc_path = 0;
105        self.current_goal = FollowEntityGoal::GoToEntity;
106    }
107
108    fn stop(&mut self, _mob: &dyn PathfinderMob) {
109        self.following = None;
110    }
111
112    fn tick(&mut self, mob: &dyn PathfinderMob, entity: &SharedEntity) {
113        let speed = if self.current_goal == FollowEntityGoal::GoInEntityDirection {
114            GO_IN_ENTITY_DIRECTION_SPEED
115        } else {
116            GO_TO_ENTITY_SPEED
117        };
118        let input = mob.travel_input();
119        mob.move_relative(
120            speed,
121            DVec3::new(
122                f64::from(input.sideways()),
123                f64::from(input.vertical()),
124                f64::from(input.forward()),
125            ),
126        );
127        let _ = Arc::clone(entity).move_entity(MoverType::SelfMovement, mob.velocity());
128
129        self.time_to_recalc_path -= 1;
130        if self.time_to_recalc_path > 0 {
131            return;
132        }
133        self.time_to_recalc_path = reduced_tick_delay(RECALC_PATH_TICKS);
134        let Some(following) = self.following.clone() else {
135            return;
136        };
137
138        let distance = mob.position().distance(following.position()) as f32;
139        match self.current_goal {
140            FollowEntityGoal::GoToEntity => {
141                let behind = following
142                    .block_position()
143                    .relative(facing(following.as_ref()).opposite())
144                    .offset(0, -1, 0);
145                mob.move_to_pos(block_coords(behind), 1.0);
146                if distance < CATCH_UP_DISTANCE {
147                    self.time_to_recalc_path = 0;
148                    self.current_goal = FollowEntityGoal::GoInEntityDirection;
149                }
150            }
151            FollowEntityGoal::GoInEntityDirection => {
152                let direction = facing(following.as_ref());
153                let (step_x, step_y, step_z) = direction.offset();
154                let go_to = following.block_position().offset(
155                    step_x * LEAD_BLOCKS,
156                    step_y * LEAD_BLOCKS - 1,
157                    step_z * LEAD_BLOCKS,
158                );
159                mob.move_to_pos(block_coords(go_to), 1.0);
160                if distance > FALL_BEHIND_DISTANCE {
161                    self.time_to_recalc_path = 0;
162                    self.current_goal = FollowEntityGoal::GoToEntity;
163                }
164            }
165        }
166    }
167}
168
169fn block_coords(pos: BlockPos) -> DVec3 {
170    DVec3::new(f64::from(pos.x()), f64::from(pos.y()), f64::from(pos.z()))
171}
