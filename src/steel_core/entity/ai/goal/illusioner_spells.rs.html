1//! Illusioner spells (vanilla `Illusioner.IllusionerMirrorSpellGoal` and
2//! `Illusioner.IllusionerBlindnessSpellGoal`).
3
4use steel_registry::sound_event::SoundEventRef;
5use steel_registry::{sound_events, vanilla_mob_effects};
6use steel_utils::types::Difficulty;
7
8use super::spellcaster_goals::Spell;
9use crate::entity::{IllagerSpell, MobEffectInstance, PathfinderMob, SharedEntity};
10
11/// Vanilla `MobEffectInstance(MobEffects.INVISIBILITY, 1200)`.
12const MIRROR_INVISIBILITY_TICKS: i32 = 1200;
13/// Vanilla `MobEffectInstance(MobEffects.BLINDNESS, 400)`.
14const BLINDNESS_TICKS: i32 = 400;
15
16/// Vanilla `Illusioner.IllusionerMirrorSpellGoal`: turn invisible.
17pub(crate) struct MirrorSpell;
18
19impl MirrorSpell {
20    #[must_use]
21    pub(crate) const fn new() -> Self {
22        Self
23    }
24}
25
26impl Spell for MirrorSpell {
27    fn spell(&self) -> IllagerSpell {
28        IllagerSpell::Disappear
29    }
30
31    fn casting_time(&self) -> i32 {
32        20
33    }
34
35    fn casting_interval(&self) -> i32 {
36        340
37    }
38
39    fn prepare_sound(&self) -> Option<SoundEventRef> {
40        Some(&sound_events::ENTITY_ILLUSIONER_PREPARE_MIRROR)
41    }
42
43    fn can_use(
44        &mut self,
45        mob: &dyn PathfinderMob,
46        default_can_use: bool,
47        _next_attack_tick_count: i32,
48    ) -> bool {
49        default_can_use && !mob.has_mob_effect(vanilla_mob_effects::INVISIBILITY)
50    }
51
52    fn perform(&mut self, mob: &dyn PathfinderMob, _entity: &SharedEntity) {
53        mob.add_mob_effect(MobEffectInstance::with_duration(
54            vanilla_mob_effects::INVISIBILITY,
55            MIRROR_INVISIBILITY_TICKS,
56            0,
57        ));
58    }
59}

61/// Vanilla `Illusioner.IllusionerBlindnessSpellGoal`: blind each new target
62/// once, on hard local difficulty.
63pub(crate) struct BlindnessSpell {
64    last_target_id: i32,
65}
66
67impl BlindnessSpell {
68    #[must_use]
69    pub(crate) const fn new() -> Self {
70        Self { last_target_id: 0 }
71    }
72}
73
74impl Spell for BlindnessSpell {
75    fn spell(&self) -> IllagerSpell {
76        IllagerSpell::Blindness
77    }
78
79    fn casting_time(&self) -> i32 {
80        20
81    }
82
83    fn casting_interval(&self) -> i32 {
84        180
85    }
86
87    fn prepare_sound(&self) -> Option<SoundEventRef> {
88        Some(&sound_events::ENTITY_ILLUSIONER_PREPARE_BLINDNESS)
89    }
90
91    fn can_use(
92        &mut self,
93        mob: &dyn PathfinderMob,
94        default_can_use: bool,
95        _next_attack_tick_count: i32,
96    ) -> bool {
97        if !default_can_use {
98            return false;
99        }
100        let Some(target) = mob.target() else {
101            return false;
102        };
103        if target.id() == self.last_target_id {
104            return false;
105        }
106        let Some(world) = mob.level() else {
107            return false;
108        };
109        // VANILLA: `isHarderThan(Difficulty.NORMAL.ordinal())`, the ordinal as a float.
110        world
111            .current_difficulty_at(mob.block_position())
112            .is_harder_than(f32::from(Difficulty::Normal as u8))
113    }
114
115    fn start(&mut self, mob: &dyn PathfinderMob) {
116        if let Some(target) = mob.target() {
117            self.last_target_id = target.id();
118        }
119    }
120
121    fn perform(&mut self, mob: &dyn PathfinderMob, _entity: &SharedEntity) {
122        let Some(target) = mob.target() else {
123            return;
124        };
125        if let Some(living) = target.as_living_entity() {
126            living.add_mob_effect_with_source(
127                MobEffectInstance::with_duration(
128                    vanilla_mob_effects::BLINDNESS,
129                    BLINDNESS_TICKS,
130                    0,
131                ),
132                Some(mob),
133            );
134        }
135    }
136}
