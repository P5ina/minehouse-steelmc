1use steel_registry::vanilla_attributes;
2use steel_registry::vanilla_game_rules::UNIVERSAL_ANGER;
3use steel_utils::WorldAabb;
4
5use super::selector::{Goal, GoalControls};
6use crate::entity::PathfinderMob;
7
8const ALERT_RANGE_Y: f64 = 10.0;
9
10/// Vanilla `ResetUniversalAngerTargetGoal`: with universal anger on, a neutral mob
11/// hit by a player forgets its target and gets angry at every player.
12///
13/// The mob must implement [`crate::entity::NeutralMob`].
14pub(crate) struct ResetUniversalAngerTargetGoal {
15    alert_others_of_same_type: bool,
16    last_hurt_by_player_timestamp: i32,
17}
18
19impl ResetUniversalAngerTargetGoal {
20    #[must_use]
21    pub(crate) const fn new(alert_others_of_same_type: bool) -> Self {
22        Self {
23            alert_others_of_same_type,
24            last_hurt_by_player_timestamp: 0,
25        }
26    }
27
28    fn was_hurt_by_player(&self, mob: &dyn PathfinderMob) -> bool {
29        mob.last_hurt_by_mob()
30            .is_some_and(|attacker| attacker.as_player().is_some())
31            && mob.last_hurt_by_mob_timestamp() > self.last_hurt_by_player_timestamp
32    }
33
34    fn alert_nearby_mobs_of_same_type(mob: &dyn PathfinderMob) {
35        let Some(world) = mob.level() else {
36            return;
37        };
38        let within = mob
39            .attributes()
40            .lock()
41            .get_value(vanilla_attributes::FOLLOW_RANGE)
42            .unwrap_or(0.0);
43        let position = mob.position();
44        let search_box = WorldAabb::new(
45            position.x,
46            position.y,
47            position.z,
48            position.x + 1.0,
49            position.y + 1.0,
50            position.z + 1.0,
51        )
52        .inflate_xyz(within, ALERT_RANGE_Y, within);
53        let mob_type_key = mob.downcast_type_key();
54        let mob_uuid = mob.uuid();
55        for entity in world.get_entities_in_aabb_matching(&search_box, |entity| {
56            entity.uuid() != mob_uuid
57                && entity.downcast_type_key() == mob_type_key
58                && !entity.is_spectator()
59        }) {
60            if let Some(other) = entity.as_neutral_mob() {
61                other.forget_current_target_and_refresh_universal_anger();
62            }
63        }
64    }
65}
66
67impl Goal for ResetUniversalAngerTargetGoal {
68    fn controls(&self) -> GoalControls {
69        GoalControls::EMPTY
70    }
71
72    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
73        mob.level()
74            .is_some_and(|world| world.get_game_rule(&UNIVERSAL_ANGER))
75            && self.was_hurt_by_player(mob)
76    }
77
78    fn start(&mut self, mob: &dyn PathfinderMob) {
79        self.last_hurt_by_player_timestamp = mob.last_hurt_by_mob_timestamp();
80        if let Some(neutral) = mob.as_neutral_mob() {
81            neutral.forget_current_target_and_refresh_universal_anger();
82        }
83        if self.alert_others_of_same_type {
84            Self::alert_nearby_mobs_of_same_type(mob);
85        }
86    }
87}
