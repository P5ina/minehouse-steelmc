15/// Vanilla `Direction.values()` order.
16const ALL_DIRECTIONS: [Direction; 6] = [
17    Direction::Down,
18    Direction::Up,
19    Direction::North,
20    Direction::South,
21    Direction::West,
22    Direction::East,
23];
24
25/// Vanilla `Direction.Plane.HORIZONTAL` order, which `getClockWise` walks.
26const HORIZONTAL_DIRECTIONS: [Direction; 4] = [
27    Direction::North,
28    Direction::East,
29    Direction::South,
30    Direction::West,
31];
32
33/// Swim node evaluator. Only water nodes are accepted, plus breach nodes
34/// (air above water) when `allow_breaching` is set, as for dolphins.
35#[derive(Debug, Clone)]
36pub struct SwimNodeEvaluator {
37    settings: MobPathSettings,
38    allow_breaching: bool,
39    nodes: NodeStore,
40    path_types_by_pos: FxHashMap<BlockPos, PathType>,
41}

43impl SwimNodeEvaluator {
44    #[must_use]
45    pub fn new(settings: MobPathSettings, allow_breaching: bool) -> Self {
46        Self {
47            settings,
48            allow_breaching,
49            nodes: NodeStore::new(),
50            path_types_by_pos: FxHashMap::default(),
51        }
52    }

54    /// Classifies the mob-sized box whose minimum corner is at `x, y, z`.
55    #[must_use]
56    pub fn get_path_type_of_mob(
57        &self,
58        context: &PathfindingContext<'_>,
59        x: i32,
60        y: i32,
61        z: i32,
62    ) -> PathType {
63        let mut pos = BlockPos::new(x, y, z);
64        for xx in x..x + self.settings.entity_width() {
65            for yy in y..y + self.settings.entity_height() {
66                for zz in z..z + self.settings.entity_depth() {
67                    pos = BlockPos::new(xx, yy, zz);
68                    let block_state = context.get_block_state(pos);
69                    let fluid_state = block_state.get_fluid_state();
70                    let below_state = context.get_block_state(pos.below());
71                    if fluid_state.is_empty()
72                        && below_state.is_pathfindable(PathComputationType::Water)
73                        && block_state.is_air()
74                    {
75                        return PathType::Breach;
76                    }
77
78                    if !fluid_state.is_water() {
79                        return PathType::Blocked;
80                    }
81                }
82            }
83        }
84
85        // Vanilla reads the last position the loop visited.
86        if context
87            .get_block_state(pos)
88            .is_pathfindable(PathComputationType::Water)
89        {
90            PathType::Water
91        } else {
92            PathType::Blocked
93        }
94    }
95
96    fn get_cached_block_type(
97        &mut self,
98        context: &PathfindingContext<'_>,
99        x: i32,
100        y: i32,
101        z: i32,
102    ) -> PathType {
103        let pos = BlockPos::new(x, y, z);
104        if let Some(path_type) = self.path_types_by_pos.get(&pos) {
105            return *path_type;
106        }
107        let path_type = self.get_path_type_of_mob(context, x, y, z);
108        self.path_types_by_pos.insert(pos, path_type);
109        path_type
110    }
111
112    fn find_accepted_node(
113        &mut self,
114        context: &PathfindingContext<'_>,
115        x: i32,
116        y: i32,
117        z: i32,
118    ) -> Option<i32> {
119        let path_type = self.get_cached_block_type(context, x, y, z);
120        if !(self.allow_breaching && path_type == PathType::Breach || path_type == PathType::Water)
121        {
122            return None;
123        }
124
125        let path_cost = self.settings.pathfinding_malus(path_type);
126        if path_cost < 0.0 {
127            return None;
128        }
129
130        let no_fluid = context
131            .get_block_state(BlockPos::new(x, y, z))
132            .get_fluid_state()
133            .is_empty();
134        let node = self.nodes.get_node(x, y, z);
135        node.path_type = path_type;
136        node.cost_malus = node.cost_malus.max(path_cost);
137        if no_fluid {
138            node.cost_malus += 8.0;
139        }
140        Some(node.hash())
141    }
142
143    fn is_node_valid(&self, node: Option<i32>) -> bool {
144        node.and_then(|hash| self.node(hash))
145            .is_some_and(|node| !node.closed)
146    }
147
148    fn has_malus(&self, node: Option<i32>) -> bool {
149        node.and_then(|hash| self.node(hash))
150            .is_some_and(|node| node.cost_malus >= 0.0)
151    }
152}

154impl NodeEvaluator for SwimNodeEvaluator {
155    fn node(&self, hash: i32) -> Option<&Node> {
156        self.nodes.get(hash)
157    }
158
159    fn node_mut(&mut self, hash: i32) -> Option<&mut Node> {
160        self.nodes.get_mut(hash)
161    }
162
163    fn nodes_mut(&mut self) -> &mut NodeStore {
164        &mut self.nodes
165    }
166
167    fn reset_search_state(&mut self) {
168        self.nodes.reset_search_state();
169    }
170
171    fn get_start(&mut self, _context: &mut PathfindingContext<'_>) -> i32 {
172        let bounding_box = self.settings.bounding_box();
173        self.nodes
174            .get_node(
175                fast_floor(bounding_box.min_x()),
176                fast_floor(bounding_box.min_y() + 0.5),
177                fast_floor(bounding_box.min_z()),
178            )
179            .hash()
180    }
181
182    fn get_neighbors(
183        &mut self,
184        context: &mut PathfindingContext<'_>,
185        _collision: &mut impl WalkNodeCollision,
186        pos_hash: i32,
187    ) -> NodeNeighbors {
188        let mut neighbors = NodeNeighbors::new();
189        let Some(pos) = self.node(pos_hash) else {
190            return neighbors;
191        };
192        let (pos_x, pos_y, pos_z) = (pos.x, pos.y, pos.z);
193
194        let mut by_direction = [None; 6];
195        for (index, direction) in ALL_DIRECTIONS.iter().copied().enumerate() {
196            let (step_x, step_y, step_z) = direction.offset();
197            let node =
198                self.find_accepted_node(context, pos_x + step_x, pos_y + step_y, pos_z + step_z);
199            by_direction[index] = node;
200            if self.is_node_valid(node)
201                && let Some(node) = node
202            {
203                neighbors.push(node);
204            }
205        }
206
207        for direction in HORIZONTAL_DIRECTIONS {
208            let second_direction = direction.rotate_y_clockwise();
209            let first = by_direction[direction_index(direction)];
210            let second = by_direction[direction_index(second_direction)];
211            if !self.has_malus(first) || !self.has_malus(second) {
212                continue;
213            }
214
215            let (step_x, _, step_z) = direction.offset();
216            let (second_step_x, _, second_step_z) = second_direction.offset();
217            let diagonal = self.find_accepted_node(
218                context,
219                pos_x + step_x + second_step_x,
220                pos_y,
221                pos_z + step_z + second_step_z,
222            );
223            if self.is_node_valid(diagonal)
224                && let Some(diagonal) = diagonal
225            {
226                neighbors.push(diagonal);
227            }
228        }
229
230        neighbors
231    }
232}

234/// Index into [`ALL_DIRECTIONS`], which follows the 3D data value order.
235const fn direction_index(direction: Direction) -> usize {
236    direction.get_3d_data_value() as usize
237}

253    /// Water at y 64..=65 for x, z in 0..=4, air above, stone everywhere else.
254    struct Pond {
255        walls: Vec<BlockPos>,
256    }
257
258    impl LevelReader for Pond {
259        fn get_block_state(&self, pos: BlockPos) -> BlockStateId {
260            let inside = (0..=4).contains(&pos.x()) && (0..=4).contains(&pos.z());
261            if !inside || pos.y() < 64 || self.walls.contains(&pos) {
262                vanilla_blocks::STONE.default_state()
263            } else if pos.y() <= 65 {
264                vanilla_blocks::WATER.default_state()
265            } else {
266                vanilla_blocks::AIR.default_state()
267            }
268        }

271            0

276        }

279            OverworldNoiseSettings::HEIGHT

287    fn fish_settings() -> MobPathSettings {
288        let mut malus = PathfindingMalus::new();
289        malus.set(PathType::Water, 0.0);
290        MobPathSettings::new(1, 1, 1, BlockPos::new(0, 64, 0), &malus)
291    }

293    fn find(level: &Pond, allow_breaching: bool, target: BlockPos) -> Option<Vec<BlockPos>> {
294        let mut context = PathfindingContext::new(level, BlockPos::new(0, 64, 0));
295        let mut evaluator = SwimNodeEvaluator::new(fish_settings(), allow_breaching);
296        let mut no_collision = |_aabb: WorldAabb| false;
297        let path = PathFinder::new(512).find_path(
298            &mut evaluator,
299            &mut context,
300            &mut no_collision,
301            PathRequest {
302                targets: &[target],
303                max_path_length: 32.0,
304                reach_range: 0,
305                max_visited_nodes_multiplier: 1.0,
306            },
307        )?;
308        path.can_reach()
309            .then(|| path.nodes().iter().map(Node::as_block_pos).collect())
310    }

312    #[test]
313    fn swims_around_a_wall() {
314        init_vanilla_registry();
315        init_behaviors();
316
317        let walls = (0..=3)
318            .flat_map(|z| [BlockPos::new(2, 64, z), BlockPos::new(2, 65, z)])
319            .collect();
320        let level = Pond { walls };
321        let Some(nodes) = find(&level, false, BlockPos::new(4, 64, 0)) else {
322            panic!("a path around the wall should exist");
323        };
324
325        assert!(
326            nodes.contains(&BlockPos::new(2, 64, 4)) || nodes.contains(&BlockPos::new(2, 65, 4))
327        );
328        assert!(nodes.iter().all(|pos| pos.y() <= 65));
329    }

333        init_vanilla_registry();
334        init_behaviors();
335
336        let level = Pond { walls: Vec::new() };
337        let above_surface = BlockPos::new(3, 66, 3);

340        let Some(nodes) = find(&level, true, above_surface) else {
