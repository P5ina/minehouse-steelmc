1//! Monsters that roam in patrols (Vanilla `PatrollingMonster`).
2//!
3//! Patrol fields live in [`PatrollingMonsterState`]; the long-distance patrol goal is
4//! [`LongDistancePatrolGoal`](crate::entity::ai::goal::LongDistancePatrolGoal).
5
6use std::sync::atomic::{AtomicBool, Ordering};
7
8use simdnbt::borrow::NbtCompound as BorrowedNbtCompoundView;
9use simdnbt::owned::{NbtCompound, NbtTag};
10use steel_registry::entity_type::EntityTypeRef;
11use steel_utils::BlockPos;
12use steel_utils::locks::SyncMutex;
13
14use crate::chunk::light::LightLayer;
15use crate::entity::{EntitySpawnReason, Monster, ominous_banner_instance};
16use crate::inventory::prelude::EquipmentSlot;
17use crate::world::ServerLevelAccessor;
18
19const TAG_PATROL_TARGET: &str = "patrol_target";
20const TAG_PATROL_LEADER: &str = "PatrolLeader";
21const TAG_PATROLLING: &str = "Patrolling";
22/// Vanilla `finalizeSpawn`: chance for a naturally spawned patroller to lead.
23const PATROL_LEADER_CHANCE: f32 = 0.06;
24/// Vanilla `checkPatrollingMonsterSpawnRules` block light limit.
25const PATROL_SPAWN_MAX_BLOCK_LIGHT: u8 = 8;
26/// Vanilla `removeWhenFarAway`: a patrolling monster only despawns beyond 128 blocks.
27const PATROLLING_DESPAWN_DISTANCE_SQR: f64 = 16384.0;
28/// Vanilla `findPatrolTarget` picks a point up to 500 blocks away on each axis.
29const PATROL_TARGET_SPREAD: i32 = 500;
30/// Vanilla `setDropChance(EquipmentSlot.HEAD, 2.0F)` for the leader's banner.
31pub(crate) const GUARANTEED_BANNER_DROP_CHANCE: f32 = 2.0;
32
33/// Patrol fields of a patrolling monster.
34pub struct PatrollingMonsterState {
35    patrol_target: SyncMutex<Option<BlockPos>>,
36    patrol_leader: AtomicBool,
37    patrolling: AtomicBool,
38}
39
40impl PatrollingMonsterState {
41    #[must_use]
42    pub(crate) const fn new() -> Self {
43        Self {
44            patrol_target: SyncMutex::new(None),
45            patrol_leader: AtomicBool::new(false),
46            patrolling: AtomicBool::new(false),
47        }
48    }
49}
50
51impl Default for PatrollingMonsterState {
52    fn default() -> Self {
53        Self::new()
54    }
55}
56
57/// Vanilla `PatrollingMonster`.
58pub trait PatrollingMonster: Monster {
59    /// Vanilla `PatrollingMonster.checkPatrollingMonsterSpawnRules`: no spawning
60    /// in block light above 8, otherwise `checkAnyLightMonsterSpawnRules`.
61    fn check_patrolling_monster_spawn_rules(
62        entity_type: EntityTypeRef,
63        level: &dyn ServerLevelAccessor,
64        spawn_reason: EntitySpawnReason,
65        pos: BlockPos,
66    ) -> bool
67    where
68        Self: Sized,
69    {
70        level.brightness(LightLayer::Block, pos) <= PATROL_SPAWN_MAX_BLOCK_LIGHT
71            && <Self as Monster>::check_any_light_monster_spawn_rules(
72                entity_type,
73                level,
74                spawn_reason,
75                pos,
76            )
77    }
78
79    fn patrolling_monster_state(&self) -> &PatrollingMonsterState;
80
81    /// Vanilla `canBeLeader`.
82    fn can_be_leader(&self) -> bool {
83        true
84    }
85
86    /// Vanilla `canJoinPatrol`.
87    fn can_join_patrol(&self) -> bool {
88        true
89    }
90
91    fn patrol_target(&self) -> Option<BlockPos> {
92        *self.patrolling_monster_state().patrol_target.lock()
93    }
94
95    fn has_patrol_target(&self) -> bool {
96        self.patrol_target().is_some()
97    }
98
99    /// Vanilla `setPatrolTarget`: also marks the monster as patrolling.
100    fn set_patrol_target(&self, target: BlockPos) {
101        *self.patrolling_monster_state().patrol_target.lock() = Some(target);
102        self.set_patrolling(true);
103    }
104
105    fn is_patrol_leader(&self) -> bool {
106        self.patrolling_monster_state()
107            .patrol_leader
108            .load(Ordering::Relaxed)
109    }
110
111    /// Vanilla `setPatrolLeader`: also marks the monster as patrolling.
112    fn set_patrol_leader(&self, leader: bool) {
113        self.patrolling_monster_state()
114            .patrol_leader
115            .store(leader, Ordering::Relaxed);
116        self.set_patrolling(true);
117    }
118
119    fn is_patrolling(&self) -> bool {
120        self.patrolling_monster_state()
121            .patrolling
122            .load(Ordering::Relaxed)
123    }
124
125    fn set_patrolling(&self, patrolling: bool) {
126        self.patrolling_monster_state()
127            .patrolling
128            .store(patrolling, Ordering::Relaxed);
129    }
130
131    /// Vanilla `findPatrolTarget`.
132    fn find_patrol_target(&self) {
133        let target = self.block_position().offset(
134            -PATROL_TARGET_SPREAD + rand::random_range(0..PATROL_TARGET_SPREAD * 2),
135            0,
136            -PATROL_TARGET_SPREAD + rand::random_range(0..PATROL_TARGET_SPREAD * 2),
137        );
138        self.set_patrol_target(target);
139    }
140
141    /// Vanilla `removeWhenFarAway`.
142    fn remove_when_far_away_patrolling(&self, dist_sqr: f64) -> bool {
143        !self.is_patrolling() || dist_sqr > PATROLLING_DESPAWN_DISTANCE_SQR
144    }
145
146    /// The patrol part of Vanilla `finalizeSpawn`, run before the mob's own.
147    fn finalize_spawn_patrolling(&self, spawn_reason: EntitySpawnReason) {
148        if !matches!(
149            spawn_reason,
150            EntitySpawnReason::Patrol | EntitySpawnReason::Event | EntitySpawnReason::Structure
151        ) && rand::random::<f32>() < PATROL_LEADER_CHANCE
152            && self.can_be_leader()
153        {
154            self.patrolling_monster_state()
155                .patrol_leader
156                .store(true, Ordering::Relaxed);
157        }
158
159        if self.is_patrol_leader() {
160            self.living_base()
161                .equipment()
162                .lock()
163                .set(EquipmentSlot::Head, ominous_banner_instance());
164            self.set_equipment_drop_chance(EquipmentSlot::Head, GUARANTEED_BANNER_DROP_CHANCE);
165        }
166
167        if spawn_reason == EntitySpawnReason::Patrol {
168            self.set_patrolling(true);
169        }
170    }
171
172    fn save_patrolling(&self, nbt: &mut NbtCompound) {
173        if let Some(target) = self.patrol_target() {
174            nbt.insert(
175                TAG_PATROL_TARGET,
176                NbtTag::IntArray(vec![target.x(), target.y(), target.z()]),
177            );
178        }
179        nbt.insert(TAG_PATROL_LEADER, i8::from(self.is_patrol_leader()));
180        nbt.insert(TAG_PATROLLING, i8::from(self.is_patrolling()));
181    }
182
183    fn load_patrolling(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
184        let state = self.patrolling_monster_state();
185        *state.patrol_target.lock() =
186            nbt.int_array(TAG_PATROL_TARGET)
187                .and_then(|pos| match pos.as_slice() {
188                    [x, y, z] => Some(BlockPos::new(*x, *y, *z)),
189                    _ => None,
190                });
191        state.patrol_leader.store(
192            nbt.byte(TAG_PATROL_LEADER).is_some_and(|b| b != 0),
193            Ordering::Relaxed,
194        );
195        state.patrolling.store(
196            nbt.byte(TAG_PATROLLING).is_some_and(|b| b != 0),
197            Ordering::Relaxed,
198        );
199    }
200}
