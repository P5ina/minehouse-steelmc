15/// Vanilla initial `AbstractHurtingProjectile.accelerationPower`.
16pub const INITIAL_ACCELERATION_POWER: f64 = 0.1;
17const ACCELERATION_POWER_TAG: &str = "acceleration_power";
18const ROTATION_SPEED: f32 = 0.2;
19const BURN_SECONDS_TICKS: i32 = 20;
20/// Vanilla `Vec3.normalize` treats shorter vectors as zero.
21const NORMALIZE_EPSILON: f64 = 1.0e-5_f32 as f64;
22
23/// Runtime fields of vanilla `AbstractHurtingProjectile`.
24#[derive(Debug)]
25pub struct HurtingProjectileBase {
26    acceleration_power: SyncMutex<f64>,
27}
28
29impl HurtingProjectileBase {
30    /// Creates the state with the default acceleration.
31    #[must_use]
32    pub const fn new() -> Self {
33        Self {
34            acceleration_power: SyncMutex::new(INITIAL_ACCELERATION_POWER),
35        }
36    }
37}
38
39impl Default for HurtingProjectileBase {
40    fn default() -> Self {
41        Self::new()
42    }
43}

45/// Vanilla `Vec3.normalize`.
46#[must_use]
47pub fn normalize_or_zero(vector: DVec3) -> DVec3 {
48    let length = vector.length();
49    if length < NORMALIZE_EPSILON {
50        DVec3::ZERO
51    } else {
52        vector / length
53    }
54}
55
56/// Vanilla `(float)(angle * 180.0F / (float)Math.PI)`.
57fn vanilla_degrees(radians: f64) -> f32 {
58    (radians * 180.0 / f64::from(PI)) as f32
59}

61/// Behavior shared by projectiles extending vanilla `AbstractHurtingProjectile`.
62pub trait AbstractHurtingProjectile: Projectile {
63    /// Returns the shared hurting-projectile state.
64    fn hurting_projectile_base(&self) -> &HurtingProjectileBase;
65
66    /// Speed added along the flight direction every tick.
67    fn acceleration_power(&self) -> f64 {
68        *self.hurting_projectile_base().acceleration_power.lock()
69    }
70
71    /// Sets the speed added along the flight direction every tick.
72    fn set_acceleration_power(&self, acceleration_power: f64) {
73        *self.hurting_projectile_base().acceleration_power.lock() = acceleration_power;
74    }
75
76    /// Velocity multiplier per tick in air.
77    fn get_inertia(&self) -> f32 {
78        0.95
79    }
80
81    /// Velocity multiplier per tick in water.
82    fn get_liquid_inertia(&self) -> f32 {
83        0.8
84    }
85
86    /// Whether the projectile sets itself on fire every tick.
87    fn should_burn(&self) -> bool {
88        true
89    }
90
91    /// Vanilla constructor with a direction: start flying along it at the
92    /// acceleration power.
93    fn assign_directional_movement(&self, direction: DVec3) {
94        self.set_velocity(normalize_or_zero(direction) * self.acceleration_power());
95        self.mark_velocity_sync();
96    }
97
98    /// Vanilla constructor from a shooter: placed at its feet, flying along
99    /// `direction`, owned by it and facing its rotation.
100    fn shoot_from_mob(&self, owner: &SharedEntity, direction: DVec3) {
101        if let Err(error) = self.try_set_position(owner.position()) {
102            log::debug!("hurting projectile could not move to its shooter: {error}");
103        }
104        self.assign_directional_movement(direction);
105        self.set_owner_entity(Some(owner));
106        self.set_rotation(owner.rotation());
107    }
108
109    /// Vanilla `applyInertia`.
110    fn apply_hurting_inertia(&self) {
111        let movement = self.velocity();
112        // VANILLA CLIENT-LOCAL: the bubbles in water are client particles.
113        let inertia = if self.is_in_water() {
114            self.get_liquid_inertia()
115        } else {
116            self.get_inertia()
117        };
118        self.set_velocity(
119            (movement + normalize_or_zero(movement) * self.acceleration_power())
120                * f64::from(inertia),
121        );
122    }
123
124    /// Vanilla `ProjectileUtil.rotateTowardsMovement`.
125    fn rotate_towards_movement(&self, rotation_speed: f32) {
126        let movement = self.velocity();
127        if movement.length_squared() == 0.0 {
128            return;
129        }
130        let horizontal = movement.x.hypot(movement.z);
131        let y_rot = vanilla_degrees(movement.z.atan2(movement.x)) + 90.0;
132        let x_rot = vanilla_degrees(horizontal.atan2(movement.y)) - 90.0;
133        let (mut y_rot_old, mut x_rot_old) = self.base().old_rotation();
134        while x_rot - x_rot_old < -180.0 {
135            x_rot_old -= 360.0;
136        }
137        while x_rot - x_rot_old >= 180.0 {
138            x_rot_old += 360.0;
139        }
140        while y_rot - y_rot_old < -180.0 {
141            y_rot_old -= 360.0;
142        }
143        while y_rot - y_rot_old >= 180.0 {
144            y_rot_old += 360.0;
145        }
146        self.base().set_old_rotation((y_rot_old, x_rot_old));
147        self.set_rotation((
148            y_rot_old + rotation_speed * (y_rot - y_rot_old),
149            x_rot_old + rotation_speed * (x_rot - x_rot_old),
150        ));
151    }
152
153    /// Vanilla `AbstractHurtingProjectile.tick`.
154    fn hurting_projectile_tick(self: Arc<Self>) {
155        self.set_old_position_to_current();
156        self.base().set_old_rotation_to_current();
157
158        let owner = self.get_owner();
159        self.apply_hurting_inertia();
160        let Some(world) = self.level() else {
161            return;
162        };
163        let owner_present = owner.is_none_or(|owner| !owner.is_removed());
164        if !owner_present || !world.is_full_chunk_loaded_at(self.block_position()) {
165            self.set_removed(RemovalReason::Discarded);
166            return;
167        }
168
169        let hit = self.get_hit_result_on_move_vector();
170        let new_position = hit.as_ref().map_or_else(
171            || self.position() + self.velocity(),
172            ProjectileHit::location,
173        );
174        self.rotate_towards_movement(ROTATION_SPEED);
175        if let Err(error) = self.try_set_position(new_position) {
176            log::debug!(
177                "failed to advance hurting projectile {}: {error}",
178                self.id()
179            );
180            self.set_removed(RemovalReason::Discarded);
181            return;
182        }
183        self.apply_effects_from_blocks();
184        self.projectile_base_tick();
185        if self.should_burn() {
186            self.ignite_for_ticks(BURN_SECONDS_TICKS);
187        }
188        if let Some(hit) = hit
189            && self.is_alive()
190            && !self.is_world_change_pending()
191        {
192            self.hit_target_or_deflect_self(&hit);
193        }
194        // VANILLA CLIENT-LOCAL: `createParticleTrail` only adds client particles.
195    }
196
197    /// Vanilla `onDeflection`: a hit resets the acceleration, other deflections halve it.
198    fn on_hurting_deflection(&self, by_attack: bool) {
199        if by_attack {
200            self.set_acceleration_power(INITIAL_ACCELERATION_POWER);
201        } else {
202            self.set_acceleration_power(self.acceleration_power() * 0.5);
203        }
204    }
205
206    /// Vanilla `canHitEntity`: also skips entities without physics.
207    fn hurting_can_hit_entity(&self, entity: &dyn Entity) -> bool {
208        self.base_can_hit_entity(entity) && !entity.no_physics()
209    }
210
211    /// Saves `acceleration_power`.
212    fn save_hurting_projectile(&self, nbt: &mut NbtCompound) {
213        nbt.insert(ACCELERATION_POWER_TAG, self.acceleration_power());
214    }
215
216    /// Loads `acceleration_power`.
217    fn load_hurting_projectile(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
218        self.set_acceleration_power(
219            nbt.double(ACCELERATION_POWER_TAG)
220                .unwrap_or(INITIAL_ACCELERATION_POWER),
221        );
222    }
223}
