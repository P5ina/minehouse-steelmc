26/// Behavior of vanilla `Fireball`: a hurting projectile showing an item (a fire
27/// charge by default).
28pub trait Fireball: AbstractHurtingProjectile {
29    /// Returns the synced item stack.
30    fn get_item(&self) -> ItemStack;
31
32    /// Writes the synced item stack.
33    fn set_synced_item(&self, item: ItemStack);
34
35    /// Vanilla `setItem`: an empty stack falls back to a fire charge.
36    fn set_item(&self, item: &ItemStack) {
37        if item.is_empty() {
38            self.set_synced_item(default_fireball_item());
39        } else {
40            self.set_synced_item(item.copy_with_count(1));
41        }
42    }
43
44    /// Saves `Item`.
45    fn save_fireball(&self, nbt: &mut NbtCompound) {
46        self.save_projectile(nbt);
47        self.save_hurting_projectile(nbt);
48        let item = self.get_item();
49        if !item.is_empty() {
50            nbt.insert("Item", item.to_nbt_tag());
51        }
52    }
53
54    /// Loads `Item`.
55    fn load_fireball(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
56        self.load_projectile(nbt);
57        self.load_hurting_projectile(nbt);
58        let item = nbt
59            .compound("Item")
60            .and_then(|tag| ItemStack::from_borrowed_compound(&tag))
61            .unwrap_or_else(default_fireball_item);
62        self.set_item(&item);
63    }
64}
65
66fn default_fireball_item() -> ItemStack {
67    ItemStack::new(&vanilla_items::FIRE_CHARGE)
68}

70/// Vanilla `DamageSources.fireball`: credited to the shooter, or to the
71/// fireball itself when nobody shot it.
72fn fireball_damage_source(fireball: SharedEntity, owner: Option<SharedEntity>) -> DamageSource {
73    match owner {
74        Some(owner) => DamageSource::environment(&vanilla_damage_types::FIREBALL)
75            .with_direct_entity(fireball)
76            .with_causing_entity(owner),
77        None => DamageSource::environment(&vanilla_damage_types::UNATTRIBUTED_FIREBALL)
78            .with_direct_entity(Arc::clone(&fireball))
79            .with_causing_entity(fireball),
80    }
81}

83/// Vanilla `EnchantmentHelper.doPostAttackEffects` for a projectile hit: the
84/// victim's armor reacts; fireballs carry no weapon.
85fn fireball_post_attack_effects(world: &World, victim: &SharedEntity, source: &DamageSource) {
86    let context = EnchantmentPostAttackContext::new(victim, source);
87    enchantment_helper::do_post_attack_effects_with_item_source(
88        world,
89        victim.as_ref(),
90        &ItemStack::empty(),
91        &context,
