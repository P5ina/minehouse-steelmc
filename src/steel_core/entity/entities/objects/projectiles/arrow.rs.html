1//! Arrow and spectral arrow entities (vanilla `Arrow`, `SpectralArrow`).
2//!
3//! Both extend vanilla `AbstractArrow`, ported in
4//! [`crate::entity::projectile::AbstractArrow`]. The arrow carries the potion
5//! of a tipped arrow and applies it on hit; the spectral arrow makes its
6//! target glow.

34/// Vanilla `Arrow.tick`: a tipped arrow loses its potion after this long in
35/// the ground.
36const POTION_FADE_TICKS: i32 = 600;
37/// Vanilla `SpectralArrow.DEFAULT_DURATION` of the glowing effect.
38const SPECTRAL_DEFAULT_DURATION: i32 = 200;
39
40/// The `Entity` / `Projectile` / `AbstractArrow` plumbing every arrow shares.
41macro_rules! arrow_entity_common {
42    () => {
43        fn base(&self) -> &EntityBase {
44            &self.base
45        }
46
47        fn entity_type(&self) -> EntityTypeRef {
48            self.entity_type
49        }
50
51        fn get_default_gravity(&self) -> f64 {
52            ARROW_GRAVITY
53        }
54
55        fn spawn_data(&self) -> i32 {
56            self.get_owner().map_or(0, |owner| owner.id())
57        }
58
59        fn restore_owner_reference(&self, owner: &SharedEntity) {
60            self.cache_owner_entity(owner);
61        }
62
63        fn projectile_owner_uuid(&self) -> Option<uuid::Uuid> {
64            self.owner_uuid()
65        }
66
67        fn projectile_owner(&self) -> Option<SharedEntity> {
68            self.get_owner()
69        }
70
71        fn synced_data(&self) -> Option<&dyn EntitySyncedData> {
72            Some(&self.entity_data)
73        }
74
75        fn player_touch(self: Arc<Self>, player: &Arc<Player>) {
76            self.arrow_player_touch(player);
77        }
78
79        fn push_impulse(&self, impulse: DVec3) {
80            // Vanilla `AbstractArrow.push`: stuck arrows ignore pushes.
81            if self.is_in_ground() || !impulse.is_finite() {
82                return;
83            }
84            self.set_velocity(self.velocity() + impulse);
85            self.mark_velocity_sync();
86        }
87
88        fn on_above_bubble_column(&self, drag_down: bool, pos: BlockPos) {
89            if !self.is_in_ground() {
90                self.on_above_bubble_column_projectile(drag_down, pos);
91            }
92        }
93
94        fn on_inside_bubble_column(&self, drag_down: bool) {
95            if !self.is_in_ground() {
96                self.on_inside_bubble_column_projectile(drag_down);
97            }
98        }
99
100        fn is_pushed_by_fluid(&self) -> bool {
101            !self.is_in_ground()
102        }
103
104        fn move_entity(self: Arc<Self>, mover_type: MoverType, delta: DVec3) -> Option<MoveResult> {
105            let result = Arc::clone(&self).default_move_entity(mover_type, delta);
106            self.arrow_after_move(mover_type);
107            result
108        }
109    };
110}

146/// The synced-data accessors over an `abstract_arrow` data layer.
147macro_rules! arrow_synced_accessors {
148    () => {
149        fn arrow_base(&self) -> &ArrowBase {
150            &self.arrow_base
151        }
152
153        fn arrow_flags(&self) -> i8 {
154            *self.entity_data.lock().abstract_arrow.id_flags.get()
155        }
156
157        fn set_arrow_flags(&self, flags: i8) {
158            self.entity_data.lock().abstract_arrow.id_flags.set(flags);
159        }
160
161        fn pierce_level(&self) -> i8 {
162            *self.entity_data.lock().abstract_arrow.pierce_level.get()
163        }
164
165        fn set_pierce_level(&self, pierce_level: i8) {
166            self.entity_data
167                .lock()
168                .abstract_arrow
169                .pierce_level
170                .set(pierce_level);
171        }
172
173        fn is_in_ground(&self) -> bool {
174            *self.entity_data.lock().abstract_arrow.in_ground.get()
175        }
176
177        fn set_in_ground(&self, in_ground: bool) {
178            self.entity_data
179                .lock()
180                .abstract_arrow
181                .in_ground
182                .set(in_ground);
183        }
184    };
185}

187/// A fired arrow, plain or tipped.
188#[entity_behavior(class = "Arrow")]
189pub struct ArrowEntity {
190    base: EntityBase,
191    entity_type: EntityTypeRef,
192    entity_data: SyncMutex<ArrowEntityData>,
193    projectile_base: ProjectileBase,
194    arrow_base: ArrowBase,
195}
196
197// SAFETY: This key is owned by Steel and uniquely identifies `ArrowEntity`.
198unsafe impl DowncastType for ArrowEntity {
199    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("steel:entity/arrow");
200}
201
202impl ArrowEntity {
203    /// Creates an arrow with no owner carrying a plain arrow item.
204    #[must_use]
205    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
206        Self {
207            base: EntityBase::new(id, position, entity_type.dimensions, world),
208            entity_type,
209            entity_data: SyncMutex::new(ArrowEntityData::new()),
210            projectile_base: ProjectileBase::new(),
211            arrow_base: ArrowBase::new(ItemStack::new(&vanilla_items::ARROW)),
212        }
213    }
214
215    /// Creates an arrow from saved base data.
216    #[must_use]
217    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
218        Self {
219            base: EntityBase::from_load(load, entity_type.dimensions),
220            entity_type,
221            entity_data: SyncMutex::new(ArrowEntityData::new()),
222            projectile_base: ProjectileBase::new(),
223            arrow_base: ArrowBase::new(ItemStack::new(&vanilla_items::ARROW)),
224        }
225    }
226
227    /// Vanilla `Arrow(level, x, y, z, pickupItemStack, firedFromWeapon)`.
228    #[must_use]
229    pub fn with_items(
230        entity_type: EntityTypeRef,
231        id: i32,
232        position: DVec3,
233        world: Weak<World>,
234        pickup_item_stack: &ItemStack,
235        fired_from_weapon: Option<&ItemStack>,
236    ) -> Self {
237        let arrow = Self::new(entity_type, id, position, world);
238        arrow.init_from_items(pickup_item_stack, fired_from_weapon);
239        arrow
240    }
241
242    fn potion_contents(&self) -> PotionContents {
243        self.pickup_item_stack()
244            .get_or_default(POTION_CONTENTS, PotionContents::empty())
245    }
246
247    /// Vanilla `Arrow.updateColor`.
248    fn update_color(&self) {
249        let contents = self.potion_contents();
250        let color = if contents == PotionContents::empty() {
251            -1
252        } else {
253            contents.get_color()
254        };
255        self.entity_data.lock().id_effect_color.set(color);
256    }
257
258    /// Vanilla `Arrow.setPotionContents`: stores the contents on the pickup
259    /// stack and refreshes the color.
260    fn set_potion_contents(&self, contents: PotionContents) {
261        let mut stack = self.pickup_item_stack();
262        stack.set(POTION_CONTENTS, contents);
263        self.set_pickup_item_stack(stack);
264    }
265
266    /// Vanilla `Arrow.addEffect`.
267    pub fn add_effect(&self, effect: RegistryMobEffectInstance) {
268        self.set_potion_contents(self.potion_contents().with_effect_added(effect));
269    }
270
271    /// Vanilla `Arrow.getColor`: the synced potion color, `-1` for none.
272    #[must_use]
273    pub fn color(&self) -> i32 {
274        *self.entity_data.lock().id_effect_color.get()
275    }
276}

278impl Entity for ArrowEntity {
279    arrow_entity_common!();
280
281    fn tick(self: Arc<Self>) {
282        Arc::clone(&self).abstract_arrow_tick();
283        // VANILLA CLIENT-LOCAL: the potion particle trail.
284        if self.is_in_ground()
285            && self.in_ground_time() != 0
286            && self.potion_contents() != PotionContents::empty()
287            && self.in_ground_time() >= POTION_FADE_TICKS
288        {
289            self.broadcast_entity_event(EntityStatus::TippedArrow);
290            self.set_pickup_item_stack(ItemStack::new(&vanilla_items::ARROW));
291        }
292    }
293
294    fn save_additional(&self, nbt: &mut NbtCompound) {
295        self.save_arrow(nbt);
296    }
297
298    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
299        self.load_arrow(nbt);
300    }
301}
302
303impl Projectile for ArrowEntity {
304    arrow_projectile_common!();
305}
306
307impl AbstractArrow for ArrowEntity {
308    arrow_synced_accessors!();
309
310    fn get_default_pickup_item(&self) -> ItemStack {
311        ItemStack::new(&vanilla_items::ARROW)
312    }
313
314    fn set_pickup_item_stack(&self, item_stack: ItemStack) {
315        self.default_set_pickup_item_stack(item_stack);
316        self.update_color();
317    }
318
319    fn do_post_hurt_effects(&self, mob: &dyn LivingEntity) {
320        let pickup = self.pickup_item_stack();
321        let contents = pickup.get_or_default(POTION_CONTENTS, PotionContents::empty());
322        let duration_scale = pickup.get_or_default(POTION_DURATION_SCALE, 1.0);
323        let owner = self.get_owner();
324        let effect_source: &dyn Entity = owner.as_deref().unwrap_or(self);
325        for effect in contents.all_effects() {
326            let scaled = effect.with_scaled_duration(duration_scale);
327            mob.add_mob_effect_with_source(
328                to_runtime_instance(&scaled, scaled.duration()),
329                Some(effect_source),
330            );
331        }
332    }
333}

394impl Entity for SpectralArrowEntity {
395    arrow_entity_common!();
396
397    fn tick(self: Arc<Self>) {
398        // VANILLA CLIENT-LOCAL: the in-flight effect particles.
399        self.abstract_arrow_tick();
400    }
401
402    fn save_additional(&self, nbt: &mut NbtCompound) {
403        self.save_arrow(nbt);
404        nbt.insert("Duration", *self.duration.lock());
405    }
406
407    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
408        self.load_arrow(nbt);
409        *self.duration.lock() = nbt.int("Duration").unwrap_or(SPECTRAL_DEFAULT_DURATION);
410    }
411}
