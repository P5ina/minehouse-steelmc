25const TAG_WARMUP: &str = "Warmup";
26const TAG_OWNER: &str = "Owner";
27/// Vanilla `lifeTicks` start: 20 ticks of attack plus `LIFE_OFFSET`.
28const START_LIFE_TICKS: i32 = 22;
29/// Vanilla bites on the eighth tick after the warmup runs out.
30const BITE_TICK: i32 = -8;
31const BITE_DAMAGE: f32 = 6.0;
32const BITE_REACH: f64 = 0.2;
33
34struct EvokerFangsState {
35    warmup_delay_ticks: i32,
36    sent_spike_event: bool,
37    life_ticks: i32,
38    owner: Option<EntityReference>,
39}

41/// Evoker fangs: bite every living thing standing on them once.
42#[entity_behavior(class = "EvokerFangs")]
43pub struct EvokerFangsEntity {
44    base: EntityBase,
45    entity_type: EntityTypeRef,
46    entity_data: SyncMutex<EvokerFangsEntityData>,
47    state: SyncMutex<EvokerFangsState>,
48}

55impl EvokerFangsEntity {
56    /// Creates fangs with no warmup and no owner.
57    #[must_use]
58    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
59        Self::new_with_base(
60            EntityBase::new(id, position, entity_type.dimensions, world),
61            entity_type,
62        )
63    }
64
65    /// Loads saved fangs.
66    #[must_use]
67    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
68        Self::new_with_base(
69            EntityBase::from_load(load, entity_type.dimensions),
70            entity_type,
71        )
72    }
73
74    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
75        Self {
76            base,
77            entity_type,
78            entity_data: SyncMutex::new(EvokerFangsEntityData::new()),
79            state: SyncMutex::new(EvokerFangsState {
80                warmup_delay_ticks: 0,
81                sent_spike_event: false,
82                life_ticks: START_LIFE_TICKS,
83                owner: None,
84            }),
85        }
86    }
87
88    /// Vanilla `new EvokerFangs(level, x, y, z, rotationRadians, warmupDelayTicks, owner)`.
89    pub fn configure(&self, rotation_radians: f32, warmup_delay_ticks: i32, owner: &SharedEntity) {
90        {
91            let mut state = self.state.lock();
92            state.warmup_delay_ticks = warmup_delay_ticks;
93            state.owner = Some(EntityReference::from_entity(owner));
94        }
95        self.set_rotation((rotation_radians.to_degrees(), self.rotation().1));
96    }
97
98    /// Vanilla `getOwner`.
99    #[must_use]
100    pub fn owner(&self) -> Option<SharedEntity> {
101        let owner = self.state.lock().owner.clone()?;
102        let world = self.level()?;
103        owner
104            .get_entity(&world)
105            .filter(|entity| entity.as_living_entity().is_some())
106    }
107
108    /// Vanilla `getWarmupDelayTicks`-style accessor for tests and commands.
109    #[must_use]
110    pub fn warmup_delay_ticks(&self) -> i32 {
111        self.state.lock().warmup_delay_ticks
112    }
113
114    fn server_tick(self: &Arc<Self>, world: &Arc<World>) {
115        let (bite, send_spike, discard) = {
116            let mut state = self.state.lock();
117            state.warmup_delay_ticks -= 1;
118            if state.warmup_delay_ticks >= 0 {
119                return;
120            }
121            let bite = state.warmup_delay_ticks == BITE_TICK;
122            let send_spike = !state.sent_spike_event;
123            state.sent_spike_event = true;
124            state.life_ticks -= 1;
125            (bite, send_spike, state.life_ticks < 0)
126        };
127
128        if bite {
129            let search_box = self.bounding_box().inflate_xyz(BITE_REACH, 0.0, BITE_REACH);
130            let victims = world.get_entities_in_aabb_matching(&search_box, |entity| {
131                entity.as_living_entity().is_some()
132            });
133            let owner = self.owner();
134            let fangs: SharedEntity = Arc::<Self>::clone(self);
135            for victim in victims {
136                deal_damage_to(world, &fangs, owner.as_ref(), &victim);
137            }
138        }
139
140        if send_spike {
141            self.broadcast_entity_event(EntityStatus::StartAttacking);
142        }
143
144        if discard {
145            self.set_removed(RemovalReason::Discarded);
146        }
147    }
148}
149
150/// Vanilla `EvokerFangs.dealDamageTo`.
151fn deal_damage_to(
152    world: &World,
153    fangs: &SharedEntity,
154    owner: Option<&SharedEntity>,
155    victim: &SharedEntity,
156) {
157    if !victim.is_alive()
158        || victim.is_invulnerable()
159        || owner.is_some_and(|owner| owner.uuid() == victim.uuid())
160    {
161        return;
162    }
163    let Some(owner) = owner else {
164        let source = DamageSource::environment(&vanilla_damage_types::MAGIC);
165        let _ = victim.hurt(world, &source, BITE_DAMAGE);
166        return;
167    };
168    if owner.is_allied_to(victim.as_ref()) {
169        return;
170    }
171    let source = DamageSource::direct(&vanilla_damage_types::INDIRECT_MAGIC, Arc::clone(fangs))
172        .with_causing_entity(Arc::clone(owner));
173    if victim.hurt(world, &source, BITE_DAMAGE) {
174        let context = EnchantmentPostAttackContext::new(victim, &source);
175        enchantment_helper::do_post_attack_effects_with_item_source(
176            world,
177            victim.as_ref(),
178            &ItemStack::empty(),
179            &context,
180        );
181    }
182}

184impl Entity for EvokerFangsEntity {
185    fn base(&self) -> &EntityBase {
186        &self.base
187    }
188
189    fn entity_type(&self) -> EntityTypeRef {
190        self.entity_type
191    }
192
193    fn tick(self: Arc<Self>) {
194        self.default_tick();
195        if let Some(world) = self.level() {
196            self.server_tick(&world);
197        }
198    }
199
200    fn hurt(&self, _world: &World, _source: &DamageSource, _amount: f32) -> bool {
201        false
202    }
203
204    fn synced_data(&self) -> Option<&dyn EntitySyncedData> {
205        Some(&self.entity_data)
206    }
207
208    fn save_additional(&self, nbt: &mut NbtCompound) {
209        let state = self.state.lock();
210        nbt.insert(TAG_WARMUP, state.warmup_delay_ticks);
211        if let Some(owner) = state.owner.as_ref() {
212            owner.store(nbt, TAG_OWNER);
213        }
214    }
215
216    fn load_additional(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
217        let mut state = self.state.lock();
218        state.warmup_delay_ticks = nbt.int(TAG_WARMUP).unwrap_or(0);
219        state.owner = EntityReference::read(&nbt, TAG_OWNER);
220    }
221}
