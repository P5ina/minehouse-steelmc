93    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
94        let living_base = LivingEntityBase::new(entity_type);
95        let mob_base = MobBase::new();
96        let mut entity_data = SnowGolemEntityData::new();
97        living_base.initialize_synced_data(&mut entity_data);
98
99        {
100            let mut goal_selector = mob_base.goal_selector().lock();
101            goal_selector.add_goal(1, RangedAttackGoal::new(1.25, 20, 10.0));
102            goal_selector.add_goal(
103                2,
104                WaterAvoidingRandomStrollGoal::with_probability(1.0, STROLL_PROBABILITY),
105            );
106            goal_selector.add_goal(3, LookAtPlayerGoal::new(6.0));
107            goal_selector.add_goal(4, RandomLookAroundGoal::new());
108
109            let mut target_selector = mob_base.target_selector().lock();
110            target_selector.add_goal(
111                1,
112                NearestAttackableTargetGoal::new_with_interval(10, true, false, |target, _| {
113                    target.as_mob().is_some() && target.is_enemy()
114                }),
115            );
116        }
117
118        Self {
119            base,
120            entity_type,
121            living_base,
122            mob_base,
123            entity_data: SyncMutex::new(entity_data),
124        }
125    }

127    /// Whether the golem still wears its carved pumpkin.
128    #[must_use]
129    pub fn has_pumpkin(&self) -> bool {
130        *self.entity_data.lock().pumpkin.get() & PUMPKIN_FLAG != 0
131    }

133    /// Puts the carved pumpkin on or takes it off.
134    pub fn set_pumpkin(&self, pumpkin: bool) {
135        let mut entity_data = self.entity_data.lock();
136        let current = *entity_data.pumpkin.get();
137        entity_data.pumpkin.set(if pumpkin {
138            current | PUMPKIN_FLAG
139        } else {
140            current & !PUMPKIN_FLAG
141        });
142    }

144    /// Melts in hot biomes and leaves snow layers where it stands.
145    fn snow_golem_ai_step(&self) {
146        let Some(world) = self.level() else {
147            return;
148        };
149        if world.snow_golem_melts(self.position()) {
150            self.hurt_server(
151                &world,
152                &DamageSource::environment(&vanilla_damage_types::ON_FIRE),
153                1.0,
154            );
155        }
156        if !world.get_game_rule(&MOB_GRIEFING) {
157            return;
158        }
159
160        let snow = vanilla_blocks::SNOW.default_state();
161        let snow_behavior = BLOCK_BEHAVIORS.get_behavior(&vanilla_blocks::SNOW);
162        let position = self.position();
163        for i in 0..4 {
164            let x = (position.x + f64::from((i % 2 * 2 - 1) as f32 * 0.25)).floor() as i32;
165            let y = position.y.floor() as i32;
166            let z = (position.z + f64::from((i / 2 % 2 * 2 - 1) as f32 * 0.25)).floor() as i32;
167            let snow_pos = BlockPos::new(x, y, z);
168            if world.get_block_state(snow_pos).is_air()
169                && snow_behavior.can_survive(snow, world.as_ref(), snow_pos)
170            {
171                world.set_block(snow_pos, snow, UpdateFlags::UPDATE_ALL);
172                world.game_event(
173                    &vanilla_game_events::BLOCK_PLACE,
174                    snow_pos,
175                    &GameEventContext::new(Some(self), Some(snow)),
176                );
177            }
178        }
179    }

186impl Shearable for SnowGolemEntity {
187    /// Vanilla `SnowGolem.shear`: drops the pumpkin from the golem's head.
188    fn shear(&self, world: &Arc<World>, sound_source: SoundSource, tool: &ItemStack) {
189        world.play_sound_at(
190            &sound_events::ENTITY_SNOW_GOLEM_SHEAR,
191            sound_source,
192            self.position(),
193            1.0,
194            1.0,
195            None,
196        );
197        self.set_pumpkin(false);
198        let eye_height = self.get_eye_height();
199        let mut rng = rand::rng();
200        for drop in shearing_loot_items_with_rng(
201            self,
202            &vanilla_loot_tables::SHEARING_SNOW_GOLEM,
203            tool,
204            &mut rng,
205        ) {
206            let _ = self.spawn_at_location(drop, eye_height);
207        }
208    }
209
210    fn ready_for_shearing(&self) -> bool {
211        self.has_pumpkin()
212    }
213}

215impl RangedAttackMob for SnowGolemEntity {
216    fn perform_ranged_attack(&self, entity: &SharedEntity, target: &SharedEntity, _power: f32) {
217        let position = self.position();
218        let target_position = target.position();
219        let xd = target_position.x - position.x;
220        let yd = target.get_eye_y() - f64::from(1.1_f32);
221        let zd = target_position.z - position.z;
222        let yo = (xd * xd + zd * zd).sqrt() * f64::from(0.2_f32);
223
224        if let Some(world) = self.level() {
225            let mut item = ItemStack::new(&vanilla_items::SNOWBALL);
226            let snowball = SnowballEntity::new(
227                &vanilla_entities::SNOWBALL,
228                next_entity_id(),
229                DVec3::new(position.x, self.get_eye_y() - 0.1, position.z),
230                Arc::downgrade(&world),
231            );
232            snowball.set_owner_entity(Some(entity));
233            snowball.set_item_clamped(item.clone());
234            snowball.shoot(
235                DVec3::new(xd, yd + yo - snowball.position().y, zd),
236                SNOWBALL_SHOOT_POWER,
237                self.ranged_attack_uncertainty(&world),
238            );
239            let snowball: SharedEntity = Arc::new(snowball);
240            match world.try_add_entity(Arc::clone(&snowball)) {
241                Ok(()) => enchantment_helper::on_projectile_spawned(
242                    &world,
243                    &mut item,
244                    snowball.as_ref(),
245                    Some(entity),
246                ),
247                Err(error) => log::debug!("world rejected a snow golem snowball: {error}"),
248            }
249        }
250
251        let pitch = 0.4 / (rand::random::<f32>() * 0.4 + 0.8);
252        self.play_sound(&sound_events::ENTITY_SNOW_GOLEM_SHOOT, 1.0, pitch);
253    }
254
255    fn ranged_attack_uncertainty(&self, _world: &World) -> f32 {
256        SNOWBALL_UNCERTAINTY
257    }

380    fn mob_interact(&self, player: &Player, hand: InteractionHand) -> InteractionResult {
381        let item_stack = {
382            let inventory = player.inventory.lock();
383            let item_stack = inventory.get_item_in_hand(hand);
384            item_stack.copy_with_count(item_stack.count())
385        };
386        if !item_stack.is(&vanilla_items::SHEARS) || !self.ready_for_shearing() {
387            return InteractionResult::Pass;
388        }
389        if let Some(world) = self.level() {
390            self.shear(&world, SoundSource::Players, &item_stack);
391            self.game_event_with_source_entity(&vanilla_game_events::SHEAR, Some(player));
392            player
393                .inventory
394                .lock()
395                .hurt_item_in_hand(hand, 1, player.has_infinite_materials());
396        }
397        InteractionResult::Success
398    }

412mod tests {
413    use std::io::Cursor;
414    use std::sync::Weak;

416    use glam::DVec3;
417    use simdnbt::borrow::read_compound;
418    use simdnbt::owned::NbtCompound;
419    use steel_registry::{init_vanilla_registry, vanilla_entities};
420
421    use super::SnowGolemEntity;
422    use crate::entity::Entity;
423
424    #[test]
425    fn pumpkin_flag_round_trips_and_defaults_on() {
426        init_vanilla_registry();
427        let golem =
428            SnowGolemEntity::new(&vanilla_entities::SNOW_GOLEM, 1, DVec3::ZERO, Weak::new());
429        assert!(golem.has_pumpkin());
430        golem.set_pumpkin(false);
431        let mut nbt = NbtCompound::new();
432        golem.save_additional(&mut nbt);
433        assert_eq!(nbt.byte("Pumpkin"), Some(0));
434
435        let mut bytes = Vec::new();
436        NbtCompound::new().write(&mut bytes);
437        let borrowed = read_compound(&mut Cursor::new(&bytes))
438            .unwrap_or_else(|error| panic!("reborrow failed: {error}"));
439        let loaded =
440            SnowGolemEntity::new(&vanilla_entities::SNOW_GOLEM, 2, DVec3::ZERO, Weak::new());
441        loaded.set_pumpkin(false);
442        loaded.load_additional((&borrowed).into());
443        assert!(
444            loaded.has_pumpkin(),
445            "a missing Pumpkin tag means the golem has one"
446        );
447    }
