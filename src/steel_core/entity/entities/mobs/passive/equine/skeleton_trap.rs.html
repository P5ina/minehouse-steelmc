24/// Vanilla `SkeletonTrapGoal.LIGHTNING_INVULNERABLE_TICKS`.
25const LIGHTNING_INVULNERABLE_TICKS: i32 = 60;
26/// Vanilla `canUse` player range.
27const TRIGGER_RANGE: f64 = 10.0;
28/// Extra horsemen the trap springs next to the trap horse.
29const EXTRA_HORSEMEN: usize = 3;
30/// Vanilla `triangle(0.0, 1.1485)` spread of the extra horses' push.
31const HORSE_PUSH_DEVIATION: f64 = 1.1485;
32
33/// Vanilla `SkeletonTrapGoal`. Steel keeps it registered for every skeleton
34/// horse and gates it on `isTrap`, which matches vanilla adding and removing
35/// it in `setTrap`: the goal has no controls, so it never blocks others.
36pub(super) struct SkeletonTrapGoal;
37
38impl SkeletonTrapGoal {
39    pub(super) const fn new() -> Self {
40        Self
41    }
42}

44fn trap_horse(mob: &dyn PathfinderMob) -> Option<&SkeletonHorseEntity> {
45    mob.downcast_ref::<SkeletonHorseEntity>()
46        .filter(|horse| horse.is_trap())
47}
48
49impl Goal for SkeletonTrapGoal {
50    fn controls(&self) -> GoalControls {
51        GoalControls::EMPTY
52    }
53
54    /// Vanilla `canUse`: `level.hasNearbyAlivePlayer(x, y, z, 10.0)`.
55    fn can_use(&mut self, mob: &dyn PathfinderMob) -> bool {
56        if trap_horse(mob).is_none() {
57            return false;
58        }
59        let Some(world) = mob.level() else {
60            return false;
61        };
62        world
63            .nearest_player(mob.position(), TRIGGER_RANGE, |player| {
64                !player.is_spectator() && Entity::is_alive(player)
65            })
66            .is_some()
67    }
68
69    /// Vanilla `tick`.
70    fn tick(&mut self, mob: &dyn PathfinderMob, entity: &SharedEntity) {
71        let Some(horse) = trap_horse(mob) else {
72            return;
73        };
74        let Some(world) = mob.level() else {
75            return;
76        };
77        let difficulty = world.current_difficulty_at(mob.block_position());
78        horse.set_trap(false);
79        horse.set_tamed(true);
80        horse.set_age(0);
81
82        let position = mob.position();
83        let Ok(bolt) = create_entity_instance(&world, &vanilla_entities::LIGHTNING_BOLT, position)
84        else {
85            return;
86        };
87        bolt.snap_to(position, 0.0, 0.0);
88        if let Some(bolt) = bolt.downcast_ref::<LightningBoltEntity>() {
89            bolt.set_visual_only(true);
90        }
91        if world.try_add_entity(bolt).is_err() {
92            return;
93        }
94
95        let Some(skeleton) = create_skeleton(&world, difficulty, position) else {
96            return;
97        };
98        if world.try_add_entity(Arc::clone(&skeleton)).is_err() {
99            return;
100        }
101        let _ = start_riding_entities(&skeleton, entity);
102
103        let mut random = LegacyRandom::from_seed(rand::random());
104        for _ in 0..EXTRA_HORSEMEN {
105            let Some(other_horse) = create_horse(&world, difficulty, position) else {
106                continue;
107            };
108            let Some(other_skeleton) = create_skeleton(&world, difficulty, position) else {
109                continue;
110            };
111            other_horse.push_impulse(DVec3::new(
112                random.triangle(0.0, HORSE_PUSH_DEVIATION),
113                0.0,
114                random.triangle(0.0, HORSE_PUSH_DEVIATION),
115            ));
116            if world.try_add_entity(Arc::clone(&other_horse)).is_err() {
117                continue;
118            }
119            if world.try_add_entity(Arc::clone(&other_skeleton)).is_ok() {
120                let _ = start_riding_entities(&other_skeleton, &other_horse);
121            }
122        }
123    }
124}

126/// Vanilla `SkeletonTrapGoal.createHorse`.
127fn create_horse(
128    world: &Arc<World>,
129    _difficulty: DifficultyInstance,
130    position: DVec3,
131) -> Option<SharedEntity> {
132    let horse = create_entity_instance(world, &vanilla_entities::SKELETON_HORSE, position).ok()?;
133    if let Some(mob) = horse.as_mob() {
134        let _ = mob.finalize_spawn(world, EntitySpawnReason::Triggered, None);
135        mob.set_persistence_required();
136    }
137    if let Some(living) = horse.as_living_entity() {
138        living
139            .living_base()
140            .set_invulnerable_time(LIGHTNING_INVULNERABLE_TICKS);
141    }
142    if let Some(skeleton_horse) = horse.downcast_ref::<SkeletonHorseEntity>() {
143        skeleton_horse.set_tamed(true);
144        skeleton_horse.set_age(0);
145    }
146    Some(horse)
147}

149/// Vanilla `SkeletonTrapGoal.createSkeleton`.
150fn create_skeleton(
151    world: &Arc<World>,
152    difficulty: DifficultyInstance,
153    position: DVec3,
154) -> Option<SharedEntity> {
155    let skeleton = create_entity_instance(world, &vanilla_entities::SKELETON, position).ok()?;
156    let mob = skeleton.as_mob()?;
157    let _ = mob.finalize_spawn(world, EntitySpawnReason::Triggered, None);
158    mob.living_base()
159        .set_invulnerable_time(LIGHTNING_INVULNERABLE_TICKS);
160    mob.set_persistence_required();
161    {
162        let mut equipment = mob.living_base().equipment().lock();
163        if equipment.get_ref(EquipmentSlot::Head).is_empty() {
164            let _ = equipment.set(
165                EquipmentSlot::Head,
166                ItemStack::new(&vanilla_items::IRON_HELMET),
167            );
168        }
169    }
170    let mut random = LegacyRandom::from_seed(rand::random());
171    reenchant_with_mob_spawn_equipment(mob, EquipmentSlot::MainHand, &mut random, difficulty);
172    reenchant_with_mob_spawn_equipment(mob, EquipmentSlot::Head, &mut random, difficulty);
173    Some(skeleton)
174}
