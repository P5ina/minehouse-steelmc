60/// An undead horse that zombies ride.
61#[entity_behavior(class = "ZombieHorse", spawn_rule)]
62pub struct ZombieHorseEntity {
63    base: EntityBase,
64    entity_type: EntityTypeRef,
65    living_base: LivingEntityBase,
66    mob_base: MobBase,
67    ageable_base: AgeableMobBase,
68    animal_base: AnimalBase,
69    horse_base: AbstractHorseBase,
70    entity_data: SyncMutex<ZombieHorseEntityData>,
71    /// The jockey from `finalizeSpawn`, seated on the first tick once this
72    /// horse has a shared handle in the world.
73    pending_rider: SyncMutex<Option<SharedEntity>>,
74}

81impl ZombieHorseEntity {
82    /// Vanilla `SpawnPlacements` predicate: `Monster.checkMonsterSpawnRules`.
83    pub(crate) const SPAWN_RULE: SpawnRule = |entity_type, level, spawn_reason, pos, random| {
84        is_spawn_allowed_on_difficulty(entity_type, level)
85            && (spawn_reason.ignores_light_requirements()
86                || is_dark_enough_to_spawn(level, pos, random))
87            && <Self as Mob>::check_mob_spawn_rules(entity_type, level, spawn_reason, pos)
88    };
89
90    /// Vanilla `ZombieHorse.generateZombieHorseJumpStrength`.
91    fn generate_jump_strength(mut probability: impl FnMut() -> f64) -> f64 {
92        BASE_JUMP_STRENGTH
93            + probability() * PER_RANDOM_JUMP_STRENGTH
94            + probability() * PER_RANDOM_JUMP_STRENGTH
95            + probability() * PER_RANDOM_JUMP_STRENGTH
96    }
97
98    /// Vanilla `ZombieHorse.generateZombieHorseSpeed`.
99    fn generate_speed(mut probability: impl FnMut() -> f64) -> f64 {
100        (BASE_SPEED
101            + probability() * PER_RANDOM_SPEED
102            + probability() * PER_RANDOM_SPEED
103            + probability() * PER_RANDOM_SPEED)
104            / f64::from(SPEED_FACTOR)
105    }
106
107    /// Creates a new zombie horse at runtime.
108    #[must_use]
109    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
110        Self::new_with_base(
111            EntityBase::new(id, position, entity_type.dimensions, world),
112            entity_type,
113        )
114    }
115
116    /// Reconstructs a zombie horse from persisted base entity state.
117    #[must_use]
118    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
119        Self::new_with_base(
120            EntityBase::from_load(load, entity_type.dimensions),
121            entity_type,
122        )
123    }
124
125    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
126        let living_base = LivingEntityBase::new(entity_type);
127        let mob_base = MobBase::new();
128        AnimalBase::initialize_pathfinding_malus(&mob_base);
129        let mut entity_data = ZombieHorseEntityData::new();
130        living_base.initialize_synced_data(&mut entity_data);
131        let horse = Self {
132            base,
133            entity_type,
134            living_base,
135            mob_base,
136            ageable_base: AgeableMobBase::new(),
137            animal_base: AnimalBase::new(),
138            horse_base: AbstractHorseBase::new(0),
139            entity_data: SyncMutex::new(entity_data),
140            pending_rider: SyncMutex::new(None),
141        };
142        horse.register_horse_goals(&horse.mob_base, add_zombie_horse_behaviour_goals);
143        horse
144    }
145}

164    /// Vanilla `ZombieHorse.mobInteract`.
165    fn mob_interact_for(&self, player: &Player, hand: InteractionHand) -> InteractionResult {
166        let should_open_inventory =
167            !AgeableMob::is_baby(self) && self.is_tamed() && player.is_secondary_use_active();
168        if !self.is_vehicle() && !should_open_inventory {
169            let item_stack = {
170                let inventory = player.inventory.lock();
171                let item_stack = inventory.get_item_in_hand(hand);
172                item_stack.copy_with_count(item_stack.count())
173            };
174            if let Some(result) = self.equine_feed_interact(player, hand, &item_stack) {
175                return result;
176            }
177        }
178        self.horse_mob_interact(player, hand)
179    }

227    /// Vanilla `ZombieHorse.finalizeSpawn`: a natural spawn brings a
228    /// spear-wielding zombie rider.
229    fn finalize_spawn_for(
230        &self,
231        world: &Arc<World>,
232        spawn_reason: EntitySpawnReason,
233        group_data: Option<SpawnGroupData>,
234    ) -> Option<SpawnGroupData> {
235        if spawn_reason == EntitySpawnReason::Natural
236            && let Ok(zombie) =
237                create_entity_instance(world, &vanilla_entities::ZOMBIE, self.position())
238        {
239            zombie.set_rotation((self.rotation().0, 0.0));
240            if let Some(zombie_mob) = zombie.as_mob() {
241                let _ = zombie_mob.finalize_spawn(world, spawn_reason, None);
242            }
243            if let Some(zombie_living) = zombie.as_living_entity() {
244                let _ = zombie_living.living_base().equipment().lock().set(
245                    EquipmentSlot::MainHand,
246                    ItemStack::new(&vanilla_items::IRON_SPEAR),
247                );
248            }
249            *self.pending_rider.lock() = Some(zombie);
250        }
251        self.finalize_horse_spawn(world, spawn_reason, group_data)
252    }
253
254    /// Adds and seats the jockey from `finalizeSpawn`.
255    fn tick_extra(&self, entity: &SharedEntity) {
256        let Some(rider) = self.pending_rider.lock().take() else {
257            return;
258        };
259        if self.is_removed() {
260            return;
261        }
262        let Some(world) = self.level() else {
263            return;
264        };
265        if world.try_add_entity(Arc::clone(&rider)).is_ok() {
266            let _ = start_riding_entities(&rider, entity);
267        }
268    }

302    /// Vanilla `ZombieHorse.randomizeAttributes`.
303    fn randomize_attributes(&self) {
304        let mut attributes = self.attributes().lock();
305        attributes.set_base_value(
306            vanilla_attributes::JUMP_STRENGTH,
307            Self::generate_jump_strength(rand::random::<f64>),
308        );
309        attributes.set_base_value(
310            vanilla_attributes::MOVEMENT_SPEED,
311            Self::generate_speed(rand::random::<f64>),
312        );
313    }

344/// Vanilla `ZombieHorse.isFood`: the zombie horse food tag.
