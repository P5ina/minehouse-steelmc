20/// Vanilla `Mob.createEquipmentSlotContainer`: one equipment slot of a mob
21/// exposed as a container.
22///
23/// Steel containers hand out item references, so this keeps a copy that is
24/// written through to the mob on every change and refreshed from it each tick.
25pub struct MountEquipmentContainer {
26    mount: WeakEntity,
27    slot: EquipmentSlot,
28    item: [ItemStack; 1],
29}
30
31// SAFETY: This key is owned by Steel and uniquely identifies `MountEquipmentContainer`.
32unsafe impl DowncastType for MountEquipmentContainer {
33    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("steel:container/mount_equipment");
34}
35
36impl MountEquipmentContainer {
37    /// Mirrors `slot` of `mount`.
38    #[must_use]
39    pub fn new(mount: &SharedEntity, slot: EquipmentSlot) -> Self {
40        let item = mount
41            .as_living_entity()
42            .map_or_else(ItemStack::empty, |living| {
43                living
44                    .living_base()
45                    .equipment()
46                    .lock()
47                    .get_ref(slot)
48                    .clone()
49            });
50        Self {
51            mount: Arc::downgrade(mount),
52            slot,
53            item: [item],
54        }
55    }
56
57    /// Re-reads the mob's equipment, picking up changes made outside the menu.
58    fn refresh(&mut self) {
59        if let Some(mount) = self.mount.upgrade()
60            && let Some(living) = mount.as_living_entity()
61        {
62            self.item[0] = living
63                .living_base()
64                .equipment()
65                .lock()
66                .get_ref(self.slot)
67                .clone();
68        }
69    }
70
71    /// Vanilla `setTheItem`: equips the mob and keeps it from despawning.
72    fn write_through(&self) {
73        let Some(mount) = self.mount.upgrade() else {
74            return;
75        };
76        let Some(mob) = mount.as_mob() else {
77            return;
78        };
79        let item = &self.item[0];
80        let previous = {
81            let mut equipment = mob.living_base().equipment().lock();
82            if equipment.get_ref(self.slot) == item {
83                return;
84            }
85            equipment.set(self.slot, item.clone())
86        };
87        mob.on_equip_item(self.slot, &previous, item);
88        if !item.is_empty() {
89            mob.set_guaranteed_drop(self.slot);
90            mob.set_persistence_required();
91        }
92    }
93}

95impl Container for MountEquipmentContainer {
96    fn items(&self) -> &[ItemStack] {
97        &self.item
98    }
99
100    fn items_mut(&mut self) -> &mut [ItemStack] {
101        &mut self.item
102    }
103
104    fn set_item(&mut self, slot: usize, stack: ItemStack) {
105        if let Some(item) = self.item.get_mut(slot) {
106            *item = stack;
107        }
108        self.write_through();
109    }
110
111    fn get_max_stack_size(&self) -> i32 {
112        1
113    }
114
115    fn set_changed(&mut self) {
116        self.write_through();
117    }
118}

120/// Vanilla `ArmorSlot` as the horse menu builds it: placement follows the
121/// mount's equippable rules, and `isActive` gates the whole slot.
122struct MountEquipmentSlot {
123    base: NormalSlot,
124    mount: WeakEntity,
125    slot: EquipmentSlot,
126    active: bool,
127}
128
129// SAFETY: This key is owned by Steel and uniquely identifies `MountEquipmentSlot`.
130unsafe impl DowncastType for MountEquipmentSlot {
131    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("steel:slot/mount_equipment");
132}
133
134impl Slot for MountEquipmentSlot {
135    fn storage(&self) -> &SlotStorage {
136        self.base.storage()
137    }
138
139    fn get_item<'a>(&self, guard: &'a ContainerLockGuard) -> &'a ItemStack {
140        self.base.get_item(guard)
141    }
142
143    fn get_item_mut<'a>(&self, guard: &'a mut ContainerLockGuard) -> &'a mut ItemStack {
144        self.base.get_item_mut(guard)
145    }
146
147    fn set_item(&self, guard: &mut ContainerLockGuard, stack: ItemStack) {
148        self.base.set_item(guard, stack);
149    }
150
151    /// Vanilla `ArmorSlot.mayPlace`.
152    fn may_place(&self, stack: &ItemStack) -> bool {
153        self.active
154            && self.mount.upgrade().is_some_and(|mount| {
155                mount
156                    .as_living_entity()
157                    .is_some_and(|living| living.is_equippable_in_slot(stack, self.slot))
158            })
159    }
160
161    /// Vanilla `ArmorSlot.mayPickup`.
162    fn may_pickup(&self, guard: &ContainerLockGuard, player: &Player) -> bool {
163        let item = self.get_item(guard);
164        self.active
165            && (item.is_empty()
166                || player.has_infinite_materials()
167                || !item.has_enchantment_effect(EnchantmentEffectComponent::PreventArmorChange))
168    }
169
170    fn get_max_stack_size(&self, _guard: &ContainerLockGuard) -> i32 {
171        1
172    }
173
174    fn set_changed(&self, guard: &mut ContainerLockGuard) {
175        self.base.set_changed(guard);
176    }
177
178    fn get_container_slot(&self) -> usize {
179        self.base.get_container_slot()
180    }
181}

183fn equipment_section(
184    builder: &mut MenuBuilder,
185    mount: &SharedEntity,
186    slot: EquipmentSlot,
187    active: bool,
188) -> (ContainerRef, Section) {
189    let container: ContainerRef = MountEquipmentContainer::new(mount, slot)
190        .into_shared()
191        .into();
192    let weak = Arc::downgrade(mount);
193    let section = builder.section_with(
194        &container,
195        1,
196        SectionKind::custom(move |container, index| {
197            Box::new(MountEquipmentSlot {
198                base: NormalSlot::new(container.clone(), index),
199                mount: Weak::clone(&weak),
200                slot,
201                active,
202            })
203        }),
204    );
205    (container, section)
206}

208/// Builds vanilla `HorseInventoryMenu` for `mount`.
209///
210/// Slot 0 is the saddle, slot 1 the body armor, then the mount's storage in
211/// `inventory_columns` columns of three rows, then the player inventory.
212#[must_use]
213pub fn horse_inventory_menu(
214    player_inventory: &Shared<PlayerInventory>,
215    container_id: u8,
216    mount: &SharedEntity,
217    mount_inventory: Shared<SimpleContainer>,
218    inventory_columns: usize,
219) -> Menu {
220    let mut builder = MenuBuilder::new(None, container_id);
221    let living = mount.as_living_entity();
222    // Vanilla checks `instanceof Llama`, which trader llamas extend.
223    let is_llama = mount.entity_type() == &vanilla_entities::LLAMA
224        || mount.entity_type() == &vanilla_entities::TRADER_LLAMA;
225    let saddle_active = living.is_some_and(|living| living.can_use_slot(EquipmentSlot::Saddle))
226        && REGISTRY
227            .entity_types
228            .is_in_tag(mount.entity_type(), &EntityTypeTag::CAN_EQUIP_SADDLE);
229    let body_active = living.is_some_and(|living| living.can_use_slot(EquipmentSlot::Body))
230        && (is_llama
231            || REGISTRY
232                .entity_types
233                .is_in_tag(mount.entity_type(), &EntityTypeTag::CAN_WEAR_HORSE_ARMOR));
234
235    let (saddle_container, saddle) =
236        equipment_section(&mut builder, mount, EquipmentSlot::Saddle, saddle_active);
237    let (body_container, body) =
238        equipment_section(&mut builder, mount, EquipmentSlot::Body, body_active);
239    let storage_size = inventory_columns * INVENTORY_ROWS;
240    let storage = builder.section(&mount_inventory, storage_size);
241    let player = builder.player_inventory(player_inventory);
242
243    builder.build(MountMenuKind {
244        mount: Arc::downgrade(mount),
245        mount_inventory,
246        equipment: [saddle_container, body_container],
247        saddle,
248        body,
249        storage,
250        player,
251    })
252}

254/// Vanilla `AbstractMountInventoryMenu` behavior.
255struct MountMenuKind {
256    mount: WeakEntity,
257    mount_inventory: Shared<SimpleContainer>,
258    equipment: [ContainerRef; 2],
259    saddle: Section,
260    body: Section,
261    storage: Section,
262    player: PlayerInventorySections,
263}

265// SAFETY: This key is owned by Steel and uniquely identifies the mount menu kind.
266unsafe impl DowncastType for MountMenuKind {
267    const TYPE_KEY: DowncastTypeKey = DowncastTypeKey::new("steel:menu/mount_inventory");
268}

270impl MountMenuKind {
271    fn equipment_slot_accepts(
272        behavior: &MenuBehavior,
273        guard: &ContainerLockGuard,
274        section: Section,
275        stack: &ItemStack,
276    ) -> bool {
277        let slot = &behavior.slots()[section.start()];
278        slot.may_place(stack) && !slot.has_item(guard)
279    }
280}

282impl MenuKind for MountMenuKind {
283    /// Vanilla `AbstractMountInventoryMenu.stillValid`, with the equipment
284    /// containers' `stillValid`.
285    fn still_valid(&self, _behavior: &MenuBehavior, player: &Player) -> bool {
286        let Some(mount) = self.mount.upgrade() else {
287            return false;
288        };
289        let inventory_changed = mount.as_abstract_horse().is_none_or(|horse| {
290            horse
291                .horse_base()
292                .has_inventory_changed(&self.mount_inventory)
293        });
294        let rides_mount = player
295            .vehicle()
296            .is_some_and(|vehicle| vehicle.id() == mount.id());
297        !inventory_changed
298            && mount.is_alive()
299            && (rides_mount
300                || player.is_within_entity_interaction_range_with_buffer(
301                    mount.bounding_box(),
302                    MOUNT_MENU_REACH_BUFFER,
303                ))
304    }
305
306    fn on_tick(
307        &mut self,
308        _behavior: &mut MenuBehavior,
309        guard: &mut ContainerLockGuard,
310        _player: &Player,
311    ) {
312        for container in &self.equipment {
313            if let Some(equipment) =
314                guard.get_typed_mut::<MountEquipmentContainer>(container.container_id())
315            {
316                equipment.refresh();
317            }
318        }
319    }
320
321    /// Vanilla `AbstractMountInventoryMenu.quickMoveStack`.
322    fn quick_move(
323        &mut self,
324        behavior: &mut MenuBehavior,
325        guard: &mut ContainerLockGuard,
326        slot_index: usize,
327        player: &Player,
328    ) -> Option<ItemStack> {
329        if slot_index >= behavior.slots().len() {
330            return Some(ItemStack::empty());
331        }
332        let clicked = behavior.slots()[slot_index].get_item(guard).clone();
333        if clicked.is_empty() {
334            return Some(ItemStack::empty());
335        }
336        let mut remaining = clicked.clone();
337        let all = self.player.all();
338
339        let from_mount = slot_index < all.start();
340        let moved = if from_mount {
341            behavior.move_item_stack_to(
342                guard,
343                slot_index,
344                &mut remaining,
345                all.start(),
346                all.end(),
347                FillDirection::Backward,
348            )
349        } else if Self::equipment_slot_accepts(behavior, guard, self.body, &clicked) {
350            behavior.move_item_stack_to(
351                guard,
352                slot_index,
353                &mut remaining,
354                self.body.start(),
355                self.body.end(),
356                FillDirection::Forward,
357            )
358        } else if Self::equipment_slot_accepts(behavior, guard, self.saddle, &clicked) {
359            behavior.move_item_stack_to(
360                guard,
361                slot_index,
362                &mut remaining,
363                self.saddle.start(),
364                self.saddle.end(),
365                FillDirection::Forward,
366            )
367        } else if !self.storage.is_empty()
368            && behavior.move_item_stack_to(
369                guard,
370                slot_index,
371                &mut remaining,
372                self.storage.start(),
373                self.storage.end(),
374                FillDirection::Forward,
375            )
376        {
377            true
378        } else {
379            // Vanilla swaps between the main inventory and the hotbar here,
380            // then ends the quick-move loop whatever happened.
381            let (start, end) = if self.player.hotbar().contains(slot_index) {
382                (self.player.main().start(), self.player.main().end())
383            } else {
384                (self.player.hotbar().start(), self.player.hotbar().end())
385            };
386            if behavior.move_item_stack_to(
387                guard,
388                slot_index,
389                &mut remaining,
390                start,
391                end,
392                FillDirection::Forward,
393            ) {
394                behavior.update_quick_move_source(guard, slot_index, &remaining, &clicked);
395            }
396            return Some(ItemStack::empty());
397        };
398
399        if !moved {
400            return Some(ItemStack::empty());
401        }
402        behavior.update_quick_move_source(guard, slot_index, &remaining, &clicked);
403        if remaining.count() == clicked.count() {
404            return Some(ItemStack::empty());
405        }
406        if let Some(remainder) = behavior.slots()[slot_index].on_take(guard, &remaining, player) {
407            player.add_item_or_drop_with_guard(guard, remainder, Prediction::Predicted);
408        }
409        Some(clicked)
410    }
411}

413/// Vanilla `AbstractHorse.openCustomInventoryScreen`: a tamed mount opens its
414/// inventory for a player who rides it or stands by while it is unridden.
415pub(super) fn open_horse_inventory<H: AbstractHorse + ?Sized>(horse: &H, player: &Player) {
416    if (horse.is_vehicle()
417        && !horse
418            .passengers()
419            .iter()
420            .any(|passenger| passenger.id() == player.id()))
421        || !horse.is_tamed()
422    {
423        return;
424    }
425    let Some(world) = horse.level() else {
426        return;
427    };
428    let Some(mount) = world.get_entity_by_id(horse.id()) else {
429        return;
430    };
431    let columns = horse.inventory_columns();
432    let mount_inventory = horse.horse_base().inventory();
433    let entity_id = horse.id();
434    player.open_mount_menu(
435        i32::try_from(columns).unwrap_or(0),
436        entity_id,
437        move |context| {
438            horse_inventory_menu(
439                &context.player.inventory,
440                context.container_id,
441                &mount,
442                mount_inventory,
443                columns,
444            )
445        },
446    );
