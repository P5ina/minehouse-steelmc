63/// Vanilla `AbstractChestedHorse`: a horse that can carry a chest.
64pub trait AbstractChestedHorse: AbstractHorse {
65    /// Vanilla `hasChest`.
66    fn has_chest(&self) -> bool;
67
68    /// Vanilla `setChest`.
69    fn set_chest(&self, chest: bool);
70
71    /// Vanilla `playChestEquipsSound`.
72    fn play_chest_equips_sound(&self) {
73        self.play_sound(
74            &sound_events::ENTITY_DONKEY_CHEST,
75            1.0,
76            (rand::random::<f32>() - rand::random::<f32>()) * 0.2 + 1.0,
77        );
78    }
79
80    /// Vanilla `equipChest`.
81    fn equip_chest(&self, player: &Player, hand: InteractionHand) {
82        self.set_chest(true);
83        self.play_chest_equips_sound();
84        if !player.has_infinite_materials() {
85            player.inventory.lock().shrink_item_in_hand(hand, 1);
86        }
87        self.create_inventory();
88    }
89
90    /// Vanilla `AbstractChestedHorse.mobInteract`.
91    fn chested_mob_interact(&self, player: &Player, hand: InteractionHand) -> InteractionResult {
92        let should_open_inventory =
93            !AgeableMob::is_baby(self) && self.is_tamed() && player.is_secondary_use_active();
94        let holding_golden_dandelion = player.is_holding(&mut |item_stack: &ItemStack| {
95            item_stack.is(&vanilla_items::GOLDEN_DANDELION)
96        });
97        if !self.is_vehicle()
98            && !should_open_inventory
99            && (!AgeableMob::is_baby(self) || !holding_golden_dandelion)
100        {
101            let item_stack = {
102                let inventory = player.inventory.lock();
103                let item_stack = inventory.get_item_in_hand(hand);
104                item_stack.copy_with_count(item_stack.count())
105            };
106            if let Some(result) = self.equine_feed_interact(player, hand, &item_stack) {
107                return result;
108            }
109            if !item_stack.is_empty() && !self.has_chest() && item_stack.is(&vanilla_items::CHEST) {
110                self.equip_chest(player, hand);
111                return InteractionResult::Success;
112            }
113        }
114        self.horse_mob_interact(player, hand)
115    }
116
117    /// Vanilla `AbstractChestedHorse.dropEquipment`, after the inventory.
118    fn drop_chest(&self) {
119        if self.has_chest() {
120            let _ = self.spawn_at_location(ItemStack::new(&vanilla_items::CHEST), 0.0);
121            self.set_chest(false);
122        }
123    }
124
125    /// Vanilla `AbstractChestedHorse.addAdditionalSaveData`.
126    fn save_chest(&self, nbt: &mut NbtCompound) {
127        nbt.insert("ChestedHorse", i8::from(self.has_chest()));
128        if !self.has_chest() {
129            return;
130        }
131        let inventory = self.horse_base().inventory();
132        let inventory = inventory.lock();
133        let items = inventory
134            .items()
135            .iter()
136            .enumerate()
137            .filter(|(_, item)| !item.is_empty())
138            .filter_map(|(slot, item)| {
139                let NbtTag::Compound(mut item_nbt) = item.clone().to_nbt_tag() else {
140                    return None;
141                };
142                item_nbt.insert("Slot", slot as i8);
143                Some(item_nbt)
144            })
145            .collect();
146        nbt.insert("Items", NbtList::Compound(items));
147    }
148
149    /// Vanilla `AbstractChestedHorse.readAdditionalSaveData`.
150    fn load_chest(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
151        self.set_chest(nbt.byte("ChestedHorse").is_some_and(|value| value != 0));
152        self.create_inventory();
153        if !self.has_chest() {
154            return;
155        }
156        let Some(compounds) = nbt.list("Items").and_then(|items| items.compounds()) else {
157            return;
158        };
159        let inventory = self.horse_base().inventory();
160        let mut inventory = inventory.lock();
161        let size = inventory.get_container_size();
162        for compound in compounds {
163            let Some(slot) = compound.byte("Slot") else {
164                continue;
165            };
166            let slot = i32::from(slot) & 0xFF;
167            if (0..size as i32).contains(&slot)
168                && let Some(item) = ItemStack::from_borrowed_compound(&compound)
169            {
170                inventory.set_item(slot as usize, item);
171            }
172        }
173    }
174
175    /// Vanilla `AbstractChestedHorse.randomizeAttributes`.
176    fn randomize_chested_attributes(&self) {
177        self.attributes().lock().set_base_value(
178            vanilla_attributes::MAX_HEALTH,
179            f64::from(generate_max_health(|bound| rand::random_range(0..bound))),
180        );
181    }
182}

220macro_rules! chested_entity {
221    ($entity:ident, $data:ty, $baby_dimensions:expr,
222        hooks { $($hook:item)* }
223        chested { $($chested:item)* }
224    ) => {
225        impl $entity {
226            /// Vanilla `SpawnPlacements` predicate: `Animal.checkAnimalSpawnRules`.
227            pub(crate) const SPAWN_RULE: crate::entity::SpawnRule =
228                |entity_type, level, spawn_reason, pos, random| {
229                    <Self as Animal>::check_animal_spawn_rules(
230                        entity_type,
231                        level,
232                        spawn_reason,
233                        pos,
234                        random,
235                    )
236                };
237
238            /// Creates a new entity at runtime.
239            #[must_use]
240            pub fn new(
241                entity_type: EntityTypeRef,
242                id: i32,
243                position: DVec3,
244                world: Weak<World>,
245            ) -> Self {
246                Self::new_with_base(
247                    EntityBase::new(id, position, entity_type.dimensions, world),
248                    entity_type,
249                )
250            }
251
252            /// Reconstructs the entity from persisted base entity state.
253            #[must_use]
254            pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
255                Self::new_with_base(
256                    EntityBase::from_load(load, entity_type.dimensions),
257                    entity_type,
258                )
259            }
260
261            fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
262                let living_base = LivingEntityBase::new(entity_type);
263                let mob_base = MobBase::new();
264                AnimalBase::initialize_pathfinding_malus(&mob_base);
265                let mut entity_data = <$data>::new();
266                living_base.initialize_synced_data(&mut entity_data);
267                let horse = Self {
268                    base,
269                    entity_type,
270                    living_base,
271                    mob_base,
272                    ageable_base: AgeableMobBase::new(),
273                    animal_base: AnimalBase::new(),
274                    horse_base: AbstractHorseBase::new(0),
275                    entity_data: SyncMutex::new(entity_data),
276                };
277                horse.register_horse_goals(&horse.mob_base, add_default_behaviour_goals);
278                horse
279            }
280        }
281
282        impl EquineHooks for $entity {
283            fn baby_dimensions(&self) -> EntityDimensions {
284                $baby_dimensions.scale(CHESTED_BABY_SCALE)
285            }
286
287            fn mob_interact_for(&self, player: &Player, hand: InteractionHand) -> InteractionResult {
288                self.chested_mob_interact(player, hand)
289            }
290
291            fn save_extra(&self, nbt: &mut NbtCompound) {
292                self.save_chest(nbt);
293            }
294
295            fn load_extra(&self, nbt: BorrowedNbtCompoundView<'_, '_>) {
296                self.load_chest(nbt);
297            }
298
299            fn ai_step_extra(&self) {}
300
301            fn drop_equipment_extra(&self) {
302                self.drop_chest();
303            }
304
305            fn can_use_slot_for(&self, slot: EquipmentSlot) -> bool {
306                self.can_use_horse_slot(slot)
307            }
308
309            fn can_fall_in_love_for(&self) -> bool {
310                self.in_love_time() <= 0
311            }
312
313            fn can_age_up_for(&self) -> bool {
314                AgeableMob::is_baby(self) && !self.is_age_locked()
315            }
316
317            fn is_food_for(&self, item_stack: &ItemStack) -> bool {
318                self.is_horse_food(item_stack)
319            }
320
321            fn remove_when_far_away_for(&self, dist_sqr: f64) -> bool {
322                self.remove_when_far_away_animal(dist_sqr)
323            }
324
325            fn finalize_spawn_for(
326                &self,
327                world: &Arc<World>,
328                spawn_reason: EntitySpawnReason,
329                group_data: Option<SpawnGroupData>,
330            ) -> Option<SpawnGroupData> {
331                self.finalize_horse_spawn(world, spawn_reason, group_data)
332            }
333
334            $($hook)*
335        }
336
337        impl AbstractChestedHorse for $entity {
338            fn has_chest(&self) -> bool {
339                *self.entity_data.lock().abstract_chested_horse().id_chest.get()
340            }
341
342            fn set_chest(&self, chest: bool) {
343                self.entity_data
344                    .lock()
345                    .abstract_chested_horse_mut()
346                    .id_chest
347                    .set(chest);
348            }
349
350            $($chested)*
351        }
352
353        impl_equine_entity!(
354            $entity, abstract_horse, abstract_horse_mut,
355            entity {}
356            living {}
357            mob {}
358        );
359    };
360}

387            ) -> Option<SharedEntity> {
388                let baby_type = if partner.entity_type() == &vanilla_entities::HORSE {
389                    &vanilla_entities::MULE
390                } else {
391                    &vanilla_entities::DONKEY

393                let baby = create_offspring(baby_type, self, world)?;
394                if let Some(baby_ageable) = baby.as_ageable_mob() {
395                    self.set_offspring_attributes(partner, baby_ageable);
396                }
397                Some(baby)
398            }
