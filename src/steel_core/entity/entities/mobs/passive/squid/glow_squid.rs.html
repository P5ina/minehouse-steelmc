36/// Vanilla `GlowSquid.checkGlowSquidSpawnRules`: deep (33 or more blocks
37/// below sea level), in complete darkness and inside a water block.
38fn check_glow_squid_spawn_rules(
39    _entity_type: EntityTypeRef,
40    level: &dyn ServerLevelAccessor,
41    _spawn_reason: EntitySpawnReason,
42    pos: BlockPos,
43    _random: &mut dyn Rng,
44) -> bool {
45    pos.y() <= level.sea_level() - GLOW_SQUID_MIN_SPAWN_DEPTH
46        && level.raw_brightness(pos, 0) == 0
47        && level.get_block_state(pos).get_block() == &vanilla_blocks::WATER
48}
49
50/// How far below sea level glow squids spawn at the least.
51const GLOW_SQUID_MIN_SPAWN_DEPTH: i32 = 33;
52
53/// Vanilla `GlowSquid`: ticks dark for 100 ticks after a mob hurts it.
54const DARK_TICKS_AFTER_HURT: i32 = 100;

56/// Vanilla glow squid: a squid with glowing ink that dims when hurt.
57#[entity_behavior(class = "GlowSquid", spawn_rule)]
58pub struct GlowSquidEntity {
59    base: EntityBase,
60    entity_type: EntityTypeRef,
61    living_base: LivingEntityBase,
62    mob_base: MobBase,
63    ageable_base: AgeableMobBase,
64    entity_data: SyncMutex<GlowSquidEntityData>,
65    state: SyncMutex<SquidState>,
66}

73impl GlowSquidEntity {
74    /// Vanilla `SpawnPlacements` predicate: `GlowSquid.checkGlowSquidSpawnRules`.
75    pub(crate) const SPAWN_RULE: SpawnRule = check_glow_squid_spawn_rules;

77    /// Creates a new glow squid.
78    #[must_use]
79    pub fn new(entity_type: EntityTypeRef, id: i32, position: DVec3, world: Weak<World>) -> Self {
80        Self::new_with_base(
81            EntityBase::new(id, position, entity_type.dimensions, world),
82            entity_type,
83        )
84    }
85
86    /// Recreates a glow squid from saved entity state.
87    #[must_use]
88    pub fn from_saved(entity_type: EntityTypeRef, load: EntityBaseLoad) -> Self {
89        Self::new_with_base(
90            EntityBase::from_load(load, entity_type.dimensions),
91            entity_type,
92        )
93    }
94
95    fn new_with_base(base: EntityBase, entity_type: EntityTypeRef) -> Self {
96        let living_base = LivingEntityBase::new(entity_type);
97        let mob_base = MobBase::new();
98        init_squid(&living_base, &mob_base);
99
100        let mut entity_data = GlowSquidEntityData::new();
101        living_base.initialize_synced_data(&mut entity_data);
102
103        Self {
104            base,
105            entity_type,
106            living_base,
107            mob_base,
108            ageable_base: AgeableMobBase::new(),
109            entity_data: SyncMutex::new(entity_data),
110            state: SyncMutex::new(SquidState::new()),
111        }
112    }
113
114    /// Ticks left until the glow squid lights up again.
115    #[must_use]
116    pub fn dark_ticks_remaining(&self) -> i32 {
117        *self.entity_data.lock().dark_ticks_remaining.get()
118    }
119
120    fn set_dark_ticks(&self, ticks: i32) {
121        self.entity_data.lock().dark_ticks_remaining.set(ticks);
122    }
123}

139impl Entity for GlowSquidEntity {
140    fn base(&self) -> &EntityBase {
141        &self.base
142    }
143
144    fn hurt(&self, world: &World, source: &DamageSource, amount: f32) -> bool {
145        let hurt = self.squid_hurt(world, source, amount);
146        if hurt {
147            self.set_dark_ticks(DARK_TICKS_AFTER_HURT);
148        }
149        hurt
150    }
151
152    fn movement_emission(&self) -> EntityMovementEmission {
153        EntityMovementEmission::Events
154    }
155
156    fn entity_type(&self) -> EntityTypeRef {
157        self.entity_type
158    }
159
160    fn base_tick(&self) {
161        let air_supply = self.air_supply();
162        Mob::base_tick_mob(self);
163        self.handle_squid_air_supply(air_supply);
164    }
165
166    fn get_default_gravity(&self) -> f64 {
167        SQUID_GRAVITY
168    }
169
170    fn dimensions_for_pose(&self, _pose: EntityPose) -> EntityDimensions {
171        let scale = LivingEntity::get_scale(self);
172        if AgeableMob::is_baby(self) {
173            SQUID_BABY_DIMENSIONS.scale(scale)
174        } else if self.entity_type.fixed {
175            self.entity_type.dimensions
176        } else {
177            self.entity_type.dimensions.scale(scale)
178        }
179    }
180
181    fn synced_data(&self) -> Option<&dyn EntitySyncedData> {
182        Some(&self.entity_data)
183    }
184
185    fn save_additional(&self, nbt: &mut owned::NbtCompound) {
186        self.save_mob(nbt);
187        self.save_ageable_mob(nbt);
188        nbt.insert("DarkTicksRemaining", self.dark_ticks_remaining());
189    }
190
191    fn load_additional(&self, nbt: borrow::NbtCompound<'_, '_>) {
192        self.load_mob(nbt);
193        self.load_ageable_mob(nbt);
194        self.set_dark_ticks(nbt.int("DarkTicksRemaining").unwrap_or(0));
195    }
196}

230        let result = self.squid_ai_step(entity);
231        let dark_ticks = self.dark_ticks_remaining();
232        if dark_ticks > 0 {
233            self.set_dark_ticks(dark_ticks - 1);
234        }
235        // Vanilla also calls `level.addParticle(GLOW, ..)`, which does nothing
236        // on a server level.
237        result
238    }
