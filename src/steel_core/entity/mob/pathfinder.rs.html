25pub(super) fn tick_path_navigation_target<M: Mob + ?Sized>(
26    mob: &M,
27    world: &Arc<World>,
28    game_time: i64,
29    can_update_path: bool,
30) {
31    let (target, speed_modifier, kind) = {
32        let mut navigation = mob.mob_base().navigation().lock();
33        let kind = navigation.kind();
34        let bounding_box = mob.bounding_box();
35        let temp_mob_position = match kind {
36            NavigationKind::Ground => {
37                ground_navigation_temp_mob_pos(mob, world.as_ref(), navigation.can_float())
38            }
39            NavigationKind::WaterBound { .. } => {
40                let position = mob.position();
41                DVec3::new(
42                    position.x,
43                    position.y + bounding_box.height() * 0.5,
44                    position.z,
45                )
46            }
47        };
48        let can_move_directly_to_next_node = match kind {
49            NavigationKind::Ground => false,
50            NavigationKind::WaterBound { .. } => navigation
51                .path()
52                .and_then(|path| path_move_target(path, bounding_box.width()))
53                .is_some_and(|next| {
54                    is_clear_for_movement_between(
55                        world.as_ref(),
56                        temp_mob_position,
57                        next,
58                        bounding_box.height(),
59                    )
60                }),
61        };
62        let context = NavigationTickContext {
63            ground_mob_position: temp_mob_position,
64            mob_position: mob.position(),
65            mob_bounding_box_width: bounding_box.width(),
66            mob_speed: mob.get_speed(),
67            game_time,
68            can_move_directly_to_next_node,
69        };
70        let next_target = if can_update_path {
71            navigation.next_move_target(context)
72        } else {
73            navigation.next_move_target_without_path_update(context, mob.on_ground())
74        };
75        let Some((target, speed_modifier)) = next_target else {
76            return;
77        };
78        (target, speed_modifier, kind)
79    };
80
81    let target_pos = BlockPos::containing(target.x, target.y, target.z);
82    let ground_y = if matches!(kind, NavigationKind::WaterBound { .. })
83        || world.get_block_state(target_pos.below()).is_air()
84    {
85        target.y
86    } else {
87        WalkNodeEvaluator::floor_level(world.as_ref(), target_pos)
88    };
89    mob.set_wanted_position(DVec3::new(target.x, ground_y, target.z), speed_modifier);
90}

92/// Vanilla `PathNavigation.isClearForMovementBetween` without fluids blocking.
93fn is_clear_for_movement_between(world: &World, start: DVec3, stop: DVec3, height: f64) -> bool {
94    let to = DVec3::new(stop.x, stop.y + height * 0.5, stop.z);
95    // TODO: vanilla clips with the mob's collision context; Steel's clip uses
96    // the empty context, which only differs for entity-dependent shapes.
97    world
98        .clip(start, to, ClipBlockShape::Collider, ClipFluid::None)
99        .miss
100}

102fn ground_navigation_temp_mob_pos<M: Mob + ?Sized>(
103    mob: &M,
104    world: &World,
105    can_float: bool,
106) -> DVec3 {
107    let position = mob.position();
108    DVec3::new(
109        position.x,
110        f64::from(ground_navigation_surface_y(mob, world, can_float)),
111        position.z,
112    )
113}

115fn ground_navigation_surface_y<M: Mob + ?Sized>(mob: &M, world: &World, can_float: bool) -> i32 {
116    if !mob.is_in_water() || !can_float {
117        return fast_floor(mob.position().y + 0.5);
118    }
119
120    let position = mob.position();
121    let block_y = mob.block_position().y();
122    let mut surface = block_y;
123    let mut state = world.get_block_state(BlockPos::containing(
124        position.x,
125        f64::from(surface),
126        position.z,
127    ));
128    let mut steps = 0;
129    while state.get_block() == &vanilla_blocks::WATER {
130        surface += 1;
131        state = world.get_block_state(BlockPos::containing(
132            position.x,
133            f64::from(surface),
134            position.z,
135        ));
136        steps += 1;
137        if steps > 16 {
138            return block_y;
139        }
140    }
141
142    surface
143}

152    /// Vanilla `getWalkTargetValue(pos)`: scores `pos` against the mob's own level.
153    fn get_walk_target_value(&self, pos: BlockPos) -> f32 {
154        self.level()
155            .map_or(0.0, |world| self.get_walk_target_value_in(pos, &*world))
156    }

158    /// Vanilla `getWalkTargetValue(pos, level)`: animals favor light, monsters
159    /// darkness. Override this one; spawn checks pass worldgen regions here.
160    fn get_walk_target_value_in(&self, pos: BlockPos, level: &dyn LevelReader) -> f32 {
161        if let Some(monster) = self.as_monster() {
162            monster.get_walk_target_value_monster(pos, level)
163        } else if let Some(animal) = self.as_animal() {
164            animal.animal_walk_target_value(pos, level)
165        } else {
166            0.0
167        }
168    }

170    fn can_update_path(&self) -> bool {
171        let kind = self.mob_base().navigation().lock().kind();
172        match kind {
173            NavigationKind::Ground => {
174                self.on_ground() || self.is_in_water() || self.is_in_lava() || self.is_passenger()
175            }
176            NavigationKind::WaterBound { allow_breaching } => {
177                allow_breaching || self.is_in_water() || self.is_in_lava()
178            }
179        }
180    }

182    fn can_path_to_targets_below_surface(&self) -> bool {
183        if let Some(vehicle) = self.controlled_pathfinder_vehicle()
184            && let Some(pathfinder) = vehicle.as_pathfinder_mob()
185        {
186            return pathfinder.can_path_to_targets_below_surface();
187        }
188
189        self.mob_base()
190            .navigation()
191            .lock()
192            .can_path_to_targets_below_surface()
193    }

195    fn can_reach_living_target(&self, target: &dyn LivingEntity) -> bool {
196        let target_pos = target.block_position();
197        self.create_path_to(target_pos, 0)
198            .is_some_and(|path| path_end_node_can_reach_target(&path, target_pos))
199    }

201    fn tick_pathfinder_path_navigation(&self) {
202        let Some(world) = self.level() else {
203            return;
204        };
205        let game_time = world.game_time();
206        // `can_update_path` reads the navigation kind, so it must run
207        // before the navigation lock is taken.
208        let can_update_path = self.can_update_path();
209        let recompute_request = {
210            let mut navigation = self.mob_base().navigation().lock();
211            if navigation.is_wall_climber() && navigation.is_done() {
212                // Vanilla `WallClimberNavigation.tick`: with no path left, steer
213                // straight at the remembered target instead of ticking the path.
214                let wanted = navigation
215                    .tick_wall_climber_without_path(self.position(), self.bounding_box().width());
216                drop(navigation);
217                if let Some((target, speed_modifier)) = wanted {
218                    self.set_wanted_position(target, speed_modifier);
219                }
220                return;
221            }
222            navigation.tick();
223            navigation.take_delayed_recompute_request(game_time, can_update_path)
224        };
225        if let Some(request) = recompute_request {
226            self.recompute_path(request);
227        }
228
229        tick_path_navigation_target(self, &world, game_time, self.can_update_path());
230    }

232    fn tick_pathfinder_goal_selectors(&self, entity: &SharedEntity)
233    where
234        Self: Sized,
235    {
236        let id_based_tick_count = self.tick_count().wrapping_add(self.id());
237        let mut target_selector = self.mob_base().target_selector().lock();
238        let mut goal_selector = self.mob_base().goal_selector().lock();
239        if id_based_tick_count % 2 != 0 && self.tick_count() > 1 {
240            target_selector.tick_running_goals(self, entity, false);
241            goal_selector.tick_running_goals(self, entity, false);
242        } else {
243            target_selector.tick(self, entity);
244            goal_selector.tick(self, entity);
245        }
246    }

248    fn is_stable_destination(&self, pos: BlockPos) -> bool {
249        let Some(world) = self.level() else {
250            return false;
251        };
252        let kind = self.mob_base().navigation().lock().kind();
253        match kind {
254            NavigationKind::Ground => world.get_block_state(pos.below()).is_solid_render(),
255            NavigationKind::WaterBound { .. } => !world.get_block_state(pos).is_solid_render(),
256        }
257    }

259    fn create_path_to(&self, target: BlockPos, reach_range: i32) -> Option<Path> {
260        if let Some(vehicle) = self.controlled_pathfinder_vehicle()
261            && let Some(pathfinder) = vehicle.as_pathfinder_mob()
262        {
263            return pathfinder.create_path_to(target, reach_range);
264        }
265
266        self.mob_base()
267            .navigation()
268            .lock()
269            .record_climber_path_to_position(target);
270        let world = self.level()?;
271        if !world.has_full_chunk(ChunkPos::from_block_pos(target)) {
272            return None;
273        }
274
275        let target = path_target_for_mob(self, world.as_ref(), target);
276        let targets = [target];
277        self.create_path_to_targets(&world, &targets, reach_range)
278    }

280    fn recompute_path(&self, request: NavigationRecomputeRequest) {
281        if let Some(vehicle) = self.controlled_pathfinder_vehicle()
282            && let Some(pathfinder) = vehicle.as_pathfinder_mob()
283        {
284            pathfinder.recompute_path(request);
285            return;
286        }
287
288        let path = self.create_path_to(request.target_pos, request.reach_range);
289        self.mob_base()
290            .navigation()
291            .lock()
292            .complete_recompute_path(path, request.game_time);
293    }

295    fn move_to_pos(&self, target: DVec3, speed_modifier: f64) -> bool {
296        self.move_to_pos_with_reach(target, 1, speed_modifier)
297    }

299    fn move_to_pos_with_reach(&self, target: DVec3, reach_range: i32, speed_modifier: f64) -> bool {
300        if let Some(vehicle) = self.controlled_pathfinder_vehicle()
301            && let Some(pathfinder) = vehicle.as_pathfinder_mob()
302        {
303            return pathfinder.move_to_pos_with_reach(target, reach_range, speed_modifier);
304        }
305
306        let target_pos = BlockPos::containing(target.x, target.y, target.z);
307        self.mob_base()
308            .navigation()
309            .lock()
310            .record_climber_path_to_position(target_pos);
311        let Some(world) = self.level() else {
312            self.mob_base().navigation().lock().stop();
313            return false;
314        };
315        if !world.has_full_chunk(ChunkPos::from_block_pos(target_pos)) {
316            self.mob_base().navigation().lock().stop();
317            return false;
318        }
319
320        let target_pos = path_target_for_mob(self, world.as_ref(), target_pos);
321        let targets = [target_pos];
322        if self
323            .mob_base()
324            .navigation()
325            .lock()
326            .reuse_current_path_to_targets(
327                world.as_ref(),
328                &targets,
329                speed_modifier,
330                self.position(),
331            )
332        {
333            return true;
334        }
335
336        let path = self.create_path_to_targets(&world, &targets, reach_range);
337        self.move_to_path(path, speed_modifier)
338    }

340    /// Vanilla `PathNavigation.moveTo(Entity, double)`.
341    ///
342    /// Ground navigation paths to the entity's block with reach 1. A wall climber
343    /// (`WallClimberNavigation.moveTo`) paths with reach 0 and, when no path is
344    /// found, still succeeds by steering straight at the entity.
345    fn move_to_entity(&self, target: &dyn Entity, speed_modifier: f64) -> bool {
346        if let Some(vehicle) = self.controlled_pathfinder_vehicle()
347            && let Some(pathfinder) = vehicle.as_pathfinder_mob()
348        {
349            return pathfinder.move_to_entity(target, speed_modifier);
350        }
351
352        if !self.mob_base().navigation().lock().is_wall_climber() {
353            return self.move_to_pos(target.position(), speed_modifier);
354        }
355
356        let target_pos = target.block_position();
357        if let Some(path) = self.create_path_to(target_pos, 0) {
358            return self.move_to_path(Some(path), speed_modifier);
359        }
360        self.mob_base()
361            .navigation()
362            .lock()
363            .move_climber_directly_to(target_pos, speed_modifier);
364        true
365    }

367    fn move_to_path(&self, path: Option<Path>, speed_modifier: f64) -> bool {
368        if let Some(vehicle) = self.controlled_pathfinder_vehicle()
369            && let Some(pathfinder) = vehicle.as_pathfinder_mob()
370        {
371            return pathfinder.move_to_path(path, speed_modifier);
372        }
373
374        let Some(world) = self.level() else {
375            self.mob_base().navigation().lock().stop();
376            return false;
377        };
378        let mut navigation = self.mob_base().navigation().lock();
379        let Some(path) = path else {
380            navigation.stop();
381            return false;
382        };
383
384        navigation.move_to(world.as_ref(), path, speed_modifier, self.position())
385    }

424    /// Re-enables MOVE and follows a nearby holder unless panicking.
425    ///
426    fn close_range_leash_behaviour_pathfinder(&self, holder: &dyn Entity) {
427        if !self.should_stay_close_to_leash_holder() || self.is_panicking() {
428            return;
429        }
430
431        self.mob_base()
432            .goal_selector()
433            .lock()
434            .enable_control(GoalControl::Move);
435        let distance_to = (holder.position() - self.position()).length();
436        let delta = (holder.position() - self.position()).normalize_or_zero()
437            * (distance_to - LEASH_CLOSE_RANGE_WANTED_DISTANCE).max(0.0);
438        self.move_to_pos(self.position() + delta, self.follow_leash_speed());
439    }

461    fn create_path_to_targets(
462        &self,
463        world: &Arc<World>,
464        targets: &[BlockPos],
465        reach_range: i32,
466    ) -> Option<Path> {
467        if let Some(vehicle) = self.controlled_pathfinder_vehicle()
468            && let Some(pathfinder) = vehicle.as_pathfinder_mob()
469        {
470            return pathfinder.create_path_to_targets(world, targets, reach_range);
471        }
472
473        if targets.is_empty()
474            || self.position().y < f64::from(world.min_y())
475            || !self.can_update_path()
476        {
477            return None;
478        }
479
480        let follow_range = self
481            .attributes()
482            .lock()
483            .required_value(vanilla_attributes::FOLLOW_RANGE);
484        let max_path_length = {
485            let mut navigation = self.mob_base().navigation().lock();
486            navigation.update_pathfinder_max_visited_nodes(follow_range);
487            navigation.max_path_length(follow_range)
488        };
489
490        let mob_position = self.block_position();
491        let settings = MobPathSettings::from_mob(self);
492        let kind = self.mob_base().navigation().lock().kind();
493        let request = NavigationPathRequest {
494            mob_position,
495            targets,
496            max_path_length,
497            reach_range,
498        };
499        if let NavigationKind::WaterBound { allow_breaching } = kind {
500            let mut evaluator = SwimNodeEvaluator::new(settings, allow_breaching);
501            let mut no_collision = |_| false;
502            return self.mob_base().navigation().lock().create_path(
503                &mut evaluator,
504                world.as_ref(),
505                &mut no_collision,
506                request,
507            );
508        }
509
510        let mut evaluator = WalkNodeEvaluator::new(settings);
511        let collision_world =
512            WorldCollisionProvider::for_path_navigation(world, self.as_entity_event_source());
513        let mut collision = |aabb| {
514            collision_world.has_entity_context_collision(
515                aabb,
516                self.position().y,
517                self.is_descending(),
518            )
519        };
520
521        self.mob_base().navigation().lock().create_path(
522            &mut evaluator,
523            world.as_ref(),
524            &mut collision,
525            request,
526        )
527    }

530pub(super) fn path_end_node_can_reach_target(path: &Path, target: BlockPos) -> bool {
531    let Some(end_node) = path.end_node() else {
532        return false;
533    };
534    let dx = end_node.x - target.x();
535    let dz = end_node.z - target.z();
536    f64::from(dx * dx + dz * dz) <= TARGET_REACH_DISTANCE_SQR
537}

539fn path_target_for_mob<M: PathfinderMob + ?Sized, L: LevelReader>(
540    mob: &M,
541    level: &L,
542    target: BlockPos,
543) -> BlockPos {
544    // Only `GroundPathNavigation.createPath` moves the target to the surface.
545    if mob.can_path_to_targets_below_surface()
546        || !mob.mob_base().navigation().lock().can_navigate_ground()
547    {
548        target
549    } else {
550        find_ground_path_target_surface(level, target)
551    }
552}

554pub(super) fn find_ground_path_target_surface(
555    level: &impl LevelReader,
556    mut pos: BlockPos,
557) -> BlockPos {
558    if level.get_block_state(pos).is_air() {
559        let mut column_pos = pos.below();
560        while column_pos.y() >= level.min_y() && level.get_block_state(column_pos).is_air() {
561            column_pos = column_pos.below();
562        }
563        if column_pos.y() >= level.min_y() {
564            return column_pos.above();
565        }
566
567        column_pos = pos.at_y(pos.y() + 1);
568        while column_pos.y() < level.max_y_exclusive() && level.get_block_state(column_pos).is_air()
569        {
570            column_pos = column_pos.above();
571        }
572        pos = column_pos;
573    }
574
575    if !level.get_block_state(pos).is_solid() {
576        return pos;
577    }
578
579    let mut column_pos = pos.above();
580    while column_pos.y() < level.max_y_exclusive() && level.get_block_state(column_pos).is_solid() {
581        column_pos = column_pos.above();
582    }
583    column_pos
584}
