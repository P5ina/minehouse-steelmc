1//! Multiblock shapes matched in any orientation (Vanilla `BlockPattern`).
2//!
3//! Golems and the wither are built by placing their head on a shape of blocks;
4//! the shape may face any direction and be tilted onto any side.

16/// Test for one cell of a pattern. `None` matches anything, even an unloaded block
17/// (Vanilla's `' '` predicate).
18pub(crate) type CellPredicate = Option<BlockPredicate>;
19
20/// Test for a block state in a pattern cell.
21pub(crate) type BlockPredicate = fn(BlockStateId) -> bool;
22
23/// A 3D grid of block predicates, indexed `[depth][height][width]`.
24pub(crate) struct BlockPattern {
25    pattern: Vec<Vec<Vec<CellPredicate>>>,
26    depth: i32,
27    height: i32,
28    width: i32,
29}
30
31impl BlockPattern {
32    /// Builds a pattern from aisles of rows, like Vanilla `BlockPatternBuilder`.
33    ///
34    /// Each aisle is one depth slice listed top row first. `' '` matches anything;
35    /// every other character must have a predicate in `lookup`.
36    ///
37    /// # Panics
38    /// Panics if the aisles are ragged or a character has no predicate, as the
39    /// Vanilla builder throws; patterns are constants, so this is a programming error.
40    #[must_use]
41    pub(crate) fn new(aisles: &[&[&str]], lookup: &[(char, BlockPredicate)]) -> Self {
42        let height = aisles[0].len();
43        let width = aisles[0][0].len();
44        let pattern = aisles
45            .iter()
46            .map(|aisle| {
47                assert_eq!(aisle.len(), height, "aisle height mismatch");
48                aisle
49                    .iter()
50                    .map(|row| {
51                        assert_eq!(row.len(), width, "aisle row width mismatch");
52                        row.chars()
53                            .map(|c| {
54                                if c == ' ' {
55                                    return None;
56                                }
57                                let predicate = lookup
58                                    .iter()
59                                    .find(|(key, _)| *key == c)
60                                    .map(|(_, predicate)| *predicate);
61                                assert!(predicate.is_some(), "no predicate for {c:?}");
62                                predicate
63                            })
64                            .collect()
65                    })
66                    .collect()
67            })
68            .collect();
69        Self {
70            pattern,
71            depth: aisles.len() as i32,
72            height: height as i32,
73            width: width as i32,
74        }
75    }
76
77    /// Finds this pattern with any orientation among the positions within
78    /// `max(width, height, depth)` of `origin` in the positive axes.
79    pub(crate) fn find(&self, world: &World, origin: BlockPos) -> Option<BlockPatternMatch> {
80        let dist = self.width.max(self.height).max(self.depth);
81        for test_pos in
82            BlockPos::between_closed(origin, origin.offset(dist - 1, dist - 1, dist - 1))
83        {
84            for forwards in Direction::ALL {
85                for up in Direction::ALL {
86                    if up == forwards || up == forwards.opposite() {
87                        continue;
88                    }
89                    if let Some(found) = self.matches(world, test_pos, forwards, up) {
90                        return Some(found);
91                    }
92                }
93            }
94        }
95        None
96    }
97
98    fn matches(
99        &self,
100        world: &World,
101        origin: BlockPos,
102        forwards: Direction,
103        up: Direction,
104    ) -> Option<BlockPatternMatch> {
105        for x in 0..self.width {
106            for y in 0..self.height {
107                for z in 0..self.depth {
108                    let Some(predicate) = self.pattern[z as usize][y as usize][x as usize] else {
109                        continue;
110                    };
111                    let pos = translate_and_rotate(origin, forwards, up, x, y, z);
112                    // Vanilla `BlockInWorld.hasState` fails on unloaded blocks.
113                    let state = world.loaded_block_state(pos)?;
114                    if !predicate(state) {
115                        return None;
116                    }
117                }
118            }
119        }
120        Some(BlockPatternMatch {
121            front_top_left: origin,
122            forwards,
123            up,
124            width: self.width,
125            height: self.height,
126        })
127    }
128}

130/// Where a pattern matched.
131pub(crate) struct BlockPatternMatch {
132    front_top_left: BlockPos,
133    forwards: Direction,
134    up: Direction,
135    width: i32,
136    height: i32,
137}

139impl BlockPatternMatch {
140    /// The world position of pattern cell (`right`, `down`, `forwards`).
141    #[must_use]
142    pub(crate) fn block(&self, right: i32, down: i32, forwards: i32) -> BlockPos {
143        translate_and_rotate(
144            self.front_top_left,
145            self.forwards,
146            self.up,
147            right,
148            down,
149            forwards,
150        )
151    }
152
153    /// Vanilla `CarvedPumpkinBlock.clearPatternBlocks`: replaces the front slice
154    /// with air, with block-break particles.
155    pub(crate) fn clear_blocks(&self, world: &Arc<World>) {
156        for x in 0..self.width {
157            for y in 0..self.height {
158                let pos = self.block(x, y, 0);
159                let state = world.get_block_state(pos);
160                world.set_block(
161                    pos,
162                    vanilla_blocks::AIR.default_state(),
163                    UpdateFlags::UPDATE_CLIENTS,
164                );
165                world.level_event(
166                    level_events::PARTICLES_DESTROY_BLOCK,
167                    pos,
168                    level_events::encode_block_state_data(u32::from(state.0)),
169                    None,
170                );
171            }
172        }
173    }
174
175    /// Vanilla `CarvedPumpkinBlock.updatePatternBlocks`.
176    pub(crate) fn update_blocks(&self, world: &Arc<World>) {
177        for x in 0..self.width {
178            for y in 0..self.height {
179                world.update_neighbors_at(self.block(x, y, 0), &vanilla_blocks::AIR);
180            }
181        }
182    }
183}

185fn translate_and_rotate(
186    origin: BlockPos,
187    forwards: Direction,
188    up: Direction,
189    right: i32,
190    down: i32,
191    forward: i32,
192) -> BlockPos {
193    let forwards_vec: IVec3 = forwards.offset_vec();
194    let up_vec: IVec3 = up.offset_vec();
195    let right_vec = forwards_vec.cross(up_vec);
196    let offset = up_vec * -down + right_vec * right + forwards_vec * forward;
197    origin.offset(offset.x, offset.y, offset.z)
198}

200#[cfg(test)]
201mod tests {
202    use steel_utils::{BlockPos, Direction};
203
204    use super::translate_and_rotate;
205
206    #[test]
207    fn rotation_follows_vanilla_cross_product() {
208        // Facing north with up = up, "right" is east (north × up = -z × y = +x).
209        let origin = BlockPos::new(0, 0, 0);
210        assert_eq!(
211            translate_and_rotate(origin, Direction::North, Direction::Up, 1, 0, 0),
212            BlockPos::new(1, 0, 0)
213        );
214        assert_eq!(
215            translate_and_rotate(origin, Direction::North, Direction::Up, 0, 2, 0),
216            BlockPos::new(0, -2, 0)
217        );
218        // Lying down: forwards = up, up = south, so "down" runs north.
219        assert_eq!(
220            translate_and_rotate(origin, Direction::Up, Direction::South, 0, 1, 0),
221            BlockPos::new(0, 0, -1)
222        );
223    }
