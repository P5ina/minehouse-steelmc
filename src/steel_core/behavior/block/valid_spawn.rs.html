98/// Vanilla `BlockStateBase.isValidSpawn`: the block's registered predicate,
99/// or the `Properties` default of a sturdy top face and light emission
100/// below [`MAX_MOB_SPAWN_LIGHT_EMISSION`].
101pub(crate) fn properties_is_valid_spawn(
102    state: BlockStateId,
103    world: &dyn LevelReader,
104    pos: BlockPos,
105    entity_type: EntityTypeRef,
106) -> bool {
107    let block = state.get_block();
108    let is_any = |blocks: &[BlockRef]| blocks.contains(&block);
109    if is_any(&NEVER) {
110        false
111    } else if is_any(&ALWAYS) {
112        true
113    } else if is_any(&OCELOT_OR_PARROT) {
114        entity_type == &vanilla_entities::OCELOT || entity_type == &vanilla_entities::PARROT
115    } else if block == &vanilla_blocks::ICE || block == &vanilla_blocks::FROSTED_ICE {
116        entity_type == &vanilla_entities::POLAR_BEAR
117    } else if block == &vanilla_blocks::MAGMA_BLOCK {
118        entity_type.fire_immune
119    } else {
120        world.is_face_sturdy_for(state, pos, Direction::Up, SupportType::Full)
121            && state.get_light_emission() < MAX_MOB_SPAWN_LIGHT_EMISSION
122    }
123}
