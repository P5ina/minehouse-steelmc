32/// The block entities a chest resolves to.
33pub enum ChestCombineResult {
34    /// The chest is blocked or has no block entity.
35    None,
36    /// A chest without a usable partner.
37    Single(SharedBlockEntity),
38    /// Both halves of a double chest, `RIGHT` half first.
39    Double(SharedBlockEntity, SharedBlockEntity),
40}
41
42impl ChestCombineResult {
43    /// The containers in the order Vanilla's `CompoundContainer` addresses them.
44    #[must_use]
45    pub fn containers(&self) -> Vec<ContainerRef> {
46        match self {
47            Self::None => Vec::new(),
48            Self::Single(chest) => chest.container_ref().into_iter().collect(),
49            Self::Double(first, second) => match (first.container_ref(), second.container_ref()) {
50                (Some(first), Some(second)) => vec![first, second],
51                _ => Vec::new(),
52            },
53        }
54    }
55}

57/// Shared chest logic for [`ChestBlock`] and [`TrappedChestBlock`].
58struct ChestBehavior {
59    block: BlockRef,
60    block_entity_type: BlockEntityTypeRef,
61    open_stat: CustomStatRef,
62}
63
64impl ChestBehavior {
65    fn get_state_for_placement(&self, context: &BlockPlaceContext<'_>) -> BlockStateId {
66        let mut chest_type = ChestType::Single;
67        let mut facing = context.horizontal_direction().opposite();
68        let secondary_use = context.is_secondary_use_active();
69        let clicked_face = context.clicked_face();
70        let world = context.world;
71        let pos = context.place_pos();
72
73        if clicked_face.is_horizontal()
74            && secondary_use
75            && let Some(partner_facing) =
76                self.candidate_partner_facing(world.as_ref(), pos, clicked_face.opposite())
77            && partner_facing.axis() != clicked_face.axis()
78        {
79            facing = partner_facing;
80            chest_type = if facing.rotate_y_counter_clockwise() == clicked_face.opposite() {
81                ChestType::Right
82            } else {
83                ChestType::Left
84            };
85        }
86
87        if chest_type == ChestType::Single && !secondary_use {
88            chest_type = self.chest_type(world.as_ref(), pos, facing);
89        }
90
91        self.block
92            .default_state()
93            .set_value(&BlockStateProperties::FACING, facing)
94            .set_value(&BlockStateProperties::CHEST_TYPE, chest_type)
95            .set_value(
96                &BlockStateProperties::WATERLOGGED,
97                context.is_water_source(),
98            )
99    }
100
101    fn chest_type(&self, world: &dyn LevelReader, pos: BlockPos, facing: Direction) -> ChestType {
102        if self.candidate_partner_facing(world, pos, facing.rotate_y_clockwise()) == Some(facing) {
103            ChestType::Left
104        } else if self.candidate_partner_facing(world, pos, facing.rotate_y_counter_clockwise())
105            == Some(facing)
106        {
107            ChestType::Right
108        } else {
109            ChestType::Single
110        }
111    }
112
113    /// The facing of a single chest of this kind next to `pos`, if there is one.
114    fn candidate_partner_facing(
115        &self,
116        world: &dyn LevelReader,
117        pos: BlockPos,
118        direction: Direction,
119    ) -> Option<Direction> {
120        let state = world.get_block_state(pos.relative(direction));
121        (self.chest_can_connect_to(state)
122            && state.get_value(&BlockStateProperties::CHEST_TYPE) == ChestType::Single)
123            .then(|| state.get_value(&BlockStateProperties::FACING))
124    }
125
126    fn chest_can_connect_to(&self, state: BlockStateId) -> bool {
127        state.get_block() == self.block
128    }
129
130    fn update_shape(
131        &self,
132        state: BlockStateId,
133        world: &dyn ScheduledTickAccess,
134        pos: BlockPos,
135        direction: Direction,
136        neighbor_state: BlockStateId,
137    ) -> BlockStateId {
138        schedule_water_tick_if_waterlogged(state, world, pos);
139        if self.chest_can_connect_to(neighbor_state) && direction.is_horizontal() {
140            let neighbor_type = neighbor_state.get_value(&BlockStateProperties::CHEST_TYPE);
141            if state.get_value(&BlockStateProperties::CHEST_TYPE) == ChestType::Single
142                && neighbor_type != ChestType::Single
143                && state.get_value(&BlockStateProperties::FACING)
144                    == neighbor_state.get_value(&BlockStateProperties::FACING)
145                && connected_direction(neighbor_state) == direction.opposite()
146            {
147                let chest_type = match neighbor_type {
148                    ChestType::Left => ChestType::Right,
149                    ChestType::Right | ChestType::Single => ChestType::Left,
150                };
151                return state.set_value(&BlockStateProperties::CHEST_TYPE, chest_type);
152            }
153        } else if connected_direction(state) == direction {
154            return state.set_value(&BlockStateProperties::CHEST_TYPE, ChestType::Single);
155        }
156        state
157    }
158
159    /// Resolves this chest and its partner. Vanilla `ChestBlock.combine`.
160    fn combine(
161        &self,
162        state: BlockStateId,
163        world: &World,
164        pos: BlockPos,
165        ignore_blocked: bool,
166    ) -> ChestCombineResult {
167        let is_blocked = |pos| !ignore_blocked && is_chest_blocked_at(world, pos);
168        let Some(chest) = self.chest_block_entity(world, pos) else {
169            return ChestCombineResult::None;
170        };
171        if is_blocked(pos) {
172            return ChestCombineResult::None;
173        }
174        let chest_type = state.get_value(&BlockStateProperties::CHEST_TYPE);
175        if chest_type == ChestType::Single {
176            return ChestCombineResult::Single(chest);
177        }
178
179        let neighbor_pos = pos.relative(connected_direction(state));
180        let neighbor_state = world.get_block_state(neighbor_pos);
181        if neighbor_state.get_block() == state.get_block() {
182            let neighbor_type = neighbor_state.get_value(&BlockStateProperties::CHEST_TYPE);
183            if neighbor_type != ChestType::Single
184                && neighbor_type != chest_type
185                && neighbor_state.get_value(&BlockStateProperties::FACING)
186                    == state.get_value(&BlockStateProperties::FACING)
187            {
188                if is_blocked(neighbor_pos) {
189                    return ChestCombineResult::None;
190                }
191                if let Some(neighbor) = self.chest_block_entity(world, neighbor_pos) {
192                    return if chest_type == ChestType::Right {
193                        ChestCombineResult::Double(chest, neighbor)
194                    } else {
195                        ChestCombineResult::Double(neighbor, chest)
196                    };
197                }
198            }
199        }
200        ChestCombineResult::Single(chest)
201    }
202
203    fn chest_block_entity(&self, world: &World, pos: BlockPos) -> Option<SharedBlockEntity> {
204        world
205            .get_block_entity(pos)
206            .filter(|entity| entity.get_type() == self.block_entity_type)
207    }
208
209    fn use_without_item(&self, state: BlockStateId, world: &World, pos: BlockPos, player: &Player) {
210        let combined = self.combine(state, world, pos, false);
211        let (title, rows) = match &combined {
212            ChestCombineResult::None => return,
213            ChestCombineResult::Single(chest) => (
214                custom_name(chest).unwrap_or_else(|| {
215                    TextComponent::translated(translations::CONTAINER_CHEST.msg())
216                }),
217                3,
218            ),
219            ChestCombineResult::Double(first, second) => (
220                custom_name(first)
221                    .or_else(|| custom_name(second))
222                    .unwrap_or_else(|| {
223                        TextComponent::translated(translations::CONTAINER_CHEST_DOUBLE.msg())
224                    }),
225                6,
226            ),
227        };
228        let containers = combined.containers();
229        if containers.is_empty() {
230            return;
231        }
232
233        let inventory = Arc::clone(&player.inventory);
234        player.open_menu(title, move |context| {
235            chest_halves(inventory, context.container_id, &containers, rows)
236        });
237        player.award_custom_stat(self.open_stat);
238        // TODO: Anger nearby piglins (PiglinAi.angerNearbyPiglins)
239        // TODO: Count viewers (ContainerOpenersCounter) for the lid, sounds and trapped chest power.
240    }
241
242    fn get_analog_output_signal(&self, state: BlockStateId, world: &World, pos: BlockPos) -> i32 {
243        let containers = self.combine(state, world, pos, false).containers();
244        if containers.is_empty() {
245            return 0;
246        }
247        let guard = ContainerLockGuard::lock_all(&containers);
248        let locked: Vec<_> = containers
249            .iter()
250            .filter_map(|container| guard.get(container.container_id()))
251            .collect();
252        calculate_redstone_signal_from_containers(&locked)
253    }
254
255    fn new_block_entity(
256        &self,
257        level: Weak<World>,
258        pos: BlockPos,
259        state: BlockStateId,
260    ) -> BlockEntityCreation {
261        BlockEntityCreation::from_registered_factory(BLOCK_ENTITIES.create(
262            self.block_entity_type,
263            level,
264            pos,
265            state,
266        ))
267    }
268}

270/// The direction from a chest half to its partner.
271#[must_use]
272pub fn connected_direction(state: BlockStateId) -> Direction {
273    let facing: Direction = state.get_value(&BlockStateProperties::FACING);
274    if state.get_value(&BlockStateProperties::CHEST_TYPE) == ChestType::Left {
275        facing.rotate_y_clockwise()
276    } else {
277        facing.rotate_y_counter_clockwise()
278    }
279}

281/// Whether a chest at `pos` cannot be opened.
282#[must_use]
283pub fn is_chest_blocked_at(world: &World, pos: BlockPos) -> bool {
284    let above = pos.above();
285    // TODO: Cats sitting on the chest block it too (isCatSittingOnChest); Steel has no cats yet.
286    is_redstone_conductor(world, world.get_block_state(above), above)
287}

289fn custom_name(chest: &SharedBlockEntity) -> Option<TextComponent> {
290    chest
291        .downcast_ref::<ChestBlockEntity>()
292        .and_then(ChestBlockEntity::custom_name)

307macro_rules! impl_chest_block {
308    ($name:ident, $entity_type:ident, $stat:ident) => {
309        impl $name {
310            /// Creates the chest behavior for `block`.
311            #[must_use]
312            pub const fn new(block: BlockRef) -> Self {
313                Self { block }
314            }
315
316            const fn chest(&self) -> ChestBehavior {
317                ChestBehavior {
318                    block: self.block,
319                    block_entity_type: &vanilla_block_entity_types::$entity_type,
320                    open_stat: &vanilla_custom_stats::$stat,
321                }
322            }
323
324            /// Resolves this chest and its partner, Vanilla `ChestBlock.combine`.
325            #[must_use]
326            pub fn combine(
327                &self,
328                state: BlockStateId,
329                world: &World,
330                pos: BlockPos,
331                ignore_blocked: bool,
332            ) -> ChestCombineResult {
333                self.chest().combine(state, world, pos, ignore_blocked)
334            }
335        }
336
337        impl BlockBehavior for $name {
338            fn get_state_for_placement(
339                &self,
340                context: &BlockPlaceContext<'_>,
341            ) -> Option<BlockStateId> {
342                Some(self.chest().get_state_for_placement(context))
343            }
344
345            fn update_shape(
346                &self,
347                state: BlockStateId,
348                world: &dyn ScheduledTickAccess,
349                pos: BlockPos,
350                direction: Direction,
351                _neighbor_pos: BlockPos,
352                neighbor_state: BlockStateId,
353            ) -> BlockStateId {
354                self.chest()
355                    .update_shape(state, world, pos, direction, neighbor_state)
356            }
357
358            fn use_without_item(
359                &self,
360                state: BlockStateId,
361                world: &Arc<World>,
362                pos: BlockPos,
363                player: &Player,
364                _hit_result: &BlockHitResult,
365                _inv: &mut InventoryAccess,
366            ) -> InteractionResult {
367                self.chest().use_without_item(state, world, pos, player);
368                InteractionResult::Success
369            }
370
371            fn affect_neighbors_after_removal(
372                &self,
373                _state: BlockStateId,
374                world: &Arc<World>,
375                pos: BlockPos,
376                _moved_by_piston: bool,
377            ) {
378                world.update_neighbor_for_output_signal(pos, self.chest().block);
379            }
380
381            fn new_block_entity(
382                &self,
383                level: Weak<World>,
384                pos: BlockPos,
385                state: BlockStateId,
386            ) -> BlockEntityCreation {
387                self.chest().new_block_entity(level, pos, state)
388            }
389
390            fn has_analog_output_signal(&self, _state: BlockStateId) -> bool {
391                true
392            }
393
394            fn get_analog_output_signal(
395                &self,
396                state: BlockStateId,
397                world: &World,
398                pos: BlockPos,
399                _direction: Direction,
400            ) -> i32 {
401                self.chest().get_analog_output_signal(state, world, pos)
402            }
403        }
404    };
405}

410/// The containers of the chest at `pos`, `RIGHT` half first, or `None` if the
411/// block is not a chest. Vanilla `ChestBlock.getContainer`.
412#[must_use]
413pub fn chest_container(
414    state: BlockStateId,
415    world: &World,
416    pos: BlockPos,
417    ignore_blocked: bool,
418) -> Option<Vec<ContainerRef>> {
419    let block = state.get_block();
420    let chest = if block == &vanilla_blocks::CHEST {
421        ChestBlock::new(block).chest()
422    } else if block == &vanilla_blocks::TRAPPED_CHEST {
423        TrappedChestBlock::new(block).chest()
424    } else {
425        return None;
426    };
427    Some(
428        chest
429            .combine(state, world, pos, ignore_blocked)
430            .containers(),
431    )
