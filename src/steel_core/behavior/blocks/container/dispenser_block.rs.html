33/// Ticks between a redstone pulse and the dispense.
34const TRIGGER_DELAY: i32 = 4;
35
36/// Dispenser block behavior.
37#[block_behavior]
38pub struct DispenserBlock {
39    block: BlockRef,
40}

42/// Dropper block behavior: like a dispenser, but always drops or inserts one item.
43#[block_behavior]
44pub struct DropperBlock {
45    block: BlockRef,
46}

48/// What differs between a dispenser and a dropper.
49#[derive(Clone, Copy)]
50struct Kind {
51    block: BlockRef,
52    entity_type: BlockEntityTypeRef,
53    title: &'static Translation<0>,
54    inspect_stat: CustomStatRef,
55    dropper: bool,
56}
57
58impl Kind {
59    fn placement(self, context: &BlockPlaceContext<'_>) -> BlockStateId {
60        self.block.default_state().set_value(
61            &BlockStateProperties::FACING,
62            context.get_nearest_looking_direction().opposite(),
63        )
64    }
65
66    /// Vanilla `DispenserBlock.neighborChanged`: the dispenser or the block above
67    /// powered starts a dispense in 4 ticks.
68    fn neighbor_changed(self, state: BlockStateId, world: &Arc<World>, pos: BlockPos) {
69        let powered = world.has_neighbor_signal(pos) || world.has_neighbor_signal(pos.above());
70        let triggered = state.get_value(&BlockStateProperties::TRIGGERED);
71        if powered && !triggered {
72            world.schedule_block_tick_default(pos, self.block, TRIGGER_DELAY);
73            world.set_block(
74                pos,
75                state.set_value(&BlockStateProperties::TRIGGERED, true),
76                UpdateFlags::UPDATE_CLIENTS,
77            );
78        } else if !powered && triggered {
79            world.set_block(
80                pos,
81                state.set_value(&BlockStateProperties::TRIGGERED, false),
82                UpdateFlags::UPDATE_CLIENTS,
83            );
84        }
85    }
86
87    fn dispense_from(self, state: BlockStateId, world: &Arc<World>, pos: BlockPos) {
88        let Some(block_entity) = world.get_block_entity(pos) else {
89            log::warn!("Ignoring dispensing attempt without a block entity at {pos:?}");
90            return;
91        };
92        let Some(entity) = block_entity.downcast_ref::<DispenserBlockEntity>() else {
93            log::warn!("Ignoring dispensing attempt without a dispenser block entity at {pos:?}");
94            return;
95        };
96        let source = BlockSource {
97            world,
98            pos,
99            state,
100            entity,
101        };
102        let Some(slot) = entity.get_random_slot(|bound| rand::random_range(0..bound)) else {
103            world.level_event(level_events::SOUND_DISPENSER_FAIL, pos, 0, None);
104            if !self.dropper {
105                world.game_event(
106                    &vanilla_game_events::BLOCK_ACTIVATE,
107                    pos,
108                    &GameEventContext::new(None, Some(state)),
109                );
110            }
111            return;
112        };
113        let stack = entity.get_item(slot);
114        let rest = if self.dropper {
115            drop_into_container_or_world(&source, &block_entity, stack)
116        } else {
117            DispenseBehavior::for_item(&stack).dispense(&source, stack)
118        };
119        entity.set_item(slot, rest);
120    }
121
122    fn open(self, world: &World, pos: BlockPos, player: &Player) -> InteractionResult {
123        let Some(block_entity) = world.get_block_entity(pos) else {
124            return InteractionResult::Success;
125        };
126        let Some(entity) = block_entity.downcast_ref::<DispenserBlockEntity>() else {
127            return InteractionResult::Success;
128        };
129        let Some(container) = block_entity.container_ref() else {
130            return InteractionResult::Success;
131        };
132        let title = entity
133            .custom_name()
134            .unwrap_or_else(|| TextComponent::translated(self.title.msg()));
135        let inventory = Arc::clone(&player.inventory);
136        player.open_menu(title, move |context| {
137            dispenser(inventory, context.container_id, container)
138        });
139        player.award_custom_stat(self.inspect_stat);
140        InteractionResult::Success
141    }
142
143    fn new_block_entity(
144        self,
145        level: Weak<World>,
146        pos: BlockPos,
147        state: BlockStateId,
148    ) -> BlockEntityCreation {
149        BlockEntityCreation::from_registered_factory(BLOCK_ENTITIES.create(
150            self.entity_type,
151            level,
152            pos,
153            state,
154        ))
155    }
156}

158/// Vanilla `DropperBlock.dispenseFrom`: insert one item into the container in
159/// front, or drop it when there is none.
160fn drop_into_container_or_world(
161    source: &BlockSource<'_>,
162    block_entity: &SharedBlockEntity,
163    stack: ItemStack,
164) -> ItemStack {
165    if stack.is_empty() {
166        return stack;
167    }
168    let facing = source.facing();
169    let Some(targets) =
170        HopperBlockEntity::get_container_at(source.world, source.pos.relative(facing))
171    else {
172        return default_dispense(source, stack);
173    };
174    let Some(own) = block_entity.container_ref() else {
175        return stack;
176    };
177    let source_id = own.container_id();
178    let target_ids: Vec<ContainerId> = targets.iter().map(ContainerRef::container_id).collect();
179    let mut locked = vec![own];
180    locked.extend(targets);
181    let rest = {
182        let mut guard = ContainerLockGuard::lock_all(&locked);
183        add_item(
184            &mut guard,
185            Some(source_id),
186            &target_ids,
187            stack.copy_with_count(1),
188            Some(facing.opposite()),
189        )
190    };
191    let mut result = stack;
192    if rest.is_empty() {
193        result.shrink(1);
194    }
195    result
196}

198fn analog_output(world: &World, pos: BlockPos) -> i32 {
199    let Some(container) = world
200        .get_block_entity(pos)
201        .and_then(ContainerRef::from_block_entity)
202    else {
203        return 0;
204    };
205    let guard = ContainerLockGuard::lock_all(&[&container]);
206    guard.get(container.container_id()).map_or(0, |container| {
207        calculate_redstone_signal_from_container(container)
208    })
209}

211macro_rules! impl_dispenser_block {
212    ($name:ident, $entity:ident, $title:ident, $stat:ident, $dropper:expr) => {
213        impl $name {
214            /// Creates the block behavior.
215            #[must_use]
216            pub const fn new(block: BlockRef) -> Self {
217                Self { block }
218            }
219
220            fn kind(&self) -> Kind {
221                Kind {
222                    block: self.block,
223                    entity_type: &vanilla_block_entity_types::$entity,
224                    title: &translations::$title,
225                    inspect_stat: &vanilla_custom_stats::$stat,
226                    dropper: $dropper,
227                }
228            }
229        }
230
231        impl BlockBehavior for $name {
232            fn get_state_for_placement(
233                &self,
234                context: &BlockPlaceContext<'_>,
235            ) -> Option<BlockStateId> {
236                Some(self.kind().placement(context))
237            }
238
239            fn handle_neighbor_changed(
240                &self,
241                state: BlockStateId,
242                world: &Arc<World>,
243                pos: BlockPos,
244                _source_block: BlockRef,
245                _moved_by_piston: bool,
246            ) {
247                self.kind().neighbor_changed(state, world, pos);
248            }
249
250            fn tick(&self, state: BlockStateId, world: &Arc<World>, pos: BlockPos) {
251                self.kind().dispense_from(state, world, pos);
252            }
253
254            fn use_without_item(
255                &self,
256                _state: BlockStateId,
257                world: &Arc<World>,
258                pos: BlockPos,
259                player: &Player,
260                _hit_result: &BlockHitResult,
261                _inv: &mut InventoryAccess,
262            ) -> InteractionResult {
263                self.kind().open(world, pos, player)
264            }
265
266            fn affect_neighbors_after_removal(
267                &self,
268                _state: BlockStateId,
269                world: &Arc<World>,
270                pos: BlockPos,
271                _moved_by_piston: bool,
272            ) {
273                world.update_neighbor_for_output_signal(pos, self.block);
274            }
275
276            fn new_block_entity(
277                &self,
278                level: Weak<World>,
279                pos: BlockPos,
280                state: BlockStateId,
281            ) -> BlockEntityCreation {
282                self.kind().new_block_entity(level, pos, state)
283            }
284
285            fn has_analog_output_signal(&self, _state: BlockStateId) -> bool {
286                true
287            }
288
289            fn get_analog_output_signal(
290                &self,
291                _state: BlockStateId,
292                world: &World,
293                pos: BlockPos,
294                _direction: Direction,
295            ) -> i32 {
296                analog_output(world, pos)
297            }
298        }
299    };
300}
