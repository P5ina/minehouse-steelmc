23/// Vanilla `InfestedBlock`.
24#[block_behavior]
25pub struct InfestedBlock {
26    block: BlockRef,
27    #[json_arg(vanilla_blocks, json = "host_block")]
28    host_block: BlockRef,
29}
30
31impl InfestedBlock {
32    /// Creates an infested block that imitates `host_block`.
33    #[must_use]
34    pub const fn new(block: BlockRef, host_block: BlockRef) -> Self {
35        Self { block, host_block }
36    }
37}
38
39impl BlockBehavior for InfestedBlock {
40    fn get_state_for_placement(&self, _context: &BlockPlaceContext<'_>) -> Option<BlockStateId> {
41        Some(self.block.default_state())
42    }
43
44    fn spawn_after_break(
45        &self,
46        _state: BlockStateId,
47        world: &Arc<World>,
48        pos: BlockPos,
49        tool: &ItemStack,

52        spawn_after_break_infested(world, pos, tool);

55    fn infested_host_block(&self) -> Option<BlockRef> {
56        Some(self.host_block)

97/// Vanilla `InfestedBlock.spawnAfterBreak`: a silverfish crawls out unless
98/// block drops are off or the tool prevents it (silk touch).
99fn spawn_after_break_infested(world: &Arc<World>, pos: BlockPos, tool: &ItemStack) {
100    if world.get_game_rule(&BLOCK_DROPS) && !prevents_infested_spawns(tool) {
101        spawn_infestation(world, pos);
102    }
103}
104
105/// Vanilla `EnchantmentHelper.hasTag(tool, PREVENTS_INFESTED_SPAWNS)`.
106fn prevents_infested_spawns(tool: &ItemStack) -> bool {
107    let Some(enchantments) = tool.get_enchantments() else {
108        return false;
109    };
110    enchantments.iter().any(|(key, _)| {
111        REGISTRY
112            .enchantments
113            .by_key(key)
114            .is_some_and(|enchantment| {
115                REGISTRY
116                    .enchantments
117                    .is_in_tag(enchantment, &EnchantmentTag::PREVENTS_INFESTED_SPAWNS)
118            })
119    })
120}
121
122/// Vanilla `InfestedBlock.spawnInfestation`.
123fn spawn_infestation(world: &Arc<World>, pos: BlockPos) {
124    let position = DVec3::new(
125        f64::from(pos.x()) + 0.5,
126        f64::from(pos.y()),
127        f64::from(pos.z()) + 0.5,
128    );
129    let Ok(silverfish) = create_entity_instance(world, &vanilla_entities::SILVERFISH, position)
130    else {
131        return;
132    };
133    silverfish.set_rotation((0.0, 0.0));
134    if world.try_add_entity(Arc::clone(&silverfish)).is_ok() {
135        silverfish.broadcast_entity_event(EntityStatus::SilverfishMergeAnim);
136    }
137}

139/// Host and infested block pairs, read from the registered behaviors.
140fn infested_pairs() -> &'static [(BlockRef, BlockRef)] {
141    static PAIRS: OnceLock<Vec<(BlockRef, BlockRef)>> = OnceLock::new();
142    PAIRS.get_or_init(|| {
143        REGISTRY
144            .blocks
145            .iter()
146            .filter_map(|(_, infested)| {
147                BLOCK_BEHAVIORS
148                    .get_behavior(infested)
149                    .infested_host_block()
150                    .map(|host| (host, infested))
151            })
152            .collect()
153    })
154}

156/// Vanilla `InfestedBlock.isCompatibleHostBlock`.
157#[must_use]
158pub(crate) fn is_compatible_host_block(state: BlockStateId) -> bool {
159    let block = state.get_block();
160    infested_pairs()
161        .iter()
162        .any(|(host, _)| ptr::eq(*host, block))
163}
164
165/// Vanilla `InfestedBlock.infestedStateByHost`, keeping matching properties.
166#[must_use]
167pub(crate) fn infested_state_by_host(host_state: BlockStateId) -> Option<BlockStateId> {
168    let block = host_state.get_block();
169    infested_pairs()
170        .iter()
171        .find(|(host, _)| ptr::eq(*host, block))
172        .map(|(_, infested)| {
173            REGISTRY
174                .blocks
175                .copy_matching_properties(host_state, infested)
176        })
177}
178
179/// Vanilla `InfestedBlock.hostStateByInfested`; `None` for blocks that are
180/// not infested.
181#[must_use]
182pub(crate) fn host_state_by_infested(infested_state: BlockStateId) -> Option<BlockStateId> {
183    BLOCK_BEHAVIORS
184        .get_behavior(infested_state.get_block())
185        .infested_host_block()
186        .map(|host| {
187            REGISTRY
188                .blocks
189                .copy_matching_properties(infested_state, host)
190        })
191}

205    fn silverfish_near(world: &Arc<World>, pos: BlockPos) -> usize {
206        let bounds = WorldAabb::new(
207            f64::from(pos.x()) - 1.0,
208            f64::from(pos.y()) - 1.0,
209            f64::from(pos.z()) - 1.0,
210            f64::from(pos.x()) + 2.0,
211            f64::from(pos.y()) + 2.0,
212            f64::from(pos.z()) + 2.0,
213        );
214        world
215            .get_entities_in_aabb(&bounds)
216            .iter()
217            .filter(|entity| entity.entity_type() == &vanilla_entities::SILVERFISH)
218            .count()
219    }

221    #[test]
222    fn host_and_infested_states_map_both_ways() {
223        init_vanilla_registry();
224        init_behaviors();
225        let stone = vanilla_blocks::STONE.default_state();
226        assert!(is_compatible_host_block(stone));
227        assert!(!is_compatible_host_block(
228            vanilla_blocks::DIRT.default_state()
229        ));
230        assert_eq!(
231            infested_state_by_host(stone),
232            Some(vanilla_blocks::INFESTED_STONE.default_state())
233        );
234        assert_eq!(
235            host_state_by_infested(vanilla_blocks::INFESTED_STONE.default_state()),
236            Some(stone)
237        );
238        assert_eq!(host_state_by_infested(stone), None);
239
240        let deepslate = vanilla_blocks::DEEPSLATE
241            .default_state()
242            .set_value(RotatedPillarBlock::AXIS, Axis::X);
243        let infested = infested_state_by_host(deepslate).expect("deepslate hosts silverfish");
244        assert_eq!(infested.get_block(), &vanilla_blocks::INFESTED_DEEPSLATE);
245        assert_eq!(infested.get_value(RotatedPillarBlock::AXIS), Axis::X);
246    }

248    #[test]
249    fn breaking_an_infested_block_releases_a_silverfish_without_silk_touch() {
250        init_vanilla_registry();
251        init_behaviors();
252        init_entities();
253        let world_fixture = fresh_test_world("infested_block_break");
254        let world = &world_fixture.world;
255        let pos = BlockPos::new(8, 64, 8);
256        insert_ready_full_chunk(world, ChunkPos::from_block_pos(pos));
257        let state = vanilla_blocks::INFESTED_STONE.default_state();
258        let behavior = BLOCK_BEHAVIORS.get_behavior(&vanilla_blocks::INFESTED_STONE);
259
260        let mut silk_touch = ItemStack::new(&vanilla_items::DIAMOND_PICKAXE);
261        silk_touch.set_enchantments(&[(Identifier::vanilla_static("silk_touch"), 1)], false);
262        behavior.spawn_after_break(state, world, pos, &silk_touch, true);
263        assert_eq!(silverfish_near(world, pos), 0, "silk touch keeps it inside");
264
265        let pickaxe = ItemStack::new(&vanilla_items::DIAMOND_PICKAXE);
266        behavior.spawn_after_break(state, world, pos, &pickaxe, true);
267        assert_eq!(silverfish_near(world, pos), 1, "a silverfish crawls out");
268    }
