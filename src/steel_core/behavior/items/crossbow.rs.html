95/// Vanilla `CrossbowItem.tryLoadProjectiles`: draws ammo for `shooter` and
96/// stores it in `crossbow`. Returns whether anything was loaded.
97pub(crate) fn try_load_projectiles(
98    world: &Arc<World>,
99    shooter: LivingEntityRef<'_>,
100    crossbow: &mut ItemStack,
101) -> bool {
102    let drawn = draw_ammo(world, shooter, crossbow);
103    if drawn.is_empty() {
104        return false;
105    }
106    // Vanilla `ChargedProjectiles.ofNonEmpty` / `ItemStackTemplate.fromNonEmptyStack`.
107    let templates = drawn
108        .iter()
109        .filter_map(|stack| ItemStackTemplate::from_stack(stack).ok())
110        .collect();
111    match ChargedProjectiles::new(templates) {
112        Ok(projectiles) => {
113            crossbow.set(CHARGED_PROJECTILES, projectiles);
114            true
115        }
116        Err(error) => {
117            log::warn!("failed to charge crossbow: {error}");
118            false
119        }
120    }
121}

123/// Vanilla `ProjectileWeaponItem.draw(heldItem, shooter.getProjectile(heldItem), shooter)`.
124fn draw_ammo(
125    world: &Arc<World>,
126    shooter: LivingEntityRef<'_>,
127    crossbow: &ItemStack,
128) -> Vec<ItemStack> {
129    let entity = shooter.entity();
130    if let Some(player) = world.players.get_by_entity_id(entity.id()) {
131        let Some((slot, projectile)) = find_player_projectile(&player, is_arrow_or_firework) else {
132            return Vec::new();
133        };
134        return draw_player_ammo(&player, crossbow, slot, &projectile);
135    }
136    draw_mob_ammo(shooter.living(), crossbow)
137}

139/// Vanilla `Monster.getProjectile` + `ProjectileWeaponItem.draw` for a mob:
140/// a held arrow or firework (taken from the hand), else a fresh arrow.
141fn draw_mob_ammo(shooter: &dyn LivingEntity, crossbow: &ItemStack) -> Vec<ItemStack> {
142    let infinite = shooter.has_infinite_materials();
143    let num_projectiles = enchantment_helper::process_projectile_count(crossbow, shooter, 1);
144    let mut equipment = shooter.living_base().equipment().lock();
145    let held_slot = [EquipmentSlot::OffHand, EquipmentSlot::MainHand]
146        .into_iter()
147        .find(|&slot| is_arrow_or_firework(equipment.get_ref(slot)));
148    let mut fresh_arrow = ItemStack::new(&vanilla_items::ARROW);
149    let projectile = match held_slot {
150        Some(slot) => equipment.get_mut(slot),
151        None => &mut fresh_arrow,
152    };
153    if projectile.is_empty() {
154        return Vec::new();
155    }
156    let projectile_copy = projectile.clone();
157    let mut drawn = Vec::with_capacity(usize::try_from(num_projectiles).unwrap_or(0));
158    for index in 0..num_projectiles {
159        let stack = if index == 0 {
160            use_ammo(crossbow, projectile, false, infinite)
161        } else {
162            use_ammo(crossbow, &mut projectile_copy.clone(), true, infinite)
163        };
164        if !stack.is_empty() {
165            drawn.push(stack);
166        }
167    }
168    drawn
169}

171/// Vanilla `CrossbowItem.performShooting`: fires everything loaded in
172/// `weapon` (at `target` when given, else along the shooter's view) and
173/// empties it. Returns whether anything was fired.
174#[expect(
175    clippy::too_many_arguments,
176    reason = "mirrors vanilla CrossbowItem.performShooting"
177)]
178pub(crate) fn perform_shooting(
179    world: &Arc<World>,
180    shooter: &SharedEntity,
181    weapon: &mut ItemStack,
182    power: f32,
183    uncertainty: f32,
184    target: Option<&SharedEntity>,
185    is_player: bool,
186    has_infinite_materials: bool,
187) -> bool {
188    let Some(charged) = weapon.get(CHARGED_PROJECTILES).cloned() else {
189        return false;
190    };
191    weapon.set(CHARGED_PROJECTILES, ChargedProjectiles::empty());
192    if charged.items().is_empty() {
193        return false;
194    }
195    let projectiles: Vec<ItemStack> = charged
196        .items()
197        .iter()
198        .map(ItemStackTemplate::create)
199        .collect();
200    shoot(
201        world,
202        shooter,
203        weapon,
204        &projectiles,
205        power,
206        uncertainty,
207        is_player,
208        target,
209        has_infinite_materials,
210    );
211    true
212}

214/// Vanilla `ProjectileWeaponItem.shoot` with the crossbow's projectile and
215/// aim overrides.
216#[expect(
217    clippy::too_many_arguments,
218    reason = "mirrors vanilla ProjectileWeaponItem.shoot"
219)]
220fn shoot(
221    world: &Arc<World>,
222    shooter: &SharedEntity,
223    weapon: &mut ItemStack,
224    projectiles: &[ItemStack],
225    power: f32,
226    uncertainty: f32,
227    is_crit: bool,
228    target: Option<&SharedEntity>,
229    has_infinite_materials: bool,
230) {
231    let max_angle = enchantment_helper::process_projectile_spread(weapon, shooter.as_ref(), 0.0);
232    let count = projectiles.len();
233    let angle_step = if count == 1 {
234        0.0
235    } else {
236        2.0 * max_angle / (count - 1) as f32
237    };
238    let angle_offset = ((count as f32 - 1.0) % 2.0) * angle_step / 2.0;
239    let mut direction = 1.0_f32;
240    for (index, projectile) in projectiles.iter().enumerate() {
241        if projectile.is_empty() {
242            continue;
243        }
244        let angle = angle_offset + direction * index.div_ceil(2) as f32 * angle_step;
245        direction = -direction;
246        let projectile_entity = create_projectile(world, shooter, weapon, projectile, is_crit);
247        shoot_projectile(
248            world,
249            shooter,
250            &projectile_entity,
251            index,
252            power,
253            uncertainty,
254            angle,
255            target,
256        );
257        spawn_projectile(world, &projectile_entity, projectile, shooter);
258        weapon.hurt_and_break(get_durability_use(projectile), has_infinite_materials);
259        if weapon.is_empty() {
260            break;
261        }
262    }
263}

274/// Vanilla `CrossbowItem.createProjectile`.
275fn create_projectile(
276    world: &Arc<World>,
277    shooter: &SharedEntity,
278    weapon: &ItemStack,
279    projectile: &ItemStack,
280    is_crit: bool,
281) -> SharedEntity {
282    if projectile.is(&vanilla_items::FIREWORK_ROCKET) {
283        let position = shooter.position();
284        let position = DVec3::new(
285            position.x,
286            shooter.get_eye_y() - FIREWORK_EYE_OFFSET,
287            position.z,
288        );
289        let world_ref: Weak<World> = Arc::downgrade(world);
290        let rocket = FireworkRocketEntity::launched(
291            &vanilla_entities::FIREWORK_ROCKET,
292            next_entity_id(),
293            position,
294            world_ref,
295            projectile.clone(),
296        );
297        rocket.set_owner_entity(Some(shooter));
298        rocket.set_shot_at_angle(true);
299        return Arc::new(rocket);
300    }
301    let arrow = create_arrow(world, projectile, shooter, Some(weapon));
302    if let Some(abstract_arrow) = as_abstract_arrow(&arrow) {
303        if is_crit {
304            abstract_arrow.set_crit_arrow(true);
305        }
306        abstract_arrow.set_sound_event(&sound_events::ITEM_CROSSBOW_HIT);
307    }
308    arrow
309}

311/// Vanilla `Projectile.spawnProjectile` tail: arrows also run their weapon's
312/// spawn effects, anything else only the ammo's.
313fn spawn_projectile(
314    world: &Arc<World>,
315    projectile_entity: &SharedEntity,
316    ammo: &ItemStack,
317    owner: &SharedEntity,
318) {
319    if as_abstract_arrow(projectile_entity).is_some() {
320        spawn_arrow(world, projectile_entity, ammo, Some(owner));
321        return;
322    }
323    if let Err(error) = world.try_add_entity(Arc::clone(projectile_entity)) {
324        log::debug!("failed to spawn crossbow projectile: {error}");
325        return;
326    }
327    let mut ammo = ammo.clone();
328    enchantment_helper::on_projectile_spawned(
329        world,
330        &mut ammo,
331        projectile_entity.as_ref(),
332        Some(owner),
333    );
334}

336/// Vanilla `CrossbowItem.shootProjectile`.
337#[expect(
338    clippy::too_many_arguments,
339    reason = "mirrors vanilla CrossbowItem.shootProjectile"
340)]
341fn shoot_projectile(
342    world: &Arc<World>,
343    shooter: &SharedEntity,
344    projectile_entity: &SharedEntity,
345    index: usize,
346    power: f32,
347    uncertainty: f32,
348    angle: f32,
349    target: Option<&SharedEntity>,
350) {
351    let shot_vector = if let Some(target) = target {
352        let shooter_position = shooter.position();
353        let target_position = target.position();
354        let xd = target_position.x - shooter_position.x;
355        let zd = target_position.z - shooter_position.z;
356        let distance_to_target = (xd * xd + zd * zd).sqrt();
357        // Vanilla `targetOverride.getY(1/3)`.
358        let target_y = target_position.y
359            + f64::from(target.bounding_box().height() as f32) * 0.333_333_333_333_333_3;
360        let yd =
361            target_y - projectile_entity.position().y + distance_to_target * f64::from(0.2_f32);
362        get_projectile_shot_vector(shooter.as_ref(), DVec3::new(xd, yd, zd), angle)
363    } else {
364        let up_vector = up_vector(shooter.as_ref()).as_vec3();
365        let view_vector = shooter.look_angle().as_vec3();
366        // Vanilla `new Quaternionf().setAngleAxis(angle, up)` then `rotate`.
367        quaternion_rotate(view_vector, angle.to_radians(), up_vector)
368    };
369    if let Some(projectile) = projectile_entity.as_projectile() {
370        projectile.shoot(shot_vector.as_dvec3(), power, uncertainty);
371    }
372    let pitch = get_shot_pitch(index);
373    world.play_sound_at(
374        &sound_events::ITEM_CROSSBOW_SHOOT,
375        shooter.sound_source(),
376        shooter.position(),
377        1.0,
378        pitch,
379        None,
380    );
381}

389/// Vanilla `CrossbowItem.getProjectileShotVector`.
390fn get_projectile_shot_vector(body: &dyn Entity, original_vector: DVec3, angle: f32) -> Vec3 {
391    let view_vec = original_vector.as_vec3().normalize();
392    let mut right_vector_pre_rot = view_vec.cross(Vec3::Y);
393    if right_vector_pre_rot.length_squared() <= 1.0E-7 {
394        right_vector_pre_rot = view_vec.cross(up_vector(body).as_vec3());
395    }
396    let view_vec_up = joml_rotate_axis(view_vec, FRAC_PI_2, right_vector_pre_rot);
397    joml_rotate_axis(view_vec, angle.to_radians(), view_vec_up)
398}

400/// JOML `Math.cosFromSin(sin, angle)`.
401fn cos_from_sin(sin: f32, angle: f32) -> f32 {
402    let cos = (1.0 - sin * sin).sqrt();
403    let tau = TAU;
404    let a = angle + FRAC_PI_2;
405    let mut b = a - (a / tau).trunc() * tau;
406    if b < 0.0 {
407        b += tau;
408    }
409    if b >= PI { -cos } else { cos }
410}

412/// JOML `Vector3f.rotateAxis(angle, x, y, z)`. The axis is not normalized,
413/// exactly like JOML, which matters for the unnormalized cross product
414/// vanilla passes in.
415fn joml_rotate_axis(vector: Vec3, angle: f32, axis: Vec3) -> Vec3 {
416    let half_angle = angle * 0.5;
417    let sin_angle = half_angle.sin();
418    let qx = axis.x * sin_angle;
419    let qy = axis.y * sin_angle;
420    let qz = axis.z * sin_angle;
421    let qw = cos_from_sin(sin_angle, half_angle);
422    let w2 = qw * qw;
423    let x2 = qx * qx;
424    let y2 = qy * qy;
425    let z2 = qz * qz;
426    let zw = qz * qw;
427    let xy = qx * qy;
428    let xz = qx * qz;
429    let yw = qy * qw;
430    let yz = qy * qz;
431    let xw = qx * qw;
432    let (x, y, z) = (vector.x, vector.y, vector.z);
433    Vec3::new(
434        (w2 + x2 - z2 - y2) * x + (-zw + xy - zw + xy) * y + (yw + xz + xz + yw) * z,
435        (xy + zw + zw + xy) * x + (y2 - z2 + w2 - x2) * y + (yz + yz - xw - xw) * z,
436        (xz - yw + xz - yw) * x + (yz + yz + xw + xw) * y + (z2 - y2 - x2 + w2) * z,
437    )
438}

440/// JOML `Quaternionf.setAngleAxis(angle, axis)` followed by
441/// `Vector3f.rotate(quaternion)` (which normalizes by the quaternion length).
442fn quaternion_rotate(vector: Vec3, angle: f32, axis: Vec3) -> Vec3 {
443    let sin = (angle * 0.5).sin();
444    let qx = axis.x * sin;
445    let qy = axis.y * sin;
446    let qz = axis.z * sin;
447    let qw = cos_from_sin(sin, angle * 0.5);
448    let xx = qx * qx;
449    let yy = qy * qy;
450    let zz = qz * qz;
451    let ww = qw * qw;
452    let xy = qx * qy;
453    let xz = qx * qz;
454    let yz = qy * qz;
455    let xw = qx * qw;
456    let zw = qz * qw;
457    let yw = qy * qw;
458    let k = 1.0 / (xx + yy + zz + ww);
459    let (x, y, z) = (vector.x, vector.y, vector.z);
460    // JOML `Math.fma` is a plain multiply-add unless `joml.useMathFma` is set.
461    Vec3::new(
462        (xx - yy - zz + ww) * k * x + (2.0 * (xy - zw) * k * y + 2.0 * (xz + yw) * k * z),
463        2.0 * (xy + zw) * k * x + ((yy - xx - zz + ww) * k * y + 2.0 * (yz - xw) * k * z),
464        2.0 * (xz - yw) * k * x + (2.0 * (yz + xw) * k * y + (zz - xx - yy + ww) * k * z),
465    )
466}

468/// Vanilla `CrossbowItem.getShotPitch`.
469fn get_shot_pitch(index: usize) -> f32 {
470    if index == 0 {
471        1.0
472    } else {
473        let range_decider = if index & 1 == 1 { 0.63 } else { 0.43 };
474        1.0 / (rand::random::<f32>() * 0.5 + 1.8) + range_decider
475    }
476}

489impl ItemBehavior for CrossbowItem {
490    fn use_item(&self, context: &mut UseItemContext) -> InteractionResult {
491        let charged = context
492            .inv
493            .with_item(|stack| stack.get(CHARGED_PROJECTILES).cloned());
494        if let Some(charged) = charged
495            && !charged.items().is_empty()
496        {
497            let power = get_shooting_power(&charged);
498            let shooter: SharedEntity = Arc::clone(context.player) as SharedEntity;
499            let infinite = context.player.has_infinite_materials();
500            let mut weapon = context.inv.with_item(|stack| stack.clone());
501            let fired = perform_shooting(
502                context.world,
503                &shooter,
504                &mut weapon,
505                power,
506                1.0,
507                None,
508                true,
509                infinite,
510            );
511            let item = weapon.item();
512            context.inv.with_item(|stack| *stack = weapon);
513            if fired {
514                // TODO: `CriteriaTriggers.SHOT_CROSSBOW` once advancement triggers exist.
515                context
516                    .player
517                    .award_stat(&vanilla_stat_types::ITEM_USED, item);
518            }
519            return InteractionResult::Consume;
520        }
521        if find_player_projectile(context.player, is_arrow_or_firework).is_some() {
522            START_SOUND_PLAYED.store(false, Ordering::Relaxed);
523            MID_LOAD_SOUND_PLAYED.store(false, Ordering::Relaxed);
524            context.player.start_using_item(context.hand);
525            return InteractionResult::Consume;
526        }
527        InteractionResult::Fail
528    }

531        ItemUseAnimation::Crossbow
532    }

538    fn on_use_tick(
539        &self,
540        world: &Arc<World>,
541        user: LivingEntityRef<'_>,
542        stack: &mut ItemStack,
543        ticks_remaining: i32,
544    ) {
545        let entity = user.entity().as_ref();
546        let sounds = enchantment_helper::pick_crossbow_charging_sounds(stack);
547        let start = sounds.map_or(Some(&sound_events::ITEM_CROSSBOW_LOADING_START), |s| {
548            s.start
549        });
550        let mid = sounds.map_or(Some(&sound_events::ITEM_CROSSBOW_LOADING_MIDDLE), |s| s.mid);
551        let end = sounds.map_or(Some(&sound_events::ITEM_CROSSBOW_LOADING_END), |s| s.end);
552        let tick_percent = (self.get_use_duration(stack, user.living()) - ticks_remaining) as f32
553            / get_charge_duration(stack) as f32;
554        if tick_percent < START_SOUND_PERCENT {
555            START_SOUND_PLAYED.store(false, Ordering::Relaxed);
556            MID_LOAD_SOUND_PLAYED.store(false, Ordering::Relaxed);
557        }
558        if tick_percent >= START_SOUND_PERCENT
559            && !START_SOUND_PLAYED.swap(true, Ordering::Relaxed)
560            && let Some(sound) = start
561        {
562            play_charging_sound(world, entity, sound, SoundSource::Players, 1.0, 0.5);
563        }
564        if tick_percent >= MID_SOUND_PERCENT
565            && !MID_LOAD_SOUND_PLAYED.swap(true, Ordering::Relaxed)
566            && let Some(sound) = mid
567        {
568            play_charging_sound(world, entity, sound, SoundSource::Players, 1.0, 0.5);
569        }
570        if tick_percent >= 1.0
571            && !is_charged(stack)
572            && try_load_projectiles(world, user, stack)
573            && let Some(sound) = end
574        {
575            let pitch = 1.0 / (rand::random::<f32>() * 0.5 + 1.0) + 0.2;
576            play_charging_sound(world, entity, sound, entity.sound_source(), pitch, 1.0);
577        }
578    }

580    fn release_using(
581        &self,
582        stack: &mut ItemStack,
583        _world: &Arc<World>,
584        user: &dyn LivingEntity,
585        time_left: i32,
586    ) -> bool {
587        let time_held = self.get_use_duration(stack, user) - time_left;
588        get_power_for_time(time_held, stack) >= 1.0 && is_charged(stack)
589    }
590
591    fn use_on_release(&self, stack: &ItemStack) -> bool {
592        stack.is(&vanilla_items::CROSSBOW)
593    }
