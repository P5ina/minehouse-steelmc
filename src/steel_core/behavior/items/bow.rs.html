1//! Bow item behavior (vanilla `BowItem`).
2
3use std::sync::Arc;
4
5use steel_macros::item_behavior;
6use steel_protocol::packets::game::SoundSource;
7use steel_registry::item_stack::ItemStack;
8use steel_registry::sound_events;
9use steel_registry::stat::vanilla_stat_types;

11use super::projectile_weapon::{draw_player_ammo, find_player_arrow, shoot_from_rotation};
12use crate::behavior::{InteractionResult, ItemBehavior, ItemUseAnimation, UseItemContext};
13use crate::entity::{Entity, LivingEntity, SharedEntity};
14use crate::world::World;
15
16/// Vanilla `BowItem.getUseDuration`.
17const USE_DURATION: i32 = 72000;
18/// Vanilla `BowItem.releaseUsing` arrow speed at full draw.
19const FULL_DRAW_POWER: f32 = 3.0;
20/// Vanilla `BowItem.releaseUsing` minimum draw that still fires.
21const MIN_DRAW_POWER: f32 = 0.1;
22
23/// Vanilla `BowItem.getPowerForTime`: draw strength after `time_held` ticks.
24#[must_use]
25pub fn get_power_for_time(time_held: i32) -> f32 {
26    let pow = time_held as f32 / 20.0;
27    ((pow * pow + pow * 2.0) / 3.0).min(1.0)
28}
29
30/// The bow.
31#[item_behavior(class = "BowItem")]
32pub struct BowItem;
33
34impl ItemBehavior for BowItem {
35    fn use_item(&self, context: &mut UseItemContext) -> InteractionResult {
36        let found_projectile = find_player_arrow(context.player).is_some();
37        if !context.player.has_infinite_materials() && !found_projectile {
38            return InteractionResult::Fail;
39        }
40        context.player.start_using_item(context.hand);
41        InteractionResult::Consume
42    }
43
44    fn get_use_animation(&self, _stack: &ItemStack) -> ItemUseAnimation {
45        ItemUseAnimation::Bow
46    }
47
48    fn get_use_duration(&self, _stack: &ItemStack, _user: &dyn LivingEntity) -> i32 {
49        USE_DURATION
50    }
51
52    fn release_using(
53        &self,
54        stack: &mut ItemStack,
55        world: &Arc<World>,
56        user: &dyn LivingEntity,
57        time_left: i32,
58    ) -> bool {
59        let Some(player) = world.players.get_by_entity_id(user.id()) else {
60            return false;
61        };
62        let Some((slot, projectile)) = find_player_arrow(&player) else {
63            return false;
64        };
65        let time_held = self.get_use_duration(stack, user) - time_left;
66        let pow = get_power_for_time(time_held);
67        if pow < MIN_DRAW_POWER {
68            return false;
69        }
70
71        let fired = draw_player_ammo(&player, stack, slot, &projectile);
72        if !fired.is_empty() {
73            let shooter: SharedEntity = Arc::clone(&player) as SharedEntity;
74            shoot_from_rotation(
75                world,
76                &shooter,
77                stack,
78                &fired,
79                pow * FULL_DRAW_POWER,
80                1.0,
81                pow >= 1.0,
82                player.has_infinite_materials(),
83            );
84        }
85
86        world.play_sound_at(
87            &sound_events::ENTITY_ARROW_SHOOT,
88            SoundSource::Players,
89            player.position(),
90            1.0,
91            1.0 / (rand::random::<f32>() * 0.4 + 1.2) + pow * 0.5,
92            None,
93        );
94        player.award_stat(&vanilla_stat_types::ITEM_USED, stack.item());
95        true
96    }
97}
